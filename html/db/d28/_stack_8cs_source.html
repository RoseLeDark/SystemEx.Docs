154 public Stack(int size) {
155 m_elements = new T[size];
156 m_end = 0;
157 m_start = m_elements.Length - 1;
158 m_current = size-1;

160 for (int i = 0; i < m_stackLayer.Length; i++) {
161 m_stackLayer[i] = new StackLayer(m_end, m_start, m_current);
162 m_stackLayer[i].Enable = false;
163 m_stackLayer[i].EndMarker = m_end;
164 m_stackLayer[i].StartMarker = m_start;

214 public bool Push(T item, uint LayerID) {
215 if ( LayerID >= 4 ) return false;
216 if ( !m_stackLayer[LayerID].Enable ) return false;
217
218 if ( m_stackLayer[LayerID].Current == m_stackLayer[LayerID].EndMarker )
219 return false;
220
221 m_elements[m_stackLayer[LayerID].Current] = item;
222 m_stackLayer[LayerID].Current--;
223 return true;
224 }

253 public int PushRange(T[] items, uint LayerID = 100) {
254 if ( LayerID >= 4 ) return -1;
255 if ( items == null ) return -1;
256
257 bool layered = LayerID <= m_stackLayer.Length;
258 int i = 0;
259
260 for ( i = 0; i < items.Length; i++ ) {
261 if ( layered ) {
262 if ( !Push(items[i], LayerID) ) break;
263 } else {
264 if ( !Push(items[i]) ) break;
265 }
266 }
267
268 return i;
269 }
270
277 public T[]? PopRange(int size, uint LayerID = 100) {
278 if ( LayerID >= 4 ) return null;
279
280 List<T> popped = new List<T>();
281 T? item;
282
283 bool layered = LayerID <= m_stackLayer.Length;
284
285 for ( int i = 0; i < size; i++ ) {
286 if ( layered ) {
287 if ( !Pop(out item, LayerID) ) break;
288 } else {
289 if ( !Pop(out item) ) break;
290 }
291
292 if ( item != null ) popped.Add(item);
293 }
294
295 return popped.ToArray();
296 }

300 public bool Pop(out T? item, uint LayerID) {
301 if ( LayerID >= 4 ) { item = default; return false; }
302 if ( (m_stackLayer[LayerID].Enable != true) ) { item = default; return false; }
303 // IsEmpty
304 if ( m_stackLayer[LayerID].Current == m_stackLayer[LayerID].StartMarker - 1 ) { item = default; return false; }
305 item = m_elements[m_stackLayer[LayerID].Current];
306
307 m_stackLayer[LayerID].Current++;
308
309 return true;
310
311 }

322
323 return true;
