153 public Quatf ( Vec3f angles ) {
154 float cos_z_2 = System.MathF.Cos(0.5f*angles.Z);
155 float cos_y_2 = System.MathF.Cos(0.5f*angles.Y);
156 float cos_x_2 = System.MathF.Cos(0.5f*angles.X);
157
158 float sin_z_2 = System.MathF.Sin(0.5f*angles.Z);
159 float sin_y_2 = System.MathF.Sin(0.5f*angles.Y);
160 float sin_x_2 = System.MathF.Sin(0.5f*angles.X);
161
162 // and now compute quaternion
163 m_s = cos_z_2 * cos_y_2 * cos_x_2 + sin_z_2 * sin_y_2 * sin_x_2;
164 m_v = new Vec3f
165 (
166 cos_z_2 * cos_y_2 * sin_x_2 - sin_z_2 * sin_y_2 * cos_x_2,
167 cos_z_2 * sin_y_2 * cos_x_2 + sin_z_2 * cos_y_2 * sin_x_2,
168 sin_z_2 * cos_y_2 * cos_x_2 - cos_z_2 * sin_y_2 * sin_x_2
169 );
170 m_comparebleMode = CompareType.Skalar;
171 }

241 public static Quatf FromAxis ( AxisAngle<Vec3f, float> axisAngle ) {
242
243 float len = Vec3f.LenghtSqrt(axisAngle.Axis);
244 Quatf temp = new Quatf();
245
246 if ( len > 1e-12f ) {
247 float inv = 1.0f / len;
248
249 // Achse normalisieren
250 float nx = axisAngle.Axis.X * inv;
251 float ny = axisAngle.Axis.Y * inv;
252 float nz = axisAngle.Axis.Z * inv;
253
254 float omega = 0.5f * axisAngle.Angle;
255 float s = MathF.Sin(omega);
256
257 temp.m_v = new Vec3f(s * nx, s * ny, s * nz);
258 temp.m_s = MathF.Cos(omega);
259 } else {
260 // Ungültige Achse → Identity
261 temp.m_s = 1.0f;
262 temp.m_v = Vec3f.Zero;
263 }
264
265 return Normalize(temp);
266

351 public static Quatf Power ( Quatf v, float Exp ) {
352 if ( System.MathF.Abs(v.m_s) > .9999f ) {
353 return v;
354 }
355
356 // Erhalte halbe angle alpha (alpha = theta/2)
357 float alpha = System.MathF.Acos(v.m_s);
358
359 // Berechne neuen alpha Wert
360 float newAlpha = alpha * Exp;
361
362 // Berechne neuen s wert
363 Quatf result = new Quatf();
364 result.m_s = System.MathF.Cos(newAlpha);
365
366 // Berechne neue xyz Werte
367
368 float mult = System.MathF.Sin(newAlpha) / System.MathF.Sin(alpha);
369 result.m_v.X = v.m_v.X * mult;
370 result.m_v.Y = v.m_v.Y * mult;
371 result.m_v.Z = v.m_v.Z * mult;
372
373 return Normalize(result);
374 }

438 public static Quatf Slerp ( Quatf q0, Quatf q1, float t ) { // Spherical linear interpolation.
439 if ( t <= 0.0f ) return q0;
440 if ( t >= 1.0f ) return q1;
441
442 float cosOmega = DotProduct(q0, q1);
443
444 float q1w = q1.m_s;
445 float q1x = q1.m_v.X;
446 float q1y = q1.m_v.Y;
447 float q1z = q1.m_v.Z;
448
449 if ( cosOmega < 0.0f ) {
450 q1w = -q1w;
451 q1x = -q1x;
452 q1y = -q1y;
453 q1z = -q1z;
454 cosOmega = -cosOmega;
455 }
456
457 Debug.Assert(cosOmega < 1.0f);
458
459 // Compute interpolation fraction, checking for quaternions
460 // almost exactly the same
461
462 float k0, k1;
463 if ( cosOmega > 0.9999f ) {
464 k0 = 1.0f - t;
465 k1 = t;
466 } else {
467 // Compute the sin of the angle using the
468 // trig identity sin^2(omega) + cos^2(omega) = 1
469
470 float sinOmega = System.MathF.Sqrt(1.0f - cosOmega*cosOmega);
471
472 // Compute the angle from its sin and cosine
473
474 float omega = System.MathF.Atan2(sinOmega, cosOmega);
475
476 // Compute inverse of denominator, so we only have
477 // to divide once
478
479 float oneOverSinOmega = 1.0f / sinOmega;
480
481 // Compute interpolation parameters
482
483 k0 = System.MathF.Sin((1.0f - t) * omega) * oneOverSinOmega;
484 k1 = System.MathF.Sin(t * omega) * oneOverSinOmega;
485 }
486
487 // Interpolate
488
489 Quatf result = new Quatf();
490
491 result.m_v.X = k0 * q0.m_v.X + k1 * q1x;
492 result.m_v.Y = k0 * q0.m_v.Y + k1 * q1y;
493 result.m_v.Z = k0 * q0.m_v.Z + k1 * q1z;
494 result.m_s = k0 * q0.m_s + k1 * q1w;
495
496 // Return it
497
498 return result;
499 }

503 public CompareResult CompareTo ( Quatf a ) {
504
505 CompareResult _ret = CompareResult.Equal;
506
507 float _a = 0;
508 float _b = 0;
509
510 switch ( m_comparebleMode) {
511 case CompareType.Skalar:
512 _ret = (CompareResult)m_s.CompareTo(a.m_s);
513 break;
514 case CompareType.Norm:
515 _a = m_s * m_s + Vec3f.Lenght(m_v);
516 _b = a.m_s * a.m_s + Vec3f.Lenght(a.m_v);
517 _ret = (CompareResult)_a.CompareTo(_b);
518 break;
519 case CompareType.Rotation:
520 _a = GetRotationAngle();
521 _b = a.GetRotationAngle();
522 _ret = (CompareResult)_a.CompareTo(_b);
523 break;
524 }
525 return _ret;
526 }

598 public static Quatf operator * ( Quatf a, Quatf b ) {
599 Quatf qtmp = new Quatf();
600
601 float ax = a.m_v.X;
602 float ay = a.m_v.Y;
603 float az = a.m_v.Z;
604
605 float bx = b.m_v.X;
606 float by = b.m_v.Y;
607 float bz = b.m_v.Z;
608
609 float as_ = a.m_s;
610 float bs_ = b.m_s;
611
612 // Skalarteil
613 qtmp.m_s = as_ * bs_ - ax * bx - ay * by - az * bz;
614
615 // Vektorteil
616 qtmp.m_v = new Vec3f(
617 as_ * bx + bs_ * ax + ay * bz - az * by,
618 as_ * by + bs_ * ay + az * bx - ax * bz,
619 as_ * bz + bs_ * az + ax * by - ay * bx
620 );
621
622 return qtmp;
623 }
