128 public ColorR16G16B16 AsScaled() {

153 public ContrastLevel GetKontrast(ColorR16G16B16 other) {
154 float c = Contrast(other);
155
156 if ( c < 0.20f ) return ContrastLevel.VeryLow;
157 if ( c < 0.40f ) return ContrastLevel.Low;
158 if ( c < 0.70f ) return ContrastLevel.High;
159 return ContrastLevel.Perfect;
160 }

162 public ColorR16G16B16 GetNextContrastColor() {
