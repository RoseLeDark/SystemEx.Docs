55 public Result Preload()
56 {
57 Result result;
58
59 if (!Directory.Exists(m_path))
60 {
61 result = new Result(new DirectoryNotFoundException($"The specified path '{m_path}' does not exist."));
62 }
63 else
64 {
65 result = new Result();
66 try {
67 foreach (string file in Directory.GetFiles(m_path))
68 {
69 string name = System.IO.Path.GetFileName(file);
70
71 if(IsNativeModule(file) == false)
72 continue;
73
74 if(IsAlreadyInList(name))
75 continue;
76
77 Module? module = Module.LoadModule(name, m_path);
78
79
80 if (module != null) {
81 result.Add("New module loaded ", module);
82 module.AddRef();
83 m_modules.Add(module);
84 }
85 }
86
87 } catch (Exception ex) {
88 result.Catch(ex);
89 return result;
90 }
91 }
92 return result;
93 }

95 public void UnloadAll()
96 {
97 for(int i = m_modules.Count - 1; i >= 0; i--)
98 {
99 if(m_modules[i].RemoveRef() == 0)
100 m_modules.RemoveAt(i);
101 }
102 }

104 public void Unload(Module module)
105 {
106 if(module == null) return;
107
108 if (m_modules.Contains(module))
109 {
110 if(module.RemoveRef() == 0)
111 m_modules.Remove(module);
112 }
113 }

115 public void Unload(string name)
116 {
117 Module? moduleToUnload = null;
118
119 foreach (Module module in m_modules)
120 {
121 if (module.Name.Equals(name, StringComparison.OrdinalIgnoreCase))
122 {
123 moduleToUnload = module;
124 break;
125 }
126 }
127
128 if (moduleToUnload != null)
129 {
130 Module.Unload(moduleToUnload);
131 m_modules.Remove(moduleToUnload);
132 }
133 }

136 private bool IsAlreadyInList(string name)
137 {
138 foreach (Module module in m_modules)

141 return true;
142 }
143 return false;
