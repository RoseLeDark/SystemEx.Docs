110 public Int4 ( bool sign, byte value ) {

113 // Signbit setzen (Bit 3)

135 public Int4 Add ( Int4 oth ) {
136 // Carry löschen
137 m_value.At(5, 0);
138 m_value.At(4, 0);
139
140 for ( byte i = 0 ; i < 4 ; i++ ) {
141 // Summe = a XOR b XOR carry
142 m_value.At(6, (byte)(m_value.Is(i) ^ oth.m_value.Is(i) ^ m_value.Is(5)));
143
144 // Carry = (a & b) | (carry & (a ^ b))
145 m_value.At(7, (byte)(
146 (m_value.Is(i) & oth.m_value.Is(i)) |
147 (m_value.Is(5) & (m_value.Is(i) ^ oth.m_value.Is(i)))
148 ));
149
150 // Ergebnisbit setzen
151 m_value.At(i, m_value.Is(6));
152
153 // Carry übernehmen
154 m_value.At(5, m_value.Is(7));
155 }
156
157 // Temp-Bits säubern
158 m_value.At(5, 0);
159 m_value.At(6, 0);
160 m_value.At(7, 0);
161
162 return this;
163 }

169 public Int4 Sub ( Int4 oth ) {
170 // Borrow löschen
171 m_value.At(5, 0);
172 m_value.At(4, 0);
173
174 for ( byte i = 0 ; i < 4 ; i++ ) {
175 // Differenz = a XOR b XOR borrow
176 m_value.At(6, (byte)(m_value.Is(i) ^ oth.m_value.Is(i) ^ m_value.Is(5)));
177
178 // Neuer Borrow = (NOT a AND b) OR (NOT (a XOR b) AND borrow)
179 // NOT wird durch XOR 1 erreicht
180 m_value.At(7, (byte)(
181 ((m_value.Is(i) ^ 1) & oth.m_value.Is(i)) |
182 (((m_value.Is(i) ^ oth.m_value.Is(i)) ^ 1) & m_value.Is(5))
183 ));
184
185 // Ergebnisbit setzen
186 m_value.At(i, m_value.Is(6));
187
188 // Borrow übernehmen
189 m_value.At(5, m_value.Is(7));
190 }
191
192 // Temp-Bits säubern
193
194 m_value.At(5, 0);
195 m_value.At(6, 0);
196 m_value.At(7, 0);
197
198 return this;
199 }
