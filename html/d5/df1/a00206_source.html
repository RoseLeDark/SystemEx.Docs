343 public bool PushFront ( Pair<T, TU> entry ) {
344 if( intContainsKey(entry.First) ) return false;
345
346 // If full, attempt to grow
347 if ( m_index + 1 >= Length ) {
348 if ( AutoGrow ) {
349 if ( !Resize(Length + GrowSize) )
350 return false;
351 } else {
352 return false;
353 }
354 }
355
356 // Shift all valid elements one slot to the right
357 // m_index is the last valid index, so we shift [0 .. m_index]
358 for ( long i = m_index ; i >= 0 ; i-- ) {
359 m_elements[i + 1] = m_elements[i];
360 m_state[i + 1] = m_state[i];
361 }
362
363 // Insert new element at the front
364 m_elements[0] = entry;
365 m_state[0] = 1;
366
367 // Increase logical count
368 m_index++;
369
370 return true;
371 }

375 public bool Insert ( long index, Pair<T, TU> entry ) {
376 if ( index < 0 ) return false;

380 if ( index >= m_elements.Length || m_index >= m_elements.Length ) {
381 if ( AutoGrow ) {
382 if ( !Resize(m_elements.Length + GrowSize) )
383 return false;

385 return false;

388
389 // Speicher nach rechts verschieben
390 for ( long i = m_index ; i > index ; i-- ) {
391 m_elements[i] = m_elements[i - 1];
392 m_state[i] = m_state[i - 1];
393 }
394
395 m_elements[index] = entry;
396 m_state[index] = 1;
397
398 if ( index >= m_index )
399 m_index = index + 1;
400
401 return true;

669 public Pair<bool, long> CopyTo ( uint sourceOffset, Map<T, TU> destination, ulong destinationOffset, ulong count ) {
670
671 long src = (long)sourceOffset;
672 long dst = (long)destinationOffset;
673
674 if ( src > Length ) src = (long)Length;
675
676 long toCopy = System.Math.Min((long)count,
677 System.Math.Min(System.Math.Max(0, (long)Length - src),
678 System.Math.Max(0, destination.Length - dst)));
679
680 if ( toCopy <= 0 ) return new Pair<bool, long>(false, 0);
681
682 Buffer.LongCopy<Pair<T, TU>>(m_elements, src, destination.m_elements, dst, toCopy);
683 Buffer.LongCopy<byte>(m_state, src, destination.m_state, dst, toCopy);
684
685 long end = dst + toCopy;
686 if ( end > destination.m_index )
687 destination.m_index = end;
688
689 return new Pair<bool, long>(true, toCopy);
690 }

697 public Pair<bool, long> CopyFrom ( Map<T, TU> source, ulong sourceOffset, ulong destinationOffset, ulong count ) {
698 long src = (long)sourceOffset;
699 long dst = (long)destinationOffset;
700
701 if ( dst > Length )
702 dst = Length;
703
704 long toCopy = System.Math.Min((long)count,
705 System.Math.Min(System.Math.Max(0, source.Length - src),
706 System.Math.Max(0, Length - dst)));
707
708 // Wenn nichts passt → prüfen ob wir wachsen müssen
709 if ( toCopy <= 0 ) {
710 if ( !AutoGrow )
711 return new Pair<bool, long>(false, 0);
712
713 long required = dst + (long)count;
714
715 long newSize = Length;
716 while ( required > newSize )
717 newSize += GrowSize;
718
719 if ( !Resize(newSize) )
720 return new Pair<bool, long>(false, 0);
721
722 // Nach Resize neu berechnen
723 toCopy = System.Math.Min((long)count,
724 System.Math.Min(System.Math.Max(0, source.Length - src),
725 System.Math.Max(0, (long)Length - dst)));
726
727 if ( toCopy <= 0 )
728 return new Pair<bool, long>(false, 0);
729 }
730
731 Buffer.LongCopy<Pair<T, TU>>(source.m_elements, src, m_elements, dst, toCopy);
732 Buffer.LongCopy<byte>(source.m_state, src, m_state, dst, toCopy);
733
734 // m_index anpassen, falls wir weiter hinten geschrieben haben
735 long end = dst + toCopy;
736 if ( end > m_index )
737 m_index = end;
738
739 return new Pair<bool, long>(true, toCopy);
740 }
