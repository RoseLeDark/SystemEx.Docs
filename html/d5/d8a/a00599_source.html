315 private static long GreatestCommonDivisor ( long a, long b ) {
316 long temp;
317
318 if ( a > b ) {
319 temp = b;
320 b = a;
321 a = temp;
322 }
323
324 while ( b != 0 ) {
325 temp = a % b;
326 a = b;
327 b = temp;
328 }
329
330 return a;
331 }
