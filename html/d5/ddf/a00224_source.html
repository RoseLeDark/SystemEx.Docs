591 public bool Insert ( long index, T entry ) {
592 if ( index < 0 ) return false;
593
594 // Grow wie im Indexer
595 if ( index >= m_elements.Length || m_index >= m_elements.Length ) {
596 if ( AutoGrow ) {
597 if ( !Resize(m_elements.Length + GrowSize) )
598 return false;
599 } else {
600 return false;
601 }
602 }
603
604 // Speicher nach rechts verschieben
605 for ( long i = m_index ; i > index ; i-- )
606 m_elements[i] = m_elements[i - 1];
607
608 m_elements[index] = entry;
609 m_index++;
610
611 return true;

918 public Pair<bool, long> CopyFrom ( Vector<T> source, ulong sourceOffset, ulong destinationOffset, ulong count ) {
919 long src = (long)sourceOffset;
920 long dst = (long)destinationOffset;
921
922 if ( dst > Length )
923 dst = Length;
924
925 long toCopy = System.Math.Min((long)count,
926 System.Math.Min(System.Math.Max(0, source.Length - src),
927 System.Math.Max(0, Length - dst)));
928
929 // Wenn nichts passt → prüfen ob wir wachsen müssen
930 if ( toCopy <= 0 ) {
931 if ( !AutoGrow )
932 return new Pair<bool, long>(false, 0);
933
934 long required = dst + (long)count;
935
936 long newSize = Length;
937 while ( required > newSize )
938 newSize += GrowSize;
939
940 if ( !Resize(newSize) )
941 return new Pair<bool, long>(false, 0);
942
943 // Nach Resize neu berechnen
944 toCopy = System.Math.Min((long)count,
945 System.Math.Min(System.Math.Max(0, source.Length - src),
946 System.Math.Max(0, (long)Length - dst)));
947
948 if ( toCopy <= 0 )
949 return new Pair<bool, long>(false, 0);
950 }
951
952 Buffer.LongCopy<T>(source.m_elements, src, m_elements, dst, toCopy);
953
954 // m_index anpassen, falls wir weiter hinten geschrieben haben
955 long end = dst + toCopy;
956 if ( end > m_index )
957 m_index = end;
958
959 return new Pair<bool, long>(true, toCopy);
960 }
