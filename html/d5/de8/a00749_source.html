552 private static void DownHeap<T> ( T[] items, int k, int n, CompFunc<T> cmp ) {
553 T temp = items[k - 1];
554
555 while ( k <= n / 2 ) {
556 int child = 2 * k;
557
558 if ( child < n && cmp(items[child - 1], items[child]) == CompareResult.AIsSmallerB )
559 child++;
560
561 if ( cmp(temp, items[child - 1]) == CompareResult.AIsSmallerB ) {
562 items[k - 1] = items[child - 1];
563 k = child;
564 } else break;
565 }
566
567 items[k - 1] = temp;
568 }
