221 public virtual Task WriteAsync(ICache src, CancellationToken cancellationToken = default) {
222 if ( !m_cache.CanWrite ) throw new InvalidOperationException("dest is Locked");
223 if ( !src.CanRead ) throw new InvalidOperationException("src is Locked");
224
225 if ( cancellationToken.IsCancellationRequested )
226 return Task.FromCanceled(cancellationToken);
227
228 try {
229 const int CHUNK = 64 * 1024; // 64 KiB
230 ulong total = (ulong)src.Length;
231 ulong pos = 0;
232
233 while ( pos < total ) {
234 if ( cancellationToken.IsCancellationRequested )
235 return Task.FromCanceled(cancellationToken);
236
237 int toRead = (int)System.Math.Min((ulong)CHUNK, total - pos);
238 byte[]? chunk = src.ReadRange(pos, (uint)toRead);
239
240 if ( chunk == null || chunk.Length == 0 )
241 break;
242
243 m_cache.WriteRange(pos, chunk);
244 pos += (ulong)chunk.Length;
245 }
246
247 return Task.CompletedTask;
248 } catch ( OperationCanceledException oce ) {
249 return Task.FromCanceled(oce.CancellationToken);
250 } catch ( Exception ex ) {
251 return Task.FromException(ex);
252 }
