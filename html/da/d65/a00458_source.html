558 public bool Insert ( long index, T entry ) {
559 if ( index < 0 ) return false;
560
561 // Grow wie im Indexer
562 if ( index >= m_elements.Length || m_index >= m_elements.Length ) {
563 if ( AutoGrow ) {
564 if ( !Resize(m_elements.Length + GrowSize) )
565 return false;
566 } else {
567 return false;
568 }
569 }

572 for ( long i = m_index ; i > index ; i-- )
573 m_elements[i] = m_elements[i - 1];

885 public Pair<bool, long> CopyFrom ( Vector<T> source, ulong sourceOffset, ulong destinationOffset, ulong count ) {
886 long src = (long)sourceOffset;
887 long dst = (long)destinationOffset;
888
889 if ( dst > Length )
890 dst = Length;
891
892 long toCopy = System.Math.Min((long)count,
893 System.Math.Min(System.Math.Max(0, source.Length - src),
894 System.Math.Max(0, Length - dst)));
895
896 // Wenn nichts passt → prüfen ob wir wachsen müssen
897 if ( toCopy <= 0 ) {
898 if ( !AutoGrow )
899 return new Pair<bool, long>(false, 0);
900
901 long required = dst + (long)count;
902
903 long newSize = Length;
904 while ( required > newSize )
905 newSize += GrowSize;
906
907 if ( !Resize(newSize) )
908 return new Pair<bool, long>(false, 0);
909
910 // Nach Resize neu berechnen
911 toCopy = System.Math.Min((long)count,
912 System.Math.Min(System.Math.Max(0, source.Length - src),
913 System.Math.Max(0, (long)Length - dst)));
914
915 if ( toCopy <= 0 )
916 return new Pair<bool, long>(false, 0);
917 }
918
919 Buffer.LongCopy<T>(source.m_elements, src, m_elements, dst, toCopy);
920
921 // m_index anpassen, falls wir weiter hinten geschrieben haben
922 long end = dst + toCopy;
923 if ( end > m_index )
924 m_index = end;
925
926 return new Pair<bool, long>(true, toCopy);
927 }
