262 public override string ToString () {

265 sb.Append("ModelPromp<");

272 sb.Append(", ");

274 // Session
275 sb.Append("SessionId=");

277 sb.Append(", ");
278
279 // Cancel
280 sb.Append("Cancel=");
281 sb.Append(Cancel);
282 sb.Append(", ");
283
284 // Context
285 sb.Append("Context=[");
286 foreach ( var kv in m_context ) {
287 sb.Append(kv.First);
288 sb.Append(":");
289 sb.Append(kv.Second);
290 sb.Append(", ");
291 }
292 if ( m_context.Count > 0 )
293 sb.Length -= 2; // letztes ", " entfernen
294 sb.Append("], ");
295
296 // Parameters
297 sb.Append("Parameters=[");
298 foreach ( var kv in m_parameter ) {
299 sb.Append(kv.First);
300 sb.Append(":");
301 sb.Append(kv.Second);
302 sb.Append(", ");
303 }
304 if ( m_parameter.Count > 0 )
305 sb.Length -= 2;
306 sb.Append("], ");
307
308 // Tags
309 sb.Append("Tags=[");
310 foreach ( var kv in m_tags ) {
311 sb.Append(kv.First);
312 sb.Append(":");
313 sb.Append(kv.Second);
314 sb.Append(", ");
315 }
316 if ( m_tags.Count > 0 )
317 sb.Length -= 2;
318 sb.Append("]");
319
320 sb.Append(")");
321
322 return sb.ToString();
323 }
