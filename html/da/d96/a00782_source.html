191 if(min == max) {
192 m_min = min;

195 m_min = System.Math.Min(min, max);

209 public long WaitOpen( TimeOut waitMS ) {
210 while(true) {
211 if ( (int)m_current.Value == m_max ) {
212 m_current.Assign(m_min);
213 break;
214 }
215
216 Thread.SpinWait(waitMS.Milliseconds);
217 }
218
219 m_hold.Unlock();
220
221 if ( !m_hold.IsHeld ) {
222
223 m_context.Lock(-1);
224
225 OnComplition?.Invoke((int)m_phase.Value, this);
226
227
228 if ( m_hold.Lock(-1) ) {
229 if ( m_drops > 0 ) {
230 if ( m_max - m_drops >= m_min )
231 m_max -= (uint)m_drops.Value;
232 else
233 throw new BarrierNewPhaseLockException("To many drops");
234 }
235 m_current.Assign(m_min);
236 m_phase++;
237 currentPhase = m_phase.Value;
238 }
239
240 m_context.Unlock();
241 }
242
243
244 return currentPhase;
