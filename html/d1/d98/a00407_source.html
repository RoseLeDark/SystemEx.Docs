60 public static ColorR8G8B8 ToColorR8G8B8(this ColorXYZ c) {
61 float X = c.X;
62 float Y = c.Y;
63 float Z = c.Z;
64
65 // XYZ → linear RGB
66 float r_lin = 3.2404542f * X - 1.5371385f * Y - 0.4985314f * Z;
67 float g_lin = -0.9692660f * X + 1.8760108f * Y + 0.0415560f * Z;
68 float b_lin = 0.0556434f * X - 0.2040259f * Y + 1.0572252f * Z;
69
70 // linear RGB → sRGB (Gamma)
71 float r = (r_lin <= 0.0031308f) ? 12.92f * r_lin : 1.055f * MathF.Pow(r_lin, 1f / 2.4f) - 0.055f;
72 float g = (g_lin <= 0.0031308f) ? 12.92f * g_lin : 1.055f * MathF.Pow(g_lin, 1f / 2.4f) - 0.055f;
73 float b = (b_lin <= 0.0031308f) ? 12.92f * b_lin : 1.055f * MathF.Pow(b_lin, 1f / 2.4f) - 0.055f;
74
75 // Clamp (sRGB muss 0..1 sein)
76 return new ColorR8G8B8(r < 0f ? 0f : (r > 1f ? 1f : r),
77 g < 0f ? 0f : (g > 1f ? 1f : g),
78 b < 0f ? 0f : (b > 1f ? 1f : b)
79 );
80 }

136 public static ColorHSV ToColorHSV(this ColorR8G8B8 x) {
137 float max = System.Math.Max(x.Red, System.Math.Max(x.Green, x.Blue));
138 float min = System.Math.Min(x.Red, System.Math.Min(x.Green, x.Blue));
139 float delta = max - min;
140
141 float h = 0f;
142 float s = 0f;
143 float v = max;
144
145 if ( delta > 0f ) {
146 s = delta / max;
147
148 if ( max == x.Red )
149 h = 60f * (((x.Green - x.Blue) / delta) % 6f);
150 else if ( max == x.Green )
151 h = 60f * (((x.Blue - x.Red) / delta) + 2f);
152 else
153 h = 60f * (((x.Red - x.Green) / delta) + 4f);
154
155 if ( h < 0f )
156 h += 360f;
157 }
158
159 return new ColorHSV(h, s, v);
160 }

164 public static ColorHWB ToColorHWB(this ColorR8G8B8 color) {
165 var c = color.AsNormalized();
166
167 float max = System.Math.Max(c.Red, System.Math.Max(c.Green, c.Blue));
168 float min = System.Math.Min(c.Red, System.Math.Min(c.Green, c.Blue));
169 float delta = max - min;
170
171 float w = min;
172 float b = 1 - max;
173 float h;
174
175 if ( delta == 0 )
176 h = 0;
177 else if ( max == c.Red )
178 h = (c.Green - c.Blue) / delta;
179 else if ( max == c.Green )
180 h = 2 + (c.Blue - c.Red) / delta;
181 else
182 h = 4 + (c.Red - c.Green) / delta;
183
184 h *= 60;
185 if ( h < 0 ) h += 360;
186
187 return new ColorHWB(h, w *= 100, b *= 100);
188 }

192 public static ColorR8G8B8 ToColorR8G8B8(this ColorHWB color) {
193 float h = Math.FMod(color.H, 360.0f) / 60.0f;
194 float w = color.Whiteness / 100.0f;
195 float bl = color.Blackness / 100.0f;
196
197 float v = 1 - bl;
198 float s = v == 0 ? 0 : 1 - w / v;
199 float c = v * s;
200 float x = c * (1 - System.Math.Abs( Math.FMod (h, 2.0f) - 1f));
201 float m = v - c;
202
203 float r, g, b;
204 if ( h < 1 ) { r = c; g = x; b = 0; }
205 else if ( h < 2 ) { r = x; g = c; b = 0; }
206 else if ( h < 3 ) { r = 0; g = c; b = x; }
207 else if ( h < 4 ) { r = 0; g = x; b = c; }
208 else if ( h < 5 ) { r = x; g = 0; b = c; }
209 else { r = c; g = 0; b = x; }
210
211 return new ColorR8G8B8(
212 ((r + m)),
213 ((g + m)),
214 ((b + m))
215 );
216 }

221 public static ColorR8G8B8 ToColorR8G8B8(this ColorHSV color) {
222 float C = color.V * color.S; // Chroma
223 float Hp = Math.FMod(color.H / 60f, 6f); // Hue' (0..6)
224 float X = C * (1f - System.Math.Abs(Math.FMod(Hp, 2f) - 1f));
225 float m = color.V - C;
226
227 float r, g, b;
228
229 if ( Hp < 1f ) { r = C; g = X; b = 0; }
230 else if ( Hp < 2f ) { r = X; g = C; b = 0; }
231 else if ( Hp < 3f ) { r = 0; g = C; b = X; }
232 else if ( Hp < 4f ) { r = 0; g = X; b = C; }
233 else if ( Hp < 5f ) { r = X; g = 0; b = C; }
234 else if ( Hp < 6f ) { r = C; g = 0; b = X; }
235 else { r = 0; g = 0; b = 0; }
236
237 return new ColorR8G8B8(r + m, g + m, b + m);
238 }

249 public static ColorHSL ToColorHSL(this ColorHSV color) {
250 // Lightness
251 float L = color.V * (1f - color.S * 0.5f);
252
253 // Saturation in HSL
254 float S_hsl = 0f;
255
256 if ( L > 0f && L < 1f ) {
257 float minL = (L < 0.5f) ? L : (1f - L);
258 if ( minL > 0f )
259 S_hsl = (color.V - L) / minL;
260 }
261
262 return new ColorHSL(color.H, S_hsl, L);
