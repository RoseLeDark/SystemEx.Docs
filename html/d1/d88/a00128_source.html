72 public long Find ( ref TContainer container, ICompared<T> comp, T value ) {
73 if ( container.Count == 0 ) return 0;
74
75 long _ret = 0;
76 long left = 0;
77 long right = container.Count - 1;
78
79 // Try to locate ANY match using midpoint probing
80 while ( left <= right ) {
81 long mid = (left + right) >> 1;
82 Optional<T> item = container.ElementAt(mid);
83 if ( item.IsNull ) continue;
84
85 if ( comp.Compare(item.Value, value) == Utils.CompareResult.Equal ) {
86 // Found one → expand to count all
87 _ret++;
88
89 // expand left
90 for ( var i = mid - 1 ; i >= 0 ; i-- ) {
91 Optional<T> _i2 = container.ElementAt(i);
92 if ( item.IsNull ) continue;
93
94 if ( comp.Compare(_i2.Value, value) == Utils.CompareResult.Equal )
95 _ret++;
96 else
97 break;
98 }
99
100 // expand right
101 for ( var i = mid + 1 ; i < container.Count ; i++ ) {
102 Optional<T> _i2 = container.ElementAt(i);
103 if ( item.IsNull ) continue;
104
105 if ( comp.Compare(_i2.Value, value) == Utils.CompareResult.Equal )
106 _ret++;
107 else
108 break;
109 }
110 break;
111 }
112
113 // Without ordering, we cannot decide direction → shrink inward
114 left++;
115 right--;
116 }
117
118 return _ret;
119 }
