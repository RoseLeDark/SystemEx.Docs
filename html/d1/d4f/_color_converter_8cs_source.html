45 public static ColorR8G8B8 ToLinear(this ColorR8G8B8 rgb) {
46 float r = (rgb.Red <= 0.04045f) ? (rgb.Red / 12.92f) : MathF.Pow((rgb.Red + 0.055f) / 1.055f, 2.4f);
47 float g = (rgb.Green <= 0.04045f) ? (rgb.Green / 12.92f) : MathF.Pow((rgb.Green + 0.055f) / 1.055f, 2.4f);

57 public static ColorR8G8B8 ToColorR8G8B8(this ColorXYZ c) {
58 float X = c.X;
59 float Y = c.Y;
60 float Z = c.Z;
61
62 // XYZ → linear RGB
63 float r_lin = 3.2404542f * X - 1.5371385f * Y - 0.4985314f * Z;
64 float g_lin = -0.9692660f * X + 1.8760108f * Y + 0.0415560f * Z;
65 float b_lin = 0.0556434f * X - 0.2040259f * Y + 1.0572252f * Z;
66
67 // linear RGB → sRGB (Gamma)
68 float r = (r_lin <= 0.0031308f) ? 12.92f * r_lin : 1.055f * MathF.Pow(r_lin, 1f / 2.4f) - 0.055f;
69 float g = (g_lin <= 0.0031308f) ? 12.92f * g_lin : 1.055f * MathF.Pow(g_lin, 1f / 2.4f) - 0.055f;
70 float b = (b_lin <= 0.0031308f) ? 12.92f * b_lin : 1.055f * MathF.Pow(b_lin, 1f / 2.4f) - 0.055f;
71
72 // Clamp (sRGB muss 0..1 sein)
73 return new ColorR8G8B8(r < 0f ? 0f : (r > 1f ? 1f : r),
74 g < 0f ? 0f : (g > 1f ? 1f : g),
75 b < 0f ? 0f : (b > 1f ? 1f : b)
76 );
77 }

133 public static ColorHSV ToColorHSV(this ColorR8G8B8 x) {
134 float max = System.Math.Max(x.Red, System.Math.Max(x.Green, x.Blue));
135 float min = System.Math.Min(x.Red, System.Math.Min(x.Green, x.Blue));
136 float delta = max - min;
137
138 float h = 0f;
139 float s = 0f;
140 float v = max;
141
142 if ( delta > 0f ) {
143 s = delta / max;
144
145 if ( max == x.Red )
146 h = 60f * (((x.Green - x.Blue) / delta) % 6f);
147 else if ( max == x.Green )
148 h = 60f * (((x.Blue - x.Red) / delta) + 2f);
149 else
150 h = 60f * (((x.Red - x.Green) / delta) + 4f);
151
152 if ( h < 0f )
153 h += 360f;
154 }
155
156 return new ColorHSV(h, s, v);
157 }

161 public static ColorHWB ToColorHWB(this ColorR8G8B8 color) {
162 var c = color.AsNormalized();
163
164 float max = System.Math.Max(c.Red, System.Math.Max(c.Green, c.Blue));
165 float min = System.Math.Min(c.Red, System.Math.Min(c.Green, c.Blue));
166 float delta = max - min;
167
168 float w = min;
169 float b = 1 - max;
170 float h;
171
172 if ( delta == 0 )
173 h = 0;
174 else if ( max == c.Red )
175 h = (c.Green - c.Blue) / delta;
176 else if ( max == c.Green )
177 h = 2 + (c.Blue - c.Red) / delta;
178 else
179 h = 4 + (c.Red - c.Green) / delta;
180
181 h *= 60;
182 if ( h < 0 ) h += 360;
183
184 return new ColorHWB(h, w *= 100, b *= 100);
185 }

189 public static ColorR8G8B8 ToColorR8G8B8(this ColorHWB color) {
190 float h = Math.FMod(color.H, 360.0f) / 60.0f;
191 float w = color.Whiteness / 100.0f;
192 float bl = color.Blackness / 100.0f;
193
194 float v = 1 - bl;
195 float s = v == 0 ? 0 : 1 - w / v;
196 float c = v * s;
197 float x = c * (1 - System.Math.Abs( Math.FMod (h, 2.0f) - 1f));
198 float m = v - c;
199
200 float r, g, b;
201 if ( h < 1 ) { r = c; g = x; b = 0; }
202 else if ( h < 2 ) { r = x; g = c; b = 0; }
203 else if ( h < 3 ) { r = 0; g = c; b = x; }
204 else if ( h < 4 ) { r = 0; g = x; b = c; }
205 else if ( h < 5 ) { r = x; g = 0; b = c; }
206 else { r = c; g = 0; b = x; }
207
208 return new ColorR8G8B8(
209 ((r + m)),
210 ((g + m)),
211 ((b + m))
212 );
213 }

218 public static ColorR8G8B8 ToColorR8G8B8(this ColorHSV color) {
219 float C = color.V * color.S; // Chroma
220 float Hp = Math.FMod(color.H / 60f, 6f); // Hue' (0..6)
221 float X = C * (1f - System.Math.Abs(Math.FMod(Hp, 2f) - 1f));
222 float m = color.V - C;
223
224 float r, g, b;
225
226 if ( Hp < 1f ) { r = C; g = X; b = 0; }
227 else if ( Hp < 2f ) { r = X; g = C; b = 0; }
228 else if ( Hp < 3f ) { r = 0; g = C; b = X; }
229 else if ( Hp < 4f ) { r = 0; g = X; b = C; }
230 else if ( Hp < 5f ) { r = X; g = 0; b = C; }
231 else if ( Hp < 6f ) { r = C; g = 0; b = X; }
232 else { r = 0; g = 0; b = 0; }
233
234 return new ColorR8G8B8(r + m, g + m, b + m);
235 }

246 public static ColorHSL ToColorHSL(this ColorHSV color) {
247 // Lightness
248 float L = color.V * (1f - color.S * 0.5f);
249
250 // Saturation in HSL
251 float S_hsl = 0f;
252
253 if ( L > 0f && L < 1f ) {
254 float minL = (L < 0.5f) ? L : (1f - L);
255 if ( minL > 0f )
256 S_hsl = (color.V - L) / minL;
257 }
258
259 return new ColorHSL(color.H, S_hsl, L);
260 }
