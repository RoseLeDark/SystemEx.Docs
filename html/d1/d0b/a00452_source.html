139 public ColorHSV Lerp(ColorHSV value, float amount) {
140 float q = amount * amount;
141
142 // Shortest-path hue interpolation
143 float dh = value.m_hue - m_hue;
144 dh = (dh + 540f) % 360f - 180f;
145
146 return new ColorHSV(
147 ClampHue(m_hue + dh * q),
148 m_saturation + (value.m_saturation - m_saturation) * q,
149 m_value + (value.m_value - m_value) * q
150 );
151 }
