145 public LightCountingSpinlock ( int minCapacity, int maxCapacity, string name = "CountingSpinlock" ) {
146 if ( maxCapacity <= 0 )

162 public virtual bool Lock ( int millisecondsTimeout = -1) {
163 bool _ret = false;
164 bool _usedTimer = millisecondsTimeout > 0;
165
166
167
168 if ( _usedTimer ) {
169 long start = Environment.TickCount64;
170
171 while ( true ) {
172
173 // Timeout erreicht?
174 if ( millisecondsTimeout >= 0 ) {
175 long elapsed = Environment.TickCount64 - start;
176 if ( elapsed >= millisecondsTimeout )
177 break; // nicht geschafft
178 }
179
180 // Wenn keine Kapazität → sofort spinnen
181 if ( IsLocked ) {
182 Thread.SpinWait(1);
183 continue;
184 }
185
186 // Atomar dekrementieren
187 m_counter--;
188 long newValue = m_counter.Value;
189
190 // Erfolg: nicht unter minCapacity gerutscht
191 if ( newValue >= m_min ) {
192 _ret = true;
193 break;
194 }
195
196 // Unter minCapacity gerutscht → revert + spin
197 m_counter++;
198 Thread.SpinWait(1);
199 }
200 } else {
201
202 while ( true ) {
203
204 // Wenn keine Kapazität → sofort spinnen
205 if ( IsLocked ) {
206 Thread.SpinWait(1);
207
208 continue;
209 }
210
211 // Atomar dekrementieren
212 m_counter--;
213 long newValue = m_counter.Value;
214
215 // Erfolg: nicht unter minCapacity gerutscht
216 if ( newValue >= m_min ) {
217 _ret = true;
218 break;
219 }
220
221 // Unter minCapacity gerutscht → revert + spin
222 m_counter.Increment();
223 Thread.SpinWait(1);
224 }
225
226 }
227
228 return _ret;
229 }

235 public virtual bool Lock ( TimeSpan span ) {
236 // Unendlich warten, wenn TimeSpan extrem groß ist
237 if ( span == Timeout.InfiniteTimeSpan || span.TotalMilliseconds > int.MaxValue )
238 return Lock(-1);
239
240 // Negative oder Null → kein Warten
241 if ( span <= TimeSpan.Zero )
242 return Lock(0);
243
244 // Normale Umrechnung
245 int ms = (int)span.TotalMilliseconds;
246 return Lock(ms);
247 }
