102 private bool internalLock ( int ms, bool ecxp ) {
103 if ( IsHeldbyCurrent ) return true;
104 if ( IsHeld ) if ( ecxp ) throw new Exception("Held by other thread"); else return false;
105
106 var lockTaken = false;
107
108 if ( ms <= 0 ) {
109
110 while ( !lockTaken ) {
111 lockTaken = Interlocked.Exchange(ref m_lock, Thread.CurrentThread.ManagedThreadId) == 0;
112 if ( !lockTaken )
113 Thread.SpinWait(1);
114 }
115 } else {
116
117 var sw = System.Diagnostics.Stopwatch.StartNew();
118
119 while ( !lockTaken && sw.ElapsedMilliseconds < ms ) {
120 lockTaken = Interlocked.Exchange(ref m_lock, Thread.CurrentThread.ManagedThreadId) == 0;
121 if ( !lockTaken )
122 Thread.SpinWait(1);
123 }
124 }
125
126 return lockTaken;
