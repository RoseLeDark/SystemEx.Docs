112
113 if ( m_waits.IsEmpty ) return;
114
115 Optional<LightThread> _th;
116
117 if ( m_waits.PopFront(out _th) ) {

133
134 while ( !m_waits.IsEmpty ) {
135 Optional<LightThread> _th;
136
137 if ( m_waits.PopFront(out _th) ) {

141#endif
142 }
