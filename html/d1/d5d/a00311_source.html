111 private static byte[] MirrorRead(byte[] data) {
112 int i = 0;
113 int j = data.Length - 1;
114
115 while ( i < j ) {
116 byte tmp = data[i];
117 data[i] = data[j];
118 data[j] = tmp;
119 i++;
120 j--;
121 }
122 return data;
123 }

128 private static byte[] Mirror(byte[] data) {
129 byte[] m = new byte[data.Length];
130 int last = data.Length - 1;

132 for ( int i = 0; i < data.Length; i++ )
