77 if ( size <= 0 ) continue;
78
79 byte[] raw = obj.ReadRange((ulong)offset, (uint)size)!;

81 entries.Add(name, raw);
82 }
83 return (T)CreateObjectFromEntrys(entries, m_endian)!;
