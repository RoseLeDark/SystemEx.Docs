108 public void Update ( byte[] input ) {
109 if ( input.Length == 0 ) return;
110 int input_pos = 0;
111 int input_len =input.Length;
112
113 var remaining = input;
114
115 // Fill existing buffer if partially full
116 if ( m_buf_len > 0 ) {
117 int take = Black3Infos.BLAKE3_CHUNK_LEN - m_buf_len;
118 if ( take > input_len )
119 take = input_len;
120
121 for ( int i = 0 ; i < take ; i++ )
122 m_buf[m_buf_len + i] = input[input_pos + i];
123
124 m_buf_len += (byte)take;
125 input_pos += take;
126 input_len -= take;
127
128 // If chunk is full, finalize it
129 if ( input_len > 0 ) {
130 Blacke3_Output output = Finalize();
131
132 byte[] chunk_cv = new byte[32];
133 output.ChainingValue(chunk_cv);
134
135 m_parent.PushCv(chunk_cv, m_chunk_counter);
136
137 Reset(m_parent.m_key, m_chunk_counter + 1);
138 } else {
139 return;
140 }
141 }
142 // Process full chunks directly
143 while ( input_len >= Black3Infos.BLAKE3_CHUNK_LEN ) {
144
145 for ( int i = 0 ; i < Black3Infos.BLAKE3_CHUNK_LEN ; i++ )
146 m_buf[i] = input[input_pos + i];
147
148 m_buf_len = Black3Infos.BLAKE3_CHUNK_LEN;
149
150 // finalize chunk
151 Blacke3_Output output = Finalize ();
152
153 byte[] chunk_cv = new byte[32];
154 output.ChainingValue(chunk_cv);
155
156 m_parent.PushCv(chunk_cv, m_chunk_counter);
157
158 Reset(m_parent.m_key, m_chunk_counter + 1);
159
160 input_pos += Black3Infos.BLAKE3_CHUNK_LEN;
161 input_len -= Black3Infos.BLAKE3_CHUNK_LEN;
162 }
163
164 // Copy remaining bytes
165 if ( input_len > 0 ) {
166 for ( int i = 0 ; i < input_len ; i++ )
167 m_buf[m_buf_len + i] = input[input_pos + i];
168
169 m_buf_len += (byte)input_len;
170 }

172 }
