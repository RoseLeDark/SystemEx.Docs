351 public bool PushFront ( Pair<T, TU> entry ) {
352 if( intContainsKey(entry.First) ) return false;
353
354 // If full, attempt to grow
355 if ( m_index + 1 >= Length ) {
356 if ( AutoGrow ) {
357 if ( !Resize(Length + GrowSize) )
358 return false;
359 } else {
360 return false;
361 }
362 }
363
364 // Shift all valid elements one slot to the right
365 // m_index is the last valid index, so we shift [0 .. m_index]
366 for ( long i = m_index ; i >= 0 ; i-- ) {
367 m_elements[i + 1] = m_elements[i];
368 m_state[i + 1] = m_state[i];
369 }
370
371 // Insert new element at the front
372 m_elements[0] = entry;
373 m_state[0] = 1;
374
375 // Increase logical count
376 m_index++;
377
378 return true;
379 }

383 public bool Insert ( long index, Pair<T, TU> entry ) {
384 if ( index < 0 ) return false;
385 if ( intContainsKey(entry.First) ) return false;
386
387 // Grow wie im Indexer
388 if ( index >= m_elements.Length || m_index >= m_elements.Length ) {
389 if ( AutoGrow ) {
390 if ( !Resize(m_elements.Length + GrowSize) )
391 return false;
392 } else {
393 return false;
394 }
395 }
396
397 // Speicher nach rechts verschieben
398 for ( long i = m_index ; i > index ; i-- ) {
399 m_elements[i] = m_elements[i - 1];
400 m_state[i] = m_state[i - 1];
401 }
402
403 m_elements[index] = entry;
404 m_state[index] = 1;
405
406 if ( index >= m_index )
407 m_index = index + 1;
408
409 return true;

677 public Pair<bool, long> CopyTo ( uint sourceOffset, Map<T, TU> destination, ulong destinationOffset, ulong count ) {
678
679 long src = (long)sourceOffset;
680 long dst = (long)destinationOffset;
681
682 if ( src > Length ) src = (long)Length;
683
684 long toCopy = System.Math.Min((long)count,
685 System.Math.Min(System.Math.Max(0, (long)Length - src),
686 System.Math.Max(0, destination.Length - dst)));
687
688 if ( toCopy <= 0 ) return new Pair<bool, long>(false, 0);
689
690 Buffer.LongCopy<Pair<T, TU>>(m_elements, src, destination.m_elements, dst, toCopy);
691 Buffer.LongCopy<byte>(m_state, src, destination.m_state, dst, toCopy);
692
693 long end = dst + toCopy;
694 if ( end > destination.m_index )
695 destination.m_index = end;
696
697 return new Pair<bool, long>(true, toCopy);
698 }

705 public Pair<bool, long> CopyFrom ( Map<T, TU> source, ulong sourceOffset, ulong destinationOffset, ulong count ) {
706 long src = (long)sourceOffset;
707 long dst = (long)destinationOffset;
708
709 if ( dst > Length )
710 dst = Length;
711
712 long toCopy = System.Math.Min((long)count,
713 System.Math.Min(System.Math.Max(0, source.Length - src),
714 System.Math.Max(0, Length - dst)));
715
716 // Wenn nichts passt → prüfen ob wir wachsen müssen
717 if ( toCopy <= 0 ) {
718 if ( !AutoGrow )
719 return new Pair<bool, long>(false, 0);
720
721 long required = dst + (long)count;
722
723 long newSize = Length;
724 while ( required > newSize )
725 newSize += GrowSize;
726
727 if ( !Resize(newSize) )
728 return new Pair<bool, long>(false, 0);
729
730 // Nach Resize neu berechnen
731 toCopy = System.Math.Min((long)count,
732 System.Math.Min(System.Math.Max(0, source.Length - src),
733 System.Math.Max(0, (long)Length - dst)));
734
735 if ( toCopy <= 0 )
736 return new Pair<bool, long>(false, 0);
737 }
738
739 Buffer.LongCopy<Pair<T, TU>>(source.m_elements, src, m_elements, dst, toCopy);
740 Buffer.LongCopy<byte>(source.m_state, src, m_state, dst, toCopy);
741
742 // m_index anpassen, falls wir weiter hinten geschrieben haben
743 long end = dst + toCopy;
744 if ( end > m_index )
745 m_index = end;
746
747 return new Pair<bool, long>(true, toCopy);
748 }
