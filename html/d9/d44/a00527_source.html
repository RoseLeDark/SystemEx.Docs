166 public bool Cost(T value, ref ulong cost, ulong budget) {
167 bool _ret = false;
168 // Priority Queue: (current cost, node)
169 var pq = new PriorityQueue<ICluster<T>, ulong>();
170 var visited = new Vector<ICluster<T>>(128, 16);
171 var costs = new Map<ICluster<T>, ulong>();
172
173 var finder = Vector<ICluster<T>>.AsFinder(ref visited);
174
175 // Start at this node with cost 0
176 pq.Enqueue(this, 0);
177 costs[this] = 0;
178
179 while ( pq.Count > 0 ) {
180 Optional<ICluster<T>> current = pq.Dequeue();
181 if ( current.IsNull ) continue;
182
183 // Skip if already visited
184 if ( finder.Exists(current.Value!) )
185 continue;
186
187 visited.PushBack(current.Value!);
188 Optional<ulong> currentCost = costs[current.Value!];
189
190 // If current cost exceeds budget, abandon this path
191 if ( currentCost.Value! > budget )
192 continue;
193
194 // Found target?
195 if ( current.Value != null && current.Value.Equals(value) ) {
196 cost = currentCost.Value!;
197 _ret = true;
198 break;
199 }
200
201 // Traverse children
202 for ( int i = 0; i < current.Value!.Child.Size; i++ ) {
203 var childPair = current.Value!.Child.ElementAt(i); // Pair<ICluster<T>, int>
204 ICluster<T> childNode = childPair.First;
205 uint edgeCost = childPair.Second;
206
207 if ( finder.Exists(childNode) )
208 continue;
209
210 ulong newCost = currentCost.Value! + (ulong)edgeCost;
211
212 // Only continue if within budget
213 if ( newCost > budget )
214 continue;
215
216 // Found a better path?
217 var newC = costs[childNode];
218
219 if ( !costs.ContainsKey(childNode) || newCost < newC.Value! ) {
220 costs[childNode] = newCost;
221 pq.Enqueue(childNode, newCost);
222 }
223 }
224 }
225
226 // Return whether target was found within budget
227 return _ret;
228 }

233 public bool find(T pradicat, SearchType type, ulong budget, ref Map<ICluster<T>, uint> steps) {
234 // PriorityQueue: (cost, node)
235 var pq = new PriorityQueue<Triple<ICluster<T>, ulong, ulong>, ulong>();
236 var visited = new Vector<ICluster<T>>(128, 16);
237 var finder = Vector<ICluster<T>>.AsFinder(ref visited);
238
239 // Start: energie = budget, cost = 0
240 pq.Enqueue( new Triple<ICluster<T>, ulong, ulong>(this, budget, 0), 0);
241
242 while ( pq.Count > 0 ) {
243 var item = pq.Dequeue();
244 if ( item.IsNull ) continue;
245
246 ICluster<T> current = item.Value!.First;
247 ulong energie = item.Value!.Second;
248 ulong cost = item.Value!.Third;
249
250 if ( finder.Exists(current) )
251 continue;
252
253 visited.PushBack(current);
254
255 // Schritt speichern
256 steps[current] = (uint)energie;
257
258 // Ziel gefunden?
259 if ( current.Value != null && (current.Value.Equals(pradicat) ) )
260 return true;
261
262 // Kinder durchgehen
263 for ( int i = 0; i < current.Child.Size; i++ ) {
264 var pair = current.Child.ElementAt(i);
265
266 var child = pair.First;
267 ulong gewicht = pair.Second;
268
269 if ( finder.Exists(child) )
270 continue;
271
272 ulong newEnergie = energie;
273 ulong newCost = cost;
274
275 switch ( type ) {
276 case SearchType.SubtractWeight:
277 if ( gewicht > newEnergie )
278 continue;
279 newEnergie -= (ulong)gewicht;
280 break;
281
282 case SearchType.SetToWeight:
283 newEnergie = (ulong)gewicht;
284 if ( newEnergie > budget )
285 continue;
286 break;
287
288 case SearchType.AccumulateCost:
289 newCost += (ulong)gewicht;
290 if ( newCost > budget )
291 continue;
292 break;
293 }
294
295 // Enqueue child with priority = accumulated cost
296 pq.Enqueue(new Triple<ICluster<T>, ulong, ulong>(child, newEnergie, newCost), newCost);
297 }
298 }
299 // Not found within budget/constraints
300 return false;
301 }
