136 public T[] Extract ( long index ) {
137 T[] _ret = Array.Empty<T>();
138
139 long count = m_pKeys.Count;
140
141 if ( index >= 0 && index < count ) {
142 long extractCount = count - index;
143 _ret = new T[extractCount];
144
145 // copy extracted elements
146 for ( long i = 0 ; i < extractCount ; i++ ) {
147 var itm = m_pKeys.ElementAt(index + i);
148 if(!itm.IsNull) {
149 _ret[i] = itm.Value!;
150 }
151 }
152
153 // remove extracted elements from container
154 for ( long i = count - 1 ; i >= index ; i-- ) {
155 m_pKeys.Erase(i);
156 }
157 }
158
159 return _ret;
160 }

395 public void Merge ( VectorSet<T, TContainer> source ) {
396 long count = source.m_pKeys.Count;
397
398 for ( long i = 0 ; i < count ; i++ ) {
399 Optional<T> value = source.m_pKeys.ElementAt(i);
400 if ( value.IsNull )
401 continue;
402
403 // check if this set already contains the value
404 long idx = m_finder.First(value.Value!);
405
406 if ( idx == -1 ) {
407 // extract from source
408 T[] extracted = source.Extract(i);
409
410 // insert into this set
411 m_pKeys.InsertRange(m_pKeys.Count, extracted);
412
413 // adjust loop because source shrinks
414 count--;
415 i--;
416 }
417 }

442 private void Sort () {
443 if ( m_sorter != null ) {
444 m_sorter(ref m_pKeys, m_compare);
445 } else {
446 // Fallback
447 for ( int i = 0 ; i < m_pKeys.Count - 1 ; i++ ) {
448 for ( int j = i + 1 ; j < m_pKeys.Count ; j++ ) {
449 bool cmp = m_compare.Compare( m_pKeys.ElementAt(i), m_pKeys.ElementAt(j) );
450
451 if (!cmp ) {
452 Swap(i, j);
453 }
454 }
455 }
