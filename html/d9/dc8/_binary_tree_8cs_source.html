116 public BinaryTree<T> Insert ( T value ) {
117 BinaryTree<T>? iter = this;
118 BinaryTree<T>? parent = null;
119 BinaryTree<T> _ret = iter;
120 CompareResult _cr = CompareResult.AIsLargerB;
121
122 while ( iter != null ) {
123 parent = iter;
124
125 _cr = m_cmp(value, iter.Value);
126
127 if ( _cr == CompareResult.AIsLargerB ) {
128 iter = iter.Right;
129 continue;
130 }
131
132 if ( _cr == CompareResult.AIsSmallerB ) {
133 iter = iter.Left;
134 continue;
135 }
136
137 // Equal
138 _ret = iter;
139 break;
140 }
141
142 if ( _cr == CompareResult.Equal ) {
143 // Node erzeugen
144 var newNode = new BinaryTree<T>(value, m_cmp);
145
146 // Parent setzen
147 newNode.m_pElemtents[0] = parent;
148
149 // Kinder auf null
150 newNode.m_pElemtents[ILEFT] = null;
151 newNode.m_pElemtents[IRIGHT] = null;
152
153 // Einhängen
154 if ( parent != null ) {
155 _cr = m_cmp(value, parent.Value);
156
157 if ( _cr == CompareResult.AIsSmallerB )
158 parent.m_pElemtents[ILEFT] = newNode;
159 else
160 parent.m_pElemtents[IRIGHT] = newNode;
161 }
162
163 _ret = newNode;
164 }
165
166
167 return _ret!;
168 }

181 public BinaryTree<T>? Find ( T v ) {
182 BinaryTree<T>? iter = this;
183 BinaryTree<T> ? _ret = null;
184
185 CompareResult _cr = CompareResult.AIsLargerB;
186
187 while ( iter != null ) {
188 T? iter_key = iter.Value;
189
190 _cr = m_cmp(iter_key, v);
191
192 if ( _cr == CompareResult.AIsLargerB )
193 iter = iter.Right;
194 else if ( _cr == CompareResult.AIsSmallerB )
195 iter = iter.Left;
196
197 else {
198 _ret = iter;
199 break;
200 }
201
202 }
203 return _ret; // not found

211 public BinaryTree<T>? FindNext ( BinaryTree<T>? n ) {
212 if ( n == null ) return null;
213
214 BinaryTree<T>? next = null;
215 BinaryTree<T>? _ret = null;
216
217 // Fall 1: rechter Teilbaum → Minimum rechts
218 if ( n.Right != null ) {
219 next = n.Right;
220 while ( next.Left != null )
221 next = next.Left;
222
223 _ret = next;
224 } else if ( (n.Parent != null) ) { // Fall 2: kein rechter Teilbaum → Parent-Logik
225
226 // Wenn wir linkes Kind sind → Parent ist der nächste
227 if ( n == n.Parent!.Left )
228 _ret = n.Parent;
229 else {
230 // Sonst hochlaufen, bis wir aus einem linken Kind kommen
231 next = n;
232
233 while ( next.Parent != null ) {
234 if ( next == next.Parent!.Right ) {
235 next = next.Parent;
236 } else {
237 _ret = next.Parent;
238 break;
239 }
240 }
241 }
242 }
243
244 // Fall 3: n ist Root und hat keinen rechten Teilbaum
245 // → Root hat keinen Next
246 return _ret;
247 }

265 public bool Erase ( BinaryTree<T>? node ) {
266 if ( node == null ) return false;
267
268 // Bestimme den tatsächlichen Knoten, der entfernt wird
269 BinaryTree<T>? toErase;
270
271 if ( node.Left == null || node.Right == null ) {
272 // Node hat höchstens ein Kind → direkt entfernen
273 toErase = node;
274 } else {
275 // Node hat zwei Kinder → successor suchen (Minimum im rechten Teilbaum)
276 toErase = node.Right;
277 while ( toErase!.Left != null )
278 toErase = toErase.Left;
279 }
280
281 // Bestimme das Kind des zu entfernenden Knotens
282 BinaryTree<T>? child = (toErase.Left != null ? toErase.Left : toErase.Right);
283
284 // Parent-Verknüpfung aktualisieren
285 if ( child != null )
286 child.Parent = toErase.Parent;
287
288 // toErase aus dem Baum entfernen
289 if ( toErase.Parent != null ) {
290 if ( toErase == toErase.Parent.Left )
291 toErase.Parent.Left = child;
292 else
293 toErase.Parent.Right = child;
294 } else {
295 // toErase war die Wurzel
296 Parent = child;
297 }
298
299 //Wert übertragen (nur wenn node != toErase)
300 if ( toErase != node )
301 node.Value = toErase.Value;
302
303
304 // Node freigeben
305 FreeNode(toErase, false);
306
307 return true;
308 }

348 public BinaryTree<T> RotateLeft ( BinaryTree<T> n ) {
349
350 BinaryTree<T>? rightChild = n.Right;
351 BinaryTree<T>? _ret = n;
352
353 if ( rightChild != null ) {
354 // Right child's left child wird n's right child
355 n.Right = rightChild.Left;
356 if ( n.Right != null )
357 n.Right.Parent = n;
358
359 // rightChild ersetzt n
360 rightChild.Parent = n.Parent;
361
362 if ( n.Parent == null ) {
363 // n war die Wurzel → rightChild wird neue Wurzel
364 Parent = rightChild;
365 } else {
366 if ( n == n.Parent.Left )
367 n.Parent.Left = rightChild;
368 else
369 n.Parent.Right = rightChild;
370 }
371
372 // n wird left child von rightChild
373 rightChild.Left = n;
374 n.Parent = rightChild;
375
376 _ret = rightChild;
377 }
378 return _ret;
379 }

456 public bool Traverse ( BinaryTree<T> n, TraverseFunc func, long depth, ref BinaryTree<T> k ) {
457 bool left = false;
458
459 if ( n.Parent != null ) {
460 left = n.Parent.Left == n;
461 }
462 if ( func(n, left, depth) == false ) {
463 k = n;
464 return false;
465 }
466
467 if ( n.Left != null ) {
468 if ( !Traverse(n.Left, func, depth + 1, ref k) ) return false;
469 }
470 if ( n.Right != null ) {
471 if ( !Traverse(n.Right, func, depth + 1, ref k) ) return false;
472 }
473
474 return true;
475 }

491 private long get_count () {
492 long c = 0;
493
494 var stack = new System.Collections.Generic.Stack<BinaryTree<T>>();
495 stack.Push(this);
496
497 while ( stack.Count > 0 ) {
498 BinaryTree<T> n = stack.Pop();
499 if ( n == null )
500 continue;
501
502 c++;
503
504 if ( n.Left != null )
505 stack.Push(n.Left);
506
507 if ( n.Right != null )
508 stack.Push(n.Right);
509 }
510
511 return c;
512 }
