34 public sealed class NetStdCompare<T> : ISimpleCompare<T> {
44 public Triple Compare ( Optional<T> a, Optional<T> b ) {
45 if ( a.IsNull || b.IsNull )
46 return triple.Nin;
47
48 int cmp = Comparer<T>.Default.Compare((T)a, (T)b);
49
50 if ( cmp < 0 ) return triple.True; // a < b
51 if ( cmp > 0 ) return triple.False; // a > b
52 return triple.Nin; // equal
