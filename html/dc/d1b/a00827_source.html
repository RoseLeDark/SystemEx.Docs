117 public static unsafe bool Check<T>(uint expectedUnmanagedSize) where T : unmanaged {
118 Type t = typeof(T);
119 long _sizeof = sizeof(T);
120 int _marshalSize = Marshal.SizeOf<T>();
121
122 // StructLayout must be Sequential
123 var attr = t.StructLayoutAttribute;
124 if ( attr == null || attr.Value != LayoutKind.Sequential )
125 throw new MissingStructLayoutSequentialException(t);
126
127 // sizeof(T) must match Marshal.SizeOf<T>()
128 if ( _marshalSize != _sizeof )
129 throw new SerializationException(
130 $"{t.Name} size mismatch: MarshalSize={_marshalSize}, sizeof={_sizeof}");
131
132 // Optional unmanaged size check
133 if ( expectedUnmanagedSize != 0 ) {
134 if ( _sizeof != expectedUnmanagedSize )
135 throw new SizeMismatchException(t, _sizeof, expectedUnmanagedSize);
136 }
137
138 return true;
139 }
140 }
