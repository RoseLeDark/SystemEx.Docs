57 public static int Clz ( ulong x ) {
58 if ( x == 0 ) return 64;
59
60 int n = 0;
61 if ( (x & 0xFFFFFFFF00000000) == 0 ) { n += 32; x <<= 32; }
62 if ( (x & 0xFFFF000000000000) == 0 ) { n += 16; x <<= 16; }
63 if ( (x & 0xFF00000000000000) == 0 ) { n += 8; x <<= 8; }
64 if ( (x & 0xF000000000000000) == 0 ) { n += 4; x <<= 4; }
65 if ( (x & 0xC000000000000000) == 0 ) { n += 2; x <<= 2; }
66 if ( (x & 0x8000000000000000) == 0 ) { n += 1; }
67
68 return n;
69 }
