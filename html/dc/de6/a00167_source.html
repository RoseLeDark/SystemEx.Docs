224 public static Half16b Add ( Half16b a, Half16b b ) {
225 if ( IsNaN(a) || IsNaN(b) ) return Half16b.NaN;
226 if ( IsInfinity(a) && IsInfinity(b) && a.Sign != b.Sign ) return Half16b.NaN;
227
228 if ( IsInfinity(a) ) return a;
229 if ( IsInfinity(b) ) return b;
230
231 if ( IsZero(a) ) return b;
232 if ( IsZero(b) ) return a;
233
234 ushort expA = a.Exponent;
235 ushort expB = b.Exponent;
236
237 uint mantA = (expA == 0) ? a.Mantissa : (uint)(a.Mantissa | a.HiddenBit);
238 uint mantB = (expB == 0) ? b.Mantissa : (uint)(b.Mantissa | b.HiddenBit);
239
240 int exp = expA;
241 int diff = expA - expB;
242
243 if ( diff > 0 )
244 mantB >>= diff;
245 else if ( diff < 0 ) {
246 mantA >>= -diff;
247 exp = expB;
248 }
249
250 uint mant;
251 ushort sign;
252
253 if ( a.Sign == b.Sign ) {
254 mant = mantA + mantB;
255 sign = (ushort)(a.Sign ? 1 : 0);
256 } else {
257 if ( mantA >= mantB ) {
258 mant = mantA - mantB;
259 sign = (ushort)(a.Sign ? 1 : 0);
260 } else {
261 mant = mantB - mantA;
262 sign = (ushort)(b.Sign ? 1 : 0);
263 }
264 }
265
266 if ( mant == 0 )
267 return new Half16b(0, 0, 0);
268
269 while ( (mant & a.HiddenBit) == 0 ) {
270 mant <<= 1;
271 exp--;
272 }
273
274 return Normalize(sign, exp, mant, a.HiddenBit);
275 }

299 public static Half16b Div ( Half16b a, Half16b b ) {
300 if ( IsNaN(a) || IsNaN(b) ) return Half16b.NaN;
301 if ( IsZero(b) ) return Half16b.NaN;
302 if ( IsZero(a) ) return Half16b.Zero;
303
304 if ( IsInfinity(a) && IsInfinity(b) ) return Half16b.NaN;
305
306 ushort sign = (ushort)((a.Sign ? 1 : 0) ^ (b.Sign ? 1 : 0));
307
308 if ( IsInfinity(a) )
309 return new Half16b(sign, 0xFF, 0);
310
311 if ( IsInfinity(b) )
312 return Half16b.Zero;
313
314 ushort expA = a.Exponent;
315 ushort expB = b.Exponent;
316
317 uint mantA = (expA == 0) ? a.Mantissa : (uint)(a.Mantissa | a.HiddenBit);
318 uint mantB = (expB == 0) ? b.Mantissa : (uint)(b.Mantissa | b.HiddenBit);
319
320 int exp = expA - expB + 127;
321
322 uint mant = (mantA << 13) / mantB;
323
324 while ( (mant & 0x80) == 0 ) {
325 mant <<= 1;
326 exp--;
327 }
328
329 return Normalize(sign, exp, mant, a.HiddenBit);
330 }

641 static Half16b Normalize ( ushort sign, int exp, uint mant, ushort hiddenbit ) {
642 if ( mant == 0 )
643 return new Half16b(sign, 0, 0);
644
645 while ( (mant & hiddenbit) == 0 ) {
646 mant <<= 1;
647 exp--;
648 }
649
650 if ( exp <= 0 )
651 return new Half16b(sign, 0, (ushort)(mant >> (1 - exp)));
652
653 if ( exp >= 0xFF )
654 return new Half16b(sign, 0xFF, 0);
655
656 ushort finalMant = (ushort)(mant & 0x7F);
657
658 return new Half16b(sign, (ushort)exp, finalMant);
659 }
660 static uint RoundToNearestEven ( uint mant ) {
661 uint guard = (mant >> 2) & 1;
662 uint round = (mant >> 1) & 1;
663 uint sticky = mant & 1;
664
665 uint add = 0;
666
667 if ( round != 0 ) {
668 if ( guard != 0 || sticky != 0 )
669 add = 1;
670 }
671
672 return (mant >> 3) + add;
673 }
