61 public static implicit operator BigDecimal ( double value ) {
62 if ( double.IsNaN(value) || double.IsInfinity(value) )
63 throw new OverflowException("Cannot convert NaN or Infinity to a BigDecimal.");
64
65 BigDecimal _ret = Zero;
66
67 if ( value != 0.0 ) {
68 long _raw = value.ToBytes(Endian.System).ToLong(Endian.System);
69
70 int _e = (int)((_raw >> 52) & 0x7FF);
71 long _man = _raw & 0xFFFFFFFFFFFFFL;
72
73 // Normalize subnormal numbers
74 if ( _e == 0 ) _e++;
75 else _man |= 0x10000000000000L;
76
77 // Convert exponent from biased IEEE‑754 to unbiased binary exponent
78 _e -= 1075;
79
80 BigInteger significand = new BigInteger(_man);
81 if ( _raw < 0 )
82 significand = -significand;
83
84 // Convert binary exponent to decimal exponent
85 if ( _e >= 0 ) {
86 _ret = new BigDecimal(significand * BigInteger.Pow(new BigInteger(2), _e), 0);

89 _ret = new BigDecimal(_uns, -_e);

126 public static explicit operator BigInteger ( BigDecimal value ) {
127 BigInteger _ret = BigInteger.Zero;
128
129 int _exp = value.Exponent;
130 var _man = value.Mantissa;
131
132 if ( _exp == 0 ) {
133 _ret = BigInteger.Zero;
134 } else if ( _exp < 0 ) {
135 var _nExp = -(long)_exp;
136 if ( _nExp > int.MaxValue && _nExp > 100_000_000 )
137 throw new OverflowException();
