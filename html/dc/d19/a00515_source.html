52 public class RCUStream<TStream> : Stream , IValueWriter, IValueReader
53 where TStream : Stream {
54
55 private TStream m_origStream;
56 private Epoch m_epochReads;
57 private ILock m_writerLock;
58
60 public RCUStreamState State
61 => ( (m_writerLock.IsHeld) ? RCUStreamState.Update : RCUStreamState.Current);
62
64 public override bool CanWrite
65 => m_epochReads.CanWrite && m_origStream.CanWrite;
66
68 public override bool CanSeek
69 => m_epochReads.CanWrite && m_origStream.CanSeek;
70
72 public override bool CanRead => m_origStream.CanRead;
73
75 public override long Length => m_origStream.Length;
76
78 public override long Position {
79
80 get {
81 using ( var _le = new UniqueEpoch(ref m_epochReads) ) {
82 return m_origStream.Position;
83 }
84 } set {
85 using ( var lk = new ScopedLock<ILock>(ref m_writerLock) ) {
86 if ( m_epochReads )
87 throw new InvalidOperationException("Readers active");
88
89 m_origStream.Position = value;
90 }
91 }
