318 public static FloatUE4M3 Add ( FloatUE4M3 a, FloatUE4M3 b ) {
319 if ( IsNaN(a) || IsNaN(b) ) return FloatUE4M3.NaN;
320 if ( IsZero(a) ) return b;
321 if ( IsZero(b) ) return a;
322
323 Fast_Byte expA = a.Exponent;
324 Fast_Byte expB = b.Exponent;
325
326 Fast_Int mantA = (expA == 0 ? a.Mantissa : (a.Mantissa | 0x08));
327 Fast_Int mantB = (expB == 0 ? b.Mantissa : (b.Mantissa | 0x08));
328
329 Fast_Byte exp = expA;
330 Fast_Int diff = expA - expB;
331
332 if ( diff > 0 )
333 mantB >>= diff;
334 else if ( diff < 0 ) {
335 mantA >>= -diff;
336 exp = expB;
337 }
338
339 Fast_Int mant;
340
341 mant = mantA + mantB;
342
343 if ( mant == 0 )
344 return FloatUE4M3.Zero;
345
346 while ( (mant & 0x08) == 0 ) {
347 mant <<= 1;
348 exp--;
349 }
350
351 if ( exp <= 0 )
352 return FloatUE4M3.Zero;
353
354 if ( exp >= 0x0F )
355 return FloatUE4M3.NaN;
356
357 Fast_Byte finalMant = (byte)(mant & 0x07);
358
359 return new FloatUE4M3(0, (byte)exp.Value, finalMant.Value);
360 }

362 public static FloatUE4M3 Sub ( FloatUE4M3 a, FloatUE4M3 b ) {
363 // 1. Sonderfälle auf Bit-Ebene abfangen
364 if ( IsNaN(a) || IsNaN(b) ) return NaN;
365
366 // Unendlich-Regeln
367 if ( IsInfinity(a) ) return IsInfinity(b) ? NaN : PositiveInfinity;
368 if ( IsInfinity(b) ) return Zero; // Sättigung: Irgendwas minus Unendlich wird zu 0 gekappt
369 if ( IsZero(b) ) return a;
370 if ( IsZero(a) ) return Zero; // Sättigung: 0 minus Irgendwas wird zu 0 gekappt
371
372 // Wenn b größer oder gleich a ist, laufen wir garantiert unter oder gegen Null
373 if ( a <= b ) return Zero;
374
375 Fast_Byte expA = a.Exponent;
376 Fast_Byte expB = b.Exponent;
377
378 // Hidden Bit hinzufügen
379 Fast_Int mantA = (expA == 0 ? a.Mantissa : (a.Mantissa | 0x08));
380 Fast_Int mantB = (expB == 0 ? b.Mantissa : (b.Mantissa | 0x08));
381
382 Fast_Int realExpA = expA == 0 ? 1 : expA;
383 Fast_Int realExpB = expB == 0 ? 1 : expB;
384
385 Fast_Int finalExp = realExpA;
386
387 // Nutzen deines größeren Registers, damit beim Shiften nichts verloren geht
388 Fast_UShort shiftedA = (ushort)(mantA << 5);
389 Fast_UShort shiftedB = (ushort)(mantB << 5);
390 Fast_UShort resMant = 0;
391
392 // Mantissen auf denselben Exponenten ausrichten
393 if ( realExpA >= realExpB ) {
394 var shift = realExpA - realExpB;
395 shiftedB >>= (int)shift;
396 finalExp = realExpA;
397 } else {
398 var shift = realExpB - realExpA;
399 shiftedA >>= (int)shift;
400 finalExp = realExpB;
401 }
402
403 // Da a > b garantiert ist, ist shiftedA immer größer als shiftedB
404 resMant = (ushort)(shiftedA - shiftedB);
405
406 if ( resMant == 0 ) return Zero;
407
408 // Renormalisierung im Shiftraster (Wir suchen das führende Bit)
409 while ( resMant >= 0x200 ) {
410 resMant >>= 1;
411 finalExp++;
412 }
413 while ( resMant < 0x100 && finalExp > 1 ) {
414 resMant <<= 1;
415 finalExp--;
416 }
417
418 // Zurück in das 3-Bit-Raster stutzen
419 resMant >>= 5;
420 resMant &= 0x07; // Hidden Bit löschen
421
422 // Überlauf zu Infinity prüfen
423 if ( finalExp >= 0x0F ) return PositiveInfinity;
424
425 // Unterlauf zu Subnormal prüfen
426 if ( finalExp <= 0 ) return new FloatUE4M3(0, 0, (byte)resMant);
427
428 return new FloatUE4M3(0, (byte)finalExp, (byte)resMant);
429 }

434 public static FloatUE4M3 Mul ( FloatUE4M3 a, FloatUE4M3 b ) {
435 if ( IsNaN(a) || IsNaN(b) ) return FloatUE4M3.NaN;
436 if ( IsZero(a) || IsZero(b) ) return FloatUE4M3.Zero;
437
438 Fast_Byte exp = a.Exponent + b.Exponent - a.ExponentBias;
439
440 Fast_Int mantA = (a.Exponent == 0 ? a.Mantissa : (a.Mantissa | 0x08));
441 Fast_Int mantB = (b.Exponent == 0 ? b.Mantissa : (b.Mantissa | 0x08));
442
443 Fast_Int mant = mantA * mantB;
444
445 while ( (mant & 0x40) != 0 ) {
446 mant >>= 1;
447 exp++;
448 }
449
450 mant >>= 3;
451
452 if ( exp <= 0 )
453 return FloatUE4M3.Zero;
454
455 if ( exp >= 0x0F )
456 return FloatUE4M3.NaN;
457
458 byte finalMant = (byte)(mant & 0x07);
459
460 return new FloatUE4M3(0, exp.Value, finalMant);
461 }

465 public static FloatUE4M3 Div ( FloatUE4M3 a, FloatUE4M3 b ) {
466 if ( IsNaN(a) || IsNaN(b) ) return FloatUE4M3.NaN;
467 if ( IsZero(b) ) return FloatUE4M3.NaN;
468 if ( IsZero(a) ) return FloatUE4M3.Zero;
469
470
471 Fast_Byte exp = a.Exponent - b.Exponent + a.ExponentBias;
472
473 Fast_Int mantA = (a.Exponent == 0 ? a.Mantissa : (a.Mantissa | 0x08));
474 Fast_Int mantB = (b.Exponent == 0 ? b.Mantissa : (b.Mantissa | 0x08));
475
476 Fast_Int mant = (mantA << 6) / mantB;
477
478 while ( (mant & 0x08) == 0 ) {
479 mant <<= 1;
480 exp--;
481 }
482
483 if ( exp <= 0 )
484 return FloatUE4M3.Zero;
485
486 if ( exp >= 0x0F )
487 return FloatUE4M3.NaN;
488
489 byte finalMant = (byte)(mant & 0x07);
490
491 return new FloatUE4M3(0, exp.Value, finalMant);
492 }

679 internal static byte Encode ( byte sign, byte exponent, byte mantissa ) {
680 Fast_Byte b = 0;
681
682 b.At(7, sign); // Signbit
683 b.At(3, (byte)((exponent >> 0) & 1));
684 b.At(4, (byte)((exponent >> 1) & 1));
685 b.At(5, (byte)((exponent >> 2) & 1));
686 b.At(6, (byte)((exponent >> 3) & 1));
687
688 b.At(0, (byte)((mantissa >> 0) & 1));
689 b.At(1, (byte)((mantissa >> 1) & 1));
690 b.At(2, (byte)((mantissa >> 2) & 1));
691
692 return b.Value;
693 }
