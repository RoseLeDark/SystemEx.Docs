119 public static BigDecimal operator + ( BigDecimal a, BigDecimal b ) {
120 // If one is zero → fast path
121 if ( a.IsZero ) return b;
122 if ( b.IsZero ) return a;
123
124 // Align exponents
125 if ( a.Exponent == b.Exponent )
126 return new BigDecimal(a.Mantissa + b.Mantissa, a.Exponent).Normalize();
127
128 if ( a.Exponent > b.Exponent ) {
129 int diff = a.Exponent - b.Exponent;
130 BigInteger bm = b.Mantissa * PowerOfTen(diff);
131 return new BigDecimal(a.Mantissa + bm, b.Exponent).Normalize();
132 } else {
133 int diff = b.Exponent - a.Exponent;
134 BigInteger am = a.Mantissa * PowerOfTen(diff);
135 return new BigDecimal(am + b.Mantissa, a.Exponent).Normalize();
136 }
137 }

178 public static BigDecimal operator / ( BigDecimal a, BigDecimal b ) {
179 if ( b.IsZero )
180 throw new DivideByZeroException();
181
182 if ( a.IsZero )
183 return Zero;
184
185 // Adjust mantissas
186 BigInteger am = a.Mantissa;
187 BigInteger bm = b.Mantissa;
188
189 // Try exact division
190 BigInteger q = BigInteger.DivRem(am, bm, out BigInteger r);
191
192 if ( !r.IsZero )
193 throw new ArithmeticException("Non-exact division in BigDecimal.");
194
195 int e = a.Exponent - b.Exponent;
196
197 return new BigDecimal(q, e).Normalize();
198 }

405 private static BigInteger PowerOfTen ( long n ) {
406 if ( n == 0 ) return BigInteger.One;
407 if ( n < 0 ) throw new ArgumentOutOfRangeException(nameof(n), "Exponent must be non-negative.");
408
409 // Efficient: use BigInteger.Pow

428 public BigDecimal Round ( int digits, RoundMode mode ) {
429 // If exponent >= digits → already exact
430 if ( Exponent >= digits )
431 return this;
432
433 int shift = digits - Exponent;
434
435 // Shift mantissa right: divide by 10^shift
436 BigInteger pow = PowerOfTen(shift);
437
438 BigInteger div = BigInteger.DivRem(Mantissa, pow, out BigInteger rem);
439
440 if ( rem.IsZero )
441 return new BigDecimal(div, digits).Normalize();
442
443 bool negative = Mantissa.Sign < 0;
444 BigInteger absRem = BigInteger.Abs(rem);
445 BigInteger half = pow >> 1; // 10^shift / 2
446
447 bool roundUp = false;
448
449 switch ( mode ) {
450 case BigDecimal.RoundMode.ToEven:
451 if ( absRem > half ) roundUp = true;
452 else if ( absRem == half && !div.IsEven ) roundUp = true;
453 break;
454
455 case BigDecimal.RoundMode.AwayFromZero:
456 roundUp = true;
457 break;
458
459 case BigDecimal.RoundMode.TowardZero:
460 roundUp = false;
461 break;
462
463 case BigDecimal.RoundMode.Floor:
464 if ( negative ) roundUp = true;
465 break;
466
467 case BigDecimal.RoundMode.Ceiling:
468 if ( !negative ) roundUp = true;
469 break;
470 }
471
472 if ( roundUp )
473 div += negative ? -1 : 1;
474
475 return new BigDecimal(div, digits).Normalize();
476 }
