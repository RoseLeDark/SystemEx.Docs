64 public long Find ( ref TContainer container, ICompared<T> comp, T value ) {
65 long _arCount= container.Count;
66 long _ret = 0;
67
68 if ( _arCount> 0 ) {
69 // find the smallest Fibonacci number that equals or is greater than the array length
70 long _fbPrevB = 0;
71 long _fbNPrev= 1;
72 long _fbNum = _fbNPrev;
73
74 while ( _fbNum <= _arCount) {
75 _fbPrevB = _fbNPrev;
76 _fbNPrev= _fbNum;
77 _fbNum = _fbPrevB + _fbNPrev;
78 }
79
80 // offset to drop the left part of the array
81 long offset = -1;
82
83 while ( _fbNum > 1 ) {
84 var index = System.Math.Min(offset + _fbPrevB, _arCount- 1);
85
86 switch ( comp.Compare(container.ElementAt(index), value) ) {
87 case CompareResult.AIsLargerB:
88 _fbNum = _fbNPrev;
89 _fbNPrev= _fbPrevB;
90 _fbPrevB = _fbNum - _fbNPrev;
91 offset = index;
92 break;
93
94 // reject approximately 2/3 of the existing array behind
95 // by moving Fibonacci numbers
96 case CompareResult.AIsSmallerB:
97 _fbNum = _fbPrevB;
98 _fbNPrev= _fbNPrev- _fbPrevB;
99 _fbPrevB = _fbNum - _fbNPrev;
100 break;
101 default:
102 _ret = index; break;
103 }
104 }
105
106
107 // check the last element
108 Optional<T> _it = container.ElementAt(_arCount - 1);
109
110 if ( _it.IsSome ) {
111 if ( _fbNPrev == 1 && comp.Compare(value, _it) == CompareResult.Equal ) {
112 _ret++;
113 }
114 }
115 }
116 return _ret;
117 }
