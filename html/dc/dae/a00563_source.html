438 internal static Half16 Normalize ( ushort sign, int exp, uint mant, ushort hiddenbit ) {
439 if ( mant == 0 ) return new Half16(sign, 0, 0);
440
441 while ( (mant & hiddenbit) == 0 ) {
442 mant <<= 1;
443 exp--;
444 }
445
446 if ( exp <= 0 )
447 return new Half16(sign, 0, (ushort)(mant >> (1 - exp)));
448
449 if ( exp >= 31 )
450 return new Half16(sign, 0x1F, 0);
451
452 ushort finalMant = (ushort)(mant & 0x3FF);
453
454 return new Half16(sign, (ushort)exp, finalMant);
455 }

460 static uint RoundToNearestEven ( uint mant ) {
461 uint guard = (mant >> 2) & 1;
462 uint round = (mant >> 1) & 1;
463 uint sticky = mant & 1;
464
465 uint add = 0;
466
467 if ( round != 0 ) {
468 if ( guard != 0 || sticky != 0 )
469 add = 1;
470 }
471
472 return (mant >> 3) + add;
473 }

487 public static Half16 Mul ( Half16 a, Half16 b ) {
488 if ( IsNaN(a) || IsNaN(b) ) return Half16.NaN;
489 if ( IsZero(a) || IsZero(b) ) return Half16.Zero;
490
491 ushort sign = (ushort)((a.Sign ? 1 : 0) ^ (b.Sign ? 1 : 0) );
492
493 if ( IsInfinity(a) || IsInfinity(b) ) {
494 return new Half16(sign, 0x1F, 0);
495 }
496
497 ushort expA = a.Exponent;
498 ushort expB = b.Exponent;
499
500 uint mantA = (expA == 0) ? a.Mantissa : (uint)(a.Mantissa | a.HiddenBit);
501 uint mantB = (expB == 0) ? b.Mantissa: (uint)(b.Mantissa | b.HiddenBit);
502
503 int exp = expA + expB - 15;
504
505 uint mant = mantA * mantB; // bis zu 20 Bits
506
507 // Normalisieren
508 while ( (mant & 0x80000) != 0 ) {
509 mant >>= 1;
510 exp++;
511 }
512
513 mant >>= 10; // zurück auf 11 Bits (leading + 10 mantissa)
514
515 return Normalize(sign, exp, mant, a.HiddenBit);
516 }

532 public static Half16 Add ( Half16 a, Half16 b ) {
533 if ( IsNaN(a) || IsNaN(b) ) return Half16.NaN;
534 if ( IsInfinity(a) && IsInfinity(b) && a.Sign != b.Sign ) return Half16.NaN;
535
536 // Sonderfälle
537 if ( IsInfinity(a) ) return a;
538 if ( IsInfinity(b) ) return b;
539
540 if ( IsZero(a) ) return b;
541 if ( IsZero(b) ) return a;
542
543 ushort expA = a.Exponent;
544 ushort expB = b.Exponent;
545
546 uint mantA = (expA == 0) ? a.Mantissa : (uint)(a.Mantissa | a.HiddenBit);
547 uint mantB = (expB == 0) ? b.Mantissa: (uint)(b.Mantissa | b.HiddenBit);
548
549 int exp = expA;
550
551 int diff = expA - expB;
552
553 if ( diff > 0 ) {
554 mantB >>= diff;
555 } else if ( diff < 0 ) {
556 mantA >>= -diff;
557 exp = expB;
558 }
559
560 uint mant;
561 ushort sign;
562
563 if ( a.Sign == b.Sign ) {
564 mant = mantA + mantB;
565 sign = (ushort)(a.Sign ? 1 : 0);
566 } else {
567 if ( mantA >= mantB ) {
568 mant = mantA - mantB;
569 sign = (ushort)(a.Sign ? 1 : 0);
570 } else {
571 mant = mantB - mantA;
572 sign = (ushort)(b.Sign ? 1 : 0);
573 }
574 }
575
576 if ( mant == 0 ) return new Half16(0, 0, 0);
577
578 while ( (mant & 0x400) == 0 ) {
579 mant <<= 1;
580 exp--;
581 }
582
583 return Normalize(sign, exp, mant, a.HiddenBit);
584 }

601 public static Half16 Div(Half16 a, Half16 b) {
602 if ( IsNaN(a) || IsNaN(b) ) return Half16.NaN;
603 if ( IsZero(b) ) return Half16.NaN;
604 if ( IsZero(a) ) return Half16.Zero;
605
606 if ( IsInfinity(a) && IsInfinity(b) ) return Half16.NaN;
607
608 ushort sign = (ushort)((a.Sign ? 1 : 0) ^ (b.Sign ? 1 : 0) );
609
610 if ( IsInfinity(a) ) {
611 return new Half16(sign, 0x1F, 0);
612 }
613 if ( IsInfinity(b) ) return Half16.Zero;
614
615 ushort expA = a.Exponent;
616 ushort expB = b.Exponent;
617
618 uint mantA = (expA == 0) ? a.Mantissa : (uint)(a.Mantissa | a.HiddenBit);
619 uint mantB = (expB == 0) ? b.Mantissa: (uint)(b.Mantissa | b.HiddenBit);
620
621 int exp = expA - expB + 15;
622
623 uint mant = (mantA << 13) / mantB;
624
625 while ( (mant & 0x400) == 0 ) {
626 mant <<= 1;
627 exp--;
628 }
629
630 return Normalize(sign, exp, mant, a.HiddenBit);
631 }
