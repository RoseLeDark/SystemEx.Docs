178 public ColorHDR Lerp(ColorHDR value, float amount) {
179 // Quadratische Kurve
180 float q = amount * amount;
181
182 // Hue shortest-path interpolation
183 float dh = value.m_hue - m_hue;
184 dh = (dh + 540f) % 360f - 180f;
185
186 return new ColorHDR(
187 ClampHue(m_hue + dh * q),
188 m_saturation + (value.m_saturation - m_saturation) * q,
189 m_value + (value.m_value - m_value) * q
190 );
191 }
