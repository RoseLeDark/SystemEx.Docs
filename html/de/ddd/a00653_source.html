143 private static void Shuffle ( ref uint a, ref uint b, ref uint c, ref uint d,
144 ref uint e, ref uint f, ref uint g, ref uint h ) {
145 a ^= b << 11; d += a; b += c;
146 b ^= c >> 2; e += b; c += d;
147 c ^= d << 8; f += c; d += e;
148 d ^= e >> 16; g += d; e += f;
149 e ^= f << 10; h += e; f += g;
150 f ^= g >> 4; a += f; g += h;
151 g ^= h << 8; b += g; h += a;
152 h ^= a >> 9; c += h; a += b;
153 }
