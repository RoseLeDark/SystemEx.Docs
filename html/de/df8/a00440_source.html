710 public Pair<bool, long> CopyTo ( uint sourceOffset, FixedVector<T> destination, ulong destinationOffset, ulong count ) {
711
712 long src = (long)sourceOffset;
713 long dst = (long)destinationOffset;
714
715 if ( src > Length ) src = (long)Length;
716
717 long toCopy = System.Math.Min((long)count,

719 System.Math.Max(0, destination.Length - dst)));

751 public Pair<bool, long> CopyFrom ( FixedVector<T> source, ulong sourceOffset, ulong destinationOffset, ulong count ) {
752 long src = (long)sourceOffset;
753 long dst = (long)destinationOffset;
754
755 // Clamp source offset
756 if ( src > source.Length )
757 src = source.Length;
758
759 // Clamp destination offset
760 if ( dst > Length )
761 dst = Length;
762
763 // Compute how much can actually be copied
764 long toCopy = System.Math.Min((long)count,
765 System.Math.Min(
766 System.Math.Max(0, source.Length - src),
767 System.Math.Max(0, Length - dst)
768 ));
769
770 // Nothing fits → return (false, 0)
771 if ( toCopy <= 0 )
772 return new Pair<bool, long>(false, 0);
773
774 // Perform the copy
775 Buffer.LongCopy<T>(source.m_elements, src, m_elements, dst, toCopy);
776
777 // Update logical length only if we wrote beyond current end
778 long end = dst + toCopy;
779 if ( end > m_index )
780 m_index = end;
781
782 return new Pair<bool, long>(true, toCopy);
783 }
