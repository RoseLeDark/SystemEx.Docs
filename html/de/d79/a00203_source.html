197 public override string ToString () {
198 if ( m_bMantissa.IsZero )
199 return "0";
200
201 var n = Normalize();
202 BigInteger man = n.Mantissa;
203 int exp = n.Exponent;
204
205 if ( exp == 0 )
206 return man.ToString();
207
208 if ( BigInteger.Abs(man) < 10 )
209 return $"{man}e{exp}";
210
211 string s = BigInteger.Abs(man).ToString();
212 bool neg = man.Sign < 0;
213
214 char first = s[0];
215 string rest = s.Substring(1);
216
217 int adjExp = exp + (s.Length - 1);
218
219 if ( rest.Length == 0 )
220 return neg ? $"-{first}e{adjExp}" : $"{first}e{adjExp}";
221
222 return neg
223 ? $"-{first}.{rest}e{adjExp}"
224 : $"{first}.{rest}e{adjExp}";
225 }

241 public string ToEngineeringString () {
242 if ( m_bMantissa.IsZero )
243 return "0";
244
245 var n = Normalize();
246 BigInteger man = n.Mantissa;
247 int exp = n.Exponent;
248
249 string s = BigInteger.Abs(man).ToString();
250 bool neg = man.Sign < 0;
251
252 int sciExp = exp + (s.Length - 1);
253 int engExp = sciExp - (sciExp % 3);
254 int shift = sciExp - engExp;
255
256 BigInteger engMan = man;
257 if ( shift > 0 )
258 engMan = man / BigInteger.Pow(10, shift);
259
260 string engStr = BigInteger.Abs(engMan).ToString();
261
262 return neg
263 ? $"-{engStr}e{engExp}"
264 : $"{engStr}e{engExp}";
265 }

272 public static BigDecimal ParseScientific ( string s ) {
273 if ( string.IsNullOrWhiteSpace(s) )
274 throw new FormatException("Input string is empty.");
275
276 s = s.Trim();
277
278 int idx = s.IndexOfAny(new[] { 'e', 'E' });
279 if ( idx < 0 )
280 throw new FormatException("Scientific notation requires an exponent.");
281
282 string mantissaPart = s.Substring(0, idx);
283 string exponentPart = s.Substring(idx + 1);
284
285 if ( !int.TryParse(exponentPart, out int exponent) )
286 throw new FormatException("Invalid exponent in scientific notation.");
287
288 bool neg = mantissaPart.StartsWith("-");
289 if ( neg )
290 mantissaPart = mantissaPart.Substring(1);
291
292 int decimalPos = mantissaPart.IndexOf('.');
293 BigInteger mantissa;
294 int scale = 0;
295
296 if ( decimalPos >= 0 ) {
297 string digits = mantissaPart.Replace(".", "");
298 mantissa = BigInteger.Parse(digits);
299 scale = -(mantissaPart.Length - decimalPos - 1);
300 } else {
301 mantissa = BigInteger.Parse(mantissaPart);
302 }
303
304 if ( neg )
305 mantissa = -mantissa;
306
307 int finalExp = exponent + scale;
308 return new BigDecimal(mantissa, finalExp).Normalize();
309 }

419 public string ToUSString ( USUnit unit ) {
420 BigDecimal factor = unit switch
421 {
422 USUnit.Inch => Inch,
423 USUnit.Foot => Foot,
424 USUnit.Yard => Yard,
425 USUnit.Mile => Mile,
426 USUnit.Ounce => Ounce,
427 USUnit.Pound => Pound,
428 USUnit.Cup => Cup,
429 USUnit.Pint => Pint,
430 USUnit.Quart => Quart,
431 USUnit.Gallon => Gallon,
432 USUnit.PSI => PSI,
433 USUnit.BTU => BTU,
434 USUnit.Horsepower => Horsepower,
435 _ => One
436 };
437
438 BigDecimal v = this / factor;
439 return $"{v.ToEngineeringString()} {unit}";
440 }
