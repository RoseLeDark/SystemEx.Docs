88 public T[] Extract ( long index ) {
89 T[] _ret = Array.Empty<T>();
90
91 long count = m_pKeys.Count;
92
93 if ( index >= 0 && index < count ) {
94 long extractCount = count - index;
95 _ret = new T[extractCount];
96
97 // copy extracted elements
98 for ( long i = 0 ; i < extractCount ; i++ ) {
99 var itm = m_pKeys.ElementAt(index + i);
100 if ( itm.IsSome ) {
101 _ret[i] = itm.Value!;
102 }
103 }
104
105 // remove extracted elements from container
106 for ( long i = count - 1 ; i >= index ; i-- ) {
107 m_pKeys.Erase(i);
108 }
109 }

269 public void Merge ( VectorUnorderedSet<T, TContainer> source ) {
270 long count = source.m_pKeys.Count;
271
272 for ( long i = 0 ; i < count ; i++ ) {
273 Optional<T> value = source.m_pKeys.ElementAt(i);
274 if ( value.IsNull)
275 continue;
276
277 // check if this set already contains the value
278 long idx = m_finder.First(value.Value!);
279
280 if ( idx == -1 ) {
281 // extract from source
282 T[] extracted = source.Extract(i);
283
