499 public void Splice ( ref Node<T> first, ref Node<T> last ) {
500 // 1. Hole die alten Nachbarn
501 Node<T>? before = this.Prev;
502 Node<T>? after = this.Next;
503
504 // 2. Entferne 'this' aus seiner Position
505 if ( before != null )
506 before.Next = after;
507 if ( after != null )
508 after.Prev = before;
509
510 // 3. Füge [first..last] an die Stelle ein, wo 'this' war
511
512 // Verbinde vor dem Block
513 if ( before != null )
514 before.Next = first;
515 first.Prev = before;
516
517 // Verbinde nach dem Block
518 if ( after != null )
519 after.Prev = last;
520 last.Next = after;
521
522 // 4. 'this' ist jetzt isoliert
523 this.Prev = null;
524 this.Next = null;

581 public void Swap ( ref Node<T> other ) {
582 if ( this == other )
583 return;
584
585 // Backup A
586 Node<T>? aPrev = this.Prev;
587 Node<T>? aNext = this.Next;
588 T? aValue = this.Value;
589
590 // Backup B
591 Node<T>? bPrev = other.Prev;
592 Node<T>? bNext = other.Next;
593 T? bValue = other.Value;
594
595 // Copy B into A
596 this.Prev = bPrev;
597 this.Next = bNext;
598 this.Value = bValue;
599
600 // Copy A into B
601 other.Prev = aPrev;
602 other.Next = aNext;
603 other.Value = aValue;
604
605 // Fix neighbors of A (null‑sicher)
606 if ( this.Prev != null )
607 this.Prev.Next = this;
608
609 if ( this.Next != null )
610 this.Next.Prev = this;
611
612 // Fix neighbors of B (null‑sicher)
613 if ( other.Prev != null )
614 other.Prev.Next = other;
615
616 if ( other.Next != null )
617 other.Next.Prev = other;
618 }
