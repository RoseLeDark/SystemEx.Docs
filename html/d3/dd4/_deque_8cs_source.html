32 public class Deque<T> {
36 private T[] m_elements;
37
41 private int m_count;
42
46 public int Count => m_count;
47
48
52 public int Size => m_elements.Length;
53
57 public bool IsEmpty => m_count == 0;
58
62 public bool IsFull => m_count == m_elements.Length;
63
67 public T Front => m_elements[0];
68
72 public T End => m_elements[m_count - 1];
73
78 public Deque(int size) {
79 m_elements = new T[size];
80 m_count = 0;
81 }
82
87 public void PushBack(T value) {
88 if ( IsFull ) return;
89 m_elements[m_count++] = value;
90 }
91
97 public bool PopBack(ref T value) {
98 if ( IsEmpty ) return false;
99 value = m_elements[m_count - 1];
100 m_count--;
101 return true;
102 }
103
109 public void PushFront(T value) {
110 if ( IsFull ) return;
111
112 // Shift all elements to the right
113 for ( int i = m_count; i > 0; i-- )
114 m_elements[i] = m_elements[i - 1];
115
116 m_elements[0] = value;
117 m_count++;
118 }
119
126 public bool PopFront(ref T value) {
127 if ( IsEmpty ) return false;
128
129 value = m_elements[0];
130
131 // Shift all elements to the left
132 for ( int i = 0; i < m_count - 1; i++ )
133 m_elements[i] = m_elements[i + 1];
134
135 m_count--;
136 return true;
137 }
138
142 public void Clear() {
143 m_count = 0;
144 }
145
157 public FlexSpan<T> AsFlexSpan ( FlexSpanMode mode = FlexSpanMode.System ) {
158 return new FlexSpan<T>(ref m_elements!, 0, m_count, mode);
159 }
160 }
