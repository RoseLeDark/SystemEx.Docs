196 private static void Permutation ( uint[] x, int q, int state, int cols, int rR, byte[,] shift, uint[] ColumnConstant ) {
197 uint[] tmp = new uint[cols];
198 uint constant = 0;
199
200 for ( int round = 0 ; round < rR ; round++ ) {
201 constant += 0x01010101u;
202
203 // AddRoundConstant
204 if ( q == 0 ) {
205 for ( int j = 0 ; j < ColumnConstant.Length ; j++ )
206 x[j] ^= ColumnConstant[j] ^ constant;
207 } else {

212 x[state - ColumnConstant.Length + j] ^= ColumnConstant[j] ^ constant;
213 }

217 int baseIndex = row * cols;

227 x[baseIndex + col] = (x[baseIndex + col] & 0xFFFFFF00u) | Ss[b];
228 }

233 for ( int r = 0 ; r < 8 ; r++ )

235 mat[r, c] = x[r * cols + c];
