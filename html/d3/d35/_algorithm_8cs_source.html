179 public static void Move<T>(T[] src, uint srcIndex, T[] dst, uint dstIndex, uint count) {
180 if ( srcIndex + count > src.Length )
181 throw new ArgumentException("src out of range");
182 if ( dstIndex + count > dst.Length )
183 throw new ArgumentException("dst out of range");
184
185 if ( src == dst && dstIndex > srcIndex ) {
186 for ( uint i = count; i > 0; i-- )
187 dst[dstIndex + i - 1] = src[srcIndex + i - 1];
188 } else {
189 for ( uint i = 0; i < count; i++ )
190 dst[dstIndex + i] = src[srcIndex + i];
191 }

235 public static bool Equal<T>(T[] a, T[] b, CompFunc<T> cmp) {
236 if ( a.Length != b.Length ) return false;

239 if ( cmp(a[i], b[i]) != CompareResult.Equal )
240 return false;

248 public static bool LexicographicalCompare<T>(T[] a, T[] b, CompFunc<T> cmp) {
249 int n = System.Math.Min(a.Length, b.Length);
250
251 for ( int i = 0; i < n; i++ ) {
252 var r = cmp(a[i], b[i]);
253 if ( r == CompareResult.AIsSmallerB ) return true;
254 if ( r == CompareResult.AIsLargerB ) return false;
255 }
256
257 return a.Length < b.Length;
258 }

262 public static void QuickSort<T> ( T[] items, CompFunc<T> cmp ) {
263 if ( items.Length <= 1 ) return;
264
265 int low = 0;
266 int high = items.Length - 1;
267
268 while ( true ) {
269 int i = low;
270 int j = high;
271 T pivot = items[(low + high) >> 1];
272
273 do {
274 while ( cmp(items[i], pivot) == CompareResult.AIsSmallerB ) i++;
275 while ( cmp(pivot, items[j]) == CompareResult.AIsSmallerB ) j--;
276
277 if ( i <= j ) {
278 Algorithm.Swap(ref items[i], ref items[j]);
279 i++;
280 j--;
281 }
282 } while ( i <= j );
283
284 if ( low < j ) {
285 QuickSortRange(items, low, j, cmp);
286 }
287
288 if ( i < high ) {
289 low = i;
290 } else break;
291 }
292 }

294 private static void QuickSortRange<T> ( T[] items, int low, int high, CompFunc<T> cmp ) {
295 int i = low;
296 int j = high;
297 T pivot = items[(low + high) >> 1];
298
299 while ( true ) {
300 while ( cmp(items[i], pivot) == CompareResult.AIsSmallerB ) i++;
301 while ( cmp(pivot, items[j]) == CompareResult.AIsSmallerB ) j--;
302
303 if ( i <= j ) {
304 Algorithm.Swap(ref items[i], ref items[j]);
305 i++;
306 j--;
307 }
308
309 if ( i > j ) break;
310 }
311
312 if ( low < j ) QuickSortRange(items, low, j, cmp);
313 if ( i < high ) QuickSortRange(items, i, high, cmp);
314 }
318 public static void HeapSort<T> ( T[] items, CompFunc<T> cmp ) {
319 int n = items.Length;
320
321 for ( int k = n / 2 ; k > 0 ; k-- )
322 DownHeap(items, k, n, cmp);
323
324 while ( n > 1 ) {
325 Algorithm.Swap(ref items[0], ref items[n - 1]);
326 n--;
327 DownHeap(items, 1, n, cmp);
328 }
329 }

331 private static void DownHeap<T> ( T[] items, int k, int n, CompFunc<T> cmp ) {
332 T temp = items[k - 1];
333
334 while ( k <= n / 2 ) {
335 int child = 2 * k;
336
337 if ( child < n && cmp(items[child - 1], items[child]) == CompareResult.AIsSmallerB )
338 child++;
339
340 if ( cmp(temp, items[child - 1]) == CompareResult.AIsSmallerB ) {
341 items[k - 1] = items[child - 1];
342 k = child;
343 } else break;
344 }
345
346 items[k - 1] = temp;
347 }

352 public static void InsertionSort<T> ( T[] items, CompFunc<T> cmp ) {
353 for ( int i = 0 ; i < items.Length ; i++ ) {
354 T t = items[i];
355 int j = i;
356
357 while ( j > 0 && cmp(t, items[j - 1]) == CompareResult.AIsSmallerB ) {
358 items[j] = items[j - 1];
359 j--;
360 }
361
362 items[j] = t;
363 }
