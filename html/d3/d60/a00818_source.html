153 public static int MaskRange(int start, int length) {
154 if ( length <= 0 ) return 0;
155 if ( start < 0 || start > 31 ) return 0;
156 if ( length >= 32 ) return unchecked((int)0xFFFFFFFF);
157
158 int mask = (1 << length) - 1;
159 return mask << start;
160 }
