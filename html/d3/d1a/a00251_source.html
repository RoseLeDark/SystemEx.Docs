347 public bool PushFront ( Pair<T, TU> entry ) {
348 if( ContainsKey(entry.First) ) return false;
349
350 // If full, attempt to grow
351 if ( m_index + 1 >= Length ) {
352 if ( AutoGrow ) {
353 if ( !Resize(Length + GrowSize) )
354 return false;
355 } else {
356 return false;
357 }
358 }
359
360 // Shift all valid elements one slot to the right
361 // m_index is the last valid index, so we shift [0 .. m_index]
362 for ( long i = m_index ; i >= 0 ; i-- ) {
363 m_elements[i + 1] = m_elements[i];
364 m_state[i + 1] = m_state[i];
365 }
366
367 // Insert new element at the front
368 m_elements[0] = entry;
369 m_state[0] = 1;
370
371 // Increase logical count
372 m_index++;
373
374 return true;
375 }

379 public bool Insert ( long index, Pair<T, TU> entry ) {

394 for ( long i = m_index ; i > index ; i-- ) {
395 m_elements[i] = m_elements[i - 1];
396 m_state[i] = m_state[i - 1];

400 m_state[index] = 1;

403 m_index = index + 1;

405 return true;

674
675 long src = (long)sourceOffset;
676 long dst = (long)destinationOffset;
677
678 if ( src > Length ) src = (long)Length;
679
680 long toCopy = System.Math.Min((long)count,
681 System.Math.Min(System.Math.Max(0, (long)Length - src),
682 System.Math.Max(0, destination.Length - dst)));

686 Buffer.LongCopy<Pair<T, TU>>(m_elements, src, destination.m_elements, dst, toCopy);

690 if ( end > destination.m_index )
691 destination.m_index = end;
692
693 return new Pair<bool, long>(true, toCopy);

701 public Pair<bool, long> CopyFrom ( Map<T, TU> source, ulong sourceOffset, ulong destinationOffset, ulong count ) {
702 long src = (long)sourceOffset;
703 long dst = (long)destinationOffset;
704
705 if ( dst > Length )
706 dst = Length;
707
708 long toCopy = System.Math.Min((long)count,
709 System.Math.Min(System.Math.Max(0, source.Length - src),
710 System.Math.Max(0, Length - dst)));
711
712 // Wenn nichts passt → prüfen ob wir wachsen müssen
713 if ( toCopy <= 0 ) {
714 if ( !AutoGrow )
715 return new Pair<bool, long>(false, 0);
716
717 long required = dst + (long)count;
718
719 long newSize = Length;
720 while ( required > newSize )
721 newSize += GrowSize;
722
723 if ( !Resize(newSize) )
724 return new Pair<bool, long>(false, 0);
725
726 // Nach Resize neu berechnen
727 toCopy = System.Math.Min((long)count,
728 System.Math.Min(System.Math.Max(0, source.Length - src),
729 System.Math.Max(0, (long)Length - dst)));
730
731 if ( toCopy <= 0 )
732 return new Pair<bool, long>(false, 0);
733 }
734
735 Buffer.LongCopy<Pair<T, TU>>(source.m_elements, src, m_elements, dst, toCopy);
736 Buffer.LongCopy<byte>(source.m_state, src, m_state, dst, toCopy);
737
738 // m_index anpassen, falls wir weiter hinten geschrieben haben
739 long end = dst + toCopy;
740 if ( end > m_index )
741 m_index = end;
742
743 return new Pair<bool, long>(true, toCopy);
744 }
