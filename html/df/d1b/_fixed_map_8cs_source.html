227 public bool InsertRange(int pos, IEnumerable<Pair<T, TU>> items) {
228 if ( IsFull ) return false;

270 public void Traverse(TraversMode mode, int startIndex, int endIndex, Action<Pair<T, TU>> func) {
271 int start = System.Math.Max(startIndex, 0);
272 int end = System.Math.Min(endIndex, m_count);
273
274 if ( mode == TraversMode.Forwards ) {
275 for ( int i = start; i < end; i++ )
276 func(m_elements[i]);
277 } else if ( mode == TraversMode.Backwards ) {
278 for ( int i = end; i >= start; i-- )
279 func(m_elements[i]);
280 }
281 }
