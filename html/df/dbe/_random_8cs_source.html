89 public static string RandPassword(int length, FixedArray<char> allowed, Endian endian) {
90
91 StringBuilder sb = new StringBuilder(length);
92
93 int i = 0, d = 0;
94 char c = '\0';
95 while(i < length ) {
96 c = RandChar((char)0, (char)short.MaxValue, endian);
97 if ( allowed.TryGet(c, out d) ) {
98 i++;
99 sb.Append(c);
100 }
101 }
102
103 return sb.ToString();
104 }

360 public static ulong RandULong(ulong min, ulong max, Endian endian) {
361 if ( min > max )
362 throw new ArgumentException("min must be <= max");
363
364 // 1. 8 zufällige Bytes erzeugen
365 byte[] _array_h = GetArray(4);
366 byte[] _array_l = GetArray(4);
367
368 // 3. Range-Mapping (raw % range)
369 ulong range = (ulong)(max - min + 1);
370 ulong value = ((_array_h.ToUInt(endian) << 32) | _array_l.ToUInt(endian)) % range;
371
372 // 4. Offset hinzufügen
373 return (ulong)(min + (ulong)value);
374 }
379 public static long RandLong(long min, long max, Endian endian) {
380 if ( min > max ) {
381 long tmp = min;
382 min = max;
383 max = tmp;
384 }
385
386 int _h = GetArray(4).ToInt(endian);
387 uint _l = GetArray(4).ToUInt(endian);
388
389 ulong raw = ((ulong)(uint)_h << 32) | _l;
390
391 ulong range = (ulong)(max - min) + 1;
392
393 long value = (long)(raw % range);
394
