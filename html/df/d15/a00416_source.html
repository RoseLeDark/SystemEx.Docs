179 public ColorHDR Lerp(ColorHDR value, float amount) {
180 // Quadratische Kurve
181 float q = amount * amount;
182
183 // Hue shortest-path interpolation
184 float dh = value.m_hue - m_hue;
185 dh = (dh + 540f) % 360f - 180f;
186
187 return new ColorHDR(
188 ClampHue(m_hue + dh * q),
189 m_saturation + (value.m_saturation - m_saturation) * q,
190 m_value + (value.m_value - m_value) * q
191 );
192 }
