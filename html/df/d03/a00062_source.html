190 private static void Shuffle ( ref uint a, ref uint b, ref uint c, ref uint d,
191 ref uint e, ref uint f, ref uint g, ref uint h ) {
192 a ^= b << 11; d += a; b += c;
193 b ^= c >> 2; e += b; c += d;
194 c ^= d << 8; f += c; d += e;
195 d ^= e >> 16; g += d; e += f;
196 e ^= f << 10; h += e; f += g;
197 f ^= g >> 4; a += f; g += h;
