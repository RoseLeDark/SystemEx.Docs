33 public struct LightLatch {
34 private int m_counter;
35 private LightConditionVariable m_cv;
36 private ILock m_lock;
37
38 public LightLatch ( int initialCount, ILock lockType, string name = "Latch" ) {
39 m_counter = initialCount;
40 m_cv = new LightConditionVariable(name + ".CV");
41 m_lock = lockType;
42 }
43
44 public void Arrive ( int maxWait = -1 ) {
45 m_lock.Lock(maxWait);
46
47 m_counter--;
48 if ( m_counter == 0 ) {
49 m_cv.Broadcast(); // alle wartenden Threads wecken
50 }
51
52 m_lock.Unlock();
53 }
54
55 public void Wait (ref LightThread thread, int maxWait = -1) {
56 m_lock.Lock(maxWait);
57
58 if ( m_counter == 0 ) {
59 m_lock.Unlock();
60 return;
61 }
62
63 m_cv.Add(thread);
64 m_lock.Unlock();
65
66 thread.Wait(ref m_cv, ref m_lock, maxWait);
67 }
68 }
