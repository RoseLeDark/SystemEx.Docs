193 public Barrier (uint min, uint max, Action<int, Barrier > onCompl ) {
194 if(min == max) {
195 m_min = min;
196 m_max = min + 1;
197 } else {
198 m_min = System.Math.Min(min, max);
199 m_max = System.Math.Max(min, max);
200 }
201 m_current = new SafeCounter(min);
202 m_phase = new SafeCounter(0);
203 m_drops = new SafeCounter(0);
204
205 OnComplition = onCompl;
206 m_hold = new AtomicLock();
207 m_hold.Lock(1000);
208
209 m_context = new AtomicLock();
210 }

212 public long WaitOpen( uint waitMS = 10 ) {
213 while(true) {
214 if ( (int)m_current.Value == m_max ) {
215 m_current.Assign(m_min);
216 break;
217 }
218
219 Thread.SpinWait((int)waitMS);
220 }
221
222 m_hold.Unlock();
223
224 if ( !m_hold.IsHeld ) {
225
226 m_context.Lock(-1);
227
228 OnComplition?.Invoke((int)m_phase.Value, this);
229
230
231 if ( m_hold.Lock(-1) ) {
232 if ( m_drops > 0 ) {
233 if ( m_max - m_drops >= m_min )
234 m_max -= (uint)m_drops.Value;
235 else
236 throw new BarrierNewPhaseLockException("To many drops");
237 }
238 m_current.Assign(m_min);
239 m_phase++;
240 currentPhase = m_phase.Value;
241 }
242
243 m_context.Unlock();
244 }
245
246
247 return currentPhase;
248 }
