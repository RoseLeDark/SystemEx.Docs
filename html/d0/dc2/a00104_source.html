66 public long Find ( ref TContainer container, ICompared<T> comp, T value ) {
67 long _arCount= container.Count;
68 long _ret = 0;
69
70 if ( _arCount> 0 ) {
71 // find the smallest Fibonacci number that equals or is greater than the array length
72 long _fbPrevB = 0;
73 long _fbNPrev= 1;
74 long _fbNum = _fbNPrev;
75
76 while ( _fbNum <= _arCount) {
77 _fbPrevB = _fbNPrev;
78 _fbNPrev= _fbNum;
79 _fbNum = _fbPrevB + _fbNPrev;
80 }
81
82 // offset to drop the left part of the array
83 long offset = -1;
84
85 while ( _fbNum > 1 ) {
86 var index = System.Math.Min(offset + _fbPrevB, _arCount- 1);
87
88 switch ( comp.Compare(container.ElementAt(index), value) ) {
89 case CompareResult.AIsLargerB:
90 _fbNum = _fbNPrev;
91 _fbNPrev= _fbPrevB;
92 _fbPrevB = _fbNum - _fbNPrev;
93 offset = index;
94 break;
95
96 // reject approximately 2/3 of the existing array behind
97 // by moving Fibonacci numbers
98 case CompareResult.AIsSmallerB:
99 _fbNum = _fbPrevB;
100 _fbNPrev= _fbNPrev- _fbPrevB;
101 _fbPrevB = _fbNum - _fbNPrev;
102 break;
103 default:
104 _ret = index; break;
105 }
106 }
107
108
109 // check the last element
110 Optional<T> _it = container.ElementAt(_arCount - 1);
111
112 if ( _it.IsSome ) {
113 if ( _fbNPrev == 1 && comp.Compare(value, _it) == CompareResult.Equal ) {
114 _ret++;
115 }
116 }
117 }
118 return _ret;
119 }
