188 public ulong Seek(SeekOrigin org, int pos) {
189 if ( m_rawBuffer.Length == 0 ) return 0;
190
191 switch ( org ) {
192
193 case SeekOrigin.Begin:
194 m_position = (ulong)System.Math.Max(pos, 0);
195 break;
196 case SeekOrigin.Current:
197 if ( pos >= 0 )
198 m_position += (ulong)pos;
199 else {
200 ulong neg = (ulong)(-pos);
201 m_position = (m_position > neg) ? (m_position - neg) : 0;
202 }
203 break;
204 case SeekOrigin.End:
205 if ( pos >= 0 ) {
206 ulong p = (ulong)pos;
207 m_position = (p > Length) ? 0 : (Length - p);
208 } else {
209 // End + negative pos = End - (-pos) = End + pos
210 ulong add = (ulong)(-pos);
211 m_position = Length + add; // wird unten gekappt
212 }
213 break;
214 }
215
216 if ( m_position > Length )
217 m_position = Length;
218
219 return m_position;
220 }

236 public virtual ulong WriteRange(ulong start, ulong iend, byte[] data) {
237 if ( m_isLocked ) throw new InvalidOperationException("is Locked");
238
239 // Start ungültig?
240 if ( start < 0 || start >= (ulong)m_rawBuffer.Length )
241 return 0;
242
243 // End über Größe → kappen
244 if ( iend > (ulong)m_rawBuffer.Length )
245 iend = (ulong)m_rawBuffer.Length;
246
247 // Bereich ungültig?
248 if ( iend <= start )
249 return 0;
250
251 ulong rangeLen = iend - start;
252 ulong writable = System.Math.Min((uint)rangeLen, (uint)data.Length);
253
254 for ( ulong i = 0; i < (ulong)writable; i++ )
255 m_rawBuffer[(int)(start + i)] = data[i];
256
257 if ( (start + writable) > m_maxUsedAdress ) m_maxUsedAdress = start + writable;
258
259 return (ulong)writable;
260 }

265 public int Read ( byte[] buffer, int offset, int count ) {
266 if ( m_isLocked ) throw new InvalidOperationException("is Locked");
267 if ( offset < 0 || count < 0 ) throw new ArgumentOutOfRangeException("offset,count");
268 ArgumentNullException.ThrowIfNull(buffer);
269 ArgumentOutOfRangeException.ThrowIfGreaterThan(offset, buffer.Length);
270
271 if ( offset + count > buffer.Length ) {
272 // Wenn der Caller mehr verlangt als im Zielbuffer Platz ist,
273 // lesen wir nur so viel, wie in den Zielbuffer passt.
274 count = buffer.Length - offset;
275 }
276
277 // Wenn die Leseposition bereits am oder jenseits des Endes ist: EOF
278 if ( (ulong)m_rawBuffer.Length <= m_position ) return 0;
279
280 // Verfügbare Bytes im Rohpuffer ab aktueller Position
281 ulong available = (ulong)m_rawBuffer.Length - m_position;
282 int toRead = (int)System.Math.Min((ulong)count, available);
283
284 if ( toRead <= 0 ) return 0;
285
286 // ReadRange erwartet position als ulong und count als uint
287 byte[]? chunk = ReadRange(m_position, (uint)toRead);
288 if ( chunk == null || chunk.Length == 0 ) return 0;
289
290 Array.Copy(chunk, 0, buffer, offset, chunk.Length);
291
292 // interne Position vorziehen
293 m_position += (ulong)chunk.Length;
294
295 return chunk.Length;
296 }

322 public int Write ( byte[] buffer, int offset, int count ) {
323 if ( m_isLocked ) throw new InvalidOperationException("is Locked");
324 if ( offset < 0 || count < 0 ) throw new ArgumentOutOfRangeException("offset,count");
325
326 ArgumentNullException.ThrowIfNull(buffer);
327 ArgumentOutOfRangeException.ThrowIfGreaterThan(offset, buffer.Length);
328
329
330
331 // Adjust count if caller asked for more than fits in the target buffer
332 if ( offset + count > buffer.Length ) {
333 count = buffer.Length - offset;
334 }
335
336 // If write position is already at or beyond end, nothing to write
337 if ( (ulong)m_rawBuffer.Length <= m_position ) return 0;
338
339 // How many bytes are available in the raw buffer from current position
340 ulong available = (ulong)m_rawBuffer.Length - m_position;
341 int toWrite = (int)System.Math.Min((ulong)count, available);
342
343 if ( toWrite <= 0 ) return 0;
344
345 for ( int i = 0 ; i < toWrite ; i++ ) {
346 m_rawBuffer[(int)(m_position + (ulong)i)] = buffer[offset + i];
347 }
348
349 // advance internal position
350 m_position += (ulong)toWrite;
351
352 if ( m_position > m_maxUsedAdress ) m_maxUsedAdress = m_position;
353
354 return toWrite;
355 }
