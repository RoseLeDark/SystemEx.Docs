102 public static string GetLibaryPath ( string somodule ) {
103 string _ret = NO_PATH;
104
105 // Check /usr/lib
106 _ret = GetFromPath("/usr/lib/", somodule);
107
108 // Check LD_LIBRARY_PATH
109 if ( _ret == NO_PATH ) {
110 string? env = Environment.GetEnvironmentVariable("LD_LIBRARY_PATH");
111 if ( env != null ) {
112 var paths = env.Split(':');
113 foreach ( string path in paths ) {
114 string candidate = GetFromPath(path, somodule);
115 if ( candidate != NO_PATH ) {
116 _ret = candidate;
117 break;
118 }
119 }
120 }
121 }
122
123 return _ret;
124 }
