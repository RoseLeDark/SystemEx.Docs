550 private static void DownHeap<T> ( T[] items, int k, int n, CompFunc<T> cmp ) {
551 T temp = items[k - 1];
552
553 while ( k <= n / 2 ) {
554 int child = 2 * k;
555
556 if ( child < n && cmp(items[child - 1], items[child]) == CompareResult.AIsSmallerB )
557 child++;
558
559 if ( cmp(temp, items[child - 1]) == CompareResult.AIsSmallerB ) {
560 items[k - 1] = items[child - 1];
561 k = child;
562 } else break;
563 }
564
565 items[k - 1] = temp;
566 }
