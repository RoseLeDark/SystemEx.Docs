119 public static void ShiftInput ( byte[] input, int len, ref int remain ) {
120 remain = len - 64;
121
122 if ( remain > 0 ) {
123 for ( int i = 0 ; i < remain ; i++ )
124 input[i] = input[i + 64];
125
126 for ( int i = remain ; i < len ; i++ )
127 input[i] = 0;
128 } else {
129 for ( int i = 0 ; i < len ; i++ )
130 input[i] = 0;
131 }

229 public static void compress_pre ( uint[] state, uint[] cv, byte[] block, byte block_len, UInt64 counter, Blake3Flags flags ) {

235 block_words[3] = block.ToUInt(12, Endian.LittleEndian);
236 block_words[4] = block.ToUInt(16, Endian.LittleEndian);

241 block_words[9] = block.ToUInt(36, Endian.LittleEndian);
242 block_words[10] = block.ToUInt(40, Endian.LittleEndian);
243 block_words[11] = block.ToUInt(44, Endian.LittleEndian);
244 block_words[12] = block.ToUInt(48, Endian.LittleEndian);
245 block_words[13] = block.ToUInt(52, Endian.LittleEndian);
246 block_words[14] = block.ToUInt(56, Endian.LittleEndian);
247 block_words[15] = block.ToUInt(60, Endian.LittleEndian);
248
249 state[0] = cv[0];
250 state[1] = cv[1];
251 state[2] = cv[2];
252 state[3] = cv[3];

254 state[5] = cv[5];

261 state[12] = Black3Utils.CounterLow(counter);

263 state[14] = (uint)block_len;
264 state[15] = (uint)flags;
