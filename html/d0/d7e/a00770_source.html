156 public static T ToBoundary<T>(this T number, T boundary) where T : INumber<T> {
157 if ( boundary == T.Zero ) return (T)number;
158 T div = number / boundary;
159 T mod = number % boundary;
160 return (T)(mod == T.Zero ? div * boundary : (div + T.One) * boundary);
161 }
