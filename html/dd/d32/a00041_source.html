180 public bool HaveCap ( BackendCapabilities pCapabilities ) {
181 object? _value;
182 bool _ret = false;
183
184 // Wenn der Key nicht existiert → Capability nicht vorhanden
185 if ( !Backend.Capabilities.TryGetValue(pCapabilities, out _value) )
186 return false;
187
188 if(_value != null ) {
189 if ( _value is int i ) _ret = i >= 1;
190 if ( _value is bool b ) _ret = b;
191 if ( _value is string stt ) _ret = stt.Contains("true", StringComparison.CurrentCultureIgnoreCase);
192 }
193
194 return _ret;
195 }

257 public bool Initialization ( Map<string, object> configuration ) {
258 // Set env
259 Map<AIEnv, object> environment = new Map<AIEnv, object>();
260
261 // OS
262 if ( OperatingSystem.IsWindows() )
263 environment.PushBack(AIEnv.AI_ENV_OS_WINDOWS, true);
264 else if ( OperatingSystem.IsLinux() )
265 environment.PushBack(AIEnv.AI_ENV_OS_LINUX, true);
266 else if ( OperatingSystem.IsMacOS() )
267 environment.PushBack(AIEnv.AI_ENV_OS_MACOS, true);
268
269 // Architecture
270 if ( RuntimeInformation.OSArchitecture == Architecture.X64 )
271 environment.PushBack(AIEnv.AI_ENV_ARCH_X64, true);
272 else if ( RuntimeInformation.OSArchitecture == Architecture.Arm64 )
273 environment.PushBack(AIEnv.AI_ENV_ARCH_ARM64, true);
274
275 // Runtime
276 environment.PushBack(AIEnv.AI_ENV_RT_DOTNET, System.Environment.Version.ToString());
277
278 // Paths
279 environment.PushBack(AIEnv.AI_ENV_FS_TEMP_DIR, Path.GetTempPath());
280 environment.PushBack(AIEnv.AI_ENV_FS_WORK_DIR, System.Environment.CurrentDirectory);
281 environment.PushBack(AIEnv.AI_ENV_FS_HOME_DIR, System.Environment.GetFolderPath(System.Environment.SpecialFolder.UserProfile));
282
283 // Locale / Culture
284 environment.PushBack(AIEnv.AI_ENV_LOCALE, CultureInfo.CurrentCulture.Name);
285 environment.PushBack(AIEnv.AI_ENV_CULTURE, CultureInfo.CurrentCulture.DisplayName);
286
287 // Network (simple check)
288 bool online = System.Net.NetworkInformation.NetworkInterface.GetIsNetworkAvailable();
289 environment.PushBack(online ? AIEnv.AI_ENV_NET_ONLINE : AIEnv.AI_ENV_NET_OFFLINE, true);
290
291 // Hardware (basic detection)
292 environment.PushBack(AIEnv.AI_ENV_HW_CPU, true);
293
294 return Initialization(configuration, environment);
295 }
