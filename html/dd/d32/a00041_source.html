193 public SeedMixed Mix<T> ( T other, SeedMix type, uint value = 2 ) where T : ISeed {
194
195 int _l = System.Math.Min(other.Length, m_usable.Length);
196
197 m_lLock.Lock(TimeOut.Infinite);
198
199 switch ( type ) {
200 case SeedMix.Addition: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] += other[i]; break;
201 case SeedMix.Subtraction: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] -= other[i]; break;
202 case SeedMix.Multiplication: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] *= other[i]; break;
203 case SeedMix.XOR: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] ^= other[i]; break;
204 case SeedMix.Or: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] |= other[i]; break;
205 case SeedMix.And: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] &= other[i]; break;
206 case SeedMix.Minimal: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] = System.Math.Min(m_usable[i], other[i]); break;
207 case SeedMix.Maximal: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] = System.Math.Max(m_usable[i], other[i]); break;
208 case SeedMix.Average: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] = (m_usable[i] + other[i]) >> 1; break;
209 case SeedMix.New: for ( int i = 0 ; i < _l ; i++ ) m_usable[i] = other[i]; break;
210 case SeedMix.ShiftRight: for ( int i = 0 ; i < _l ; i++ ) { var _x = (m_usable[i] >> (int)(other[i] & value)); m_usable[i] = _x; } break;
211 case SeedMix.ShiftLeft: for ( int i = 0 ; i < _l ; i++ ) { var _x = (m_usable[i] << (int)(other[i] & value)); m_usable[i] = _x; } break;
212 case SeedMix.RotateLeft: for ( int i = 0 ; i < _l ; i++ ) { int _r = (int)(other[i] + value) & 31; m_usable[i] = (m_usable[i] << _r) | (m_usable[i] >> (32 - _r)); } break;
213 case SeedMix.RotateRight: for ( int i = 0 ; i < _l ; i++ ) { int _r = (int)(other[i] + value) & 31; m_usable[i] = (m_usable[i] >> _r) | (m_usable[i] << (32 - _r)); } break;
214 default: if ( OnUserMix != null ) m_usable = OnUserMix.Invoke(m_usable, other); break;
215 }
216
217 m_lLock.Unlock();
218
219 return this;
220 }
221 }
