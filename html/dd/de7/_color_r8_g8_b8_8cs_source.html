76 public ColorR8G8B8(int rgb) {

78 byte g = (byte)((rgb >> 8) & 0xFF);
79 byte b = (byte)(rgb & 0xFF);

82 m_green = g * 0.00392156862745098f;
83 m_blue = b * 0.00392156862745098f;

277 public ContrastLevel GetKontrast(ColorR8G8B8 other) {
278 float c = Contrast(other);
279
280 if ( c < 0.20f )
281 return ContrastLevel.VeryLow;
282
283 if ( c < 0.40f )
284 return ContrastLevel.Low;
285
286 if ( c < 0.70f )
287 return ContrastLevel.High;
288
289 return ContrastLevel.Perfect;

294 public ColorR8G8B8 GetNextContrastColor() {
