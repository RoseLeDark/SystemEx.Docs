196 private bool Set(T key, TU value) {
197 for ( int i = 0; i < m_elements.Count; i++ ) {
198 if ( m_elements[i].EqualFirst(key) ) {
199
200 if ( m_elements[i].EqualSecond(value) )
201 return true;
202
203 int last = m_elements.Count - 1;
204 m_elements[i] = m_elements[last];
205 m_elements.RemoveAt(last);
206
207 m_elements.Add(new Pair<T, TU>(key, value));
208 return true;
209 }
210 }
211
212 return TryAdd(key, value);
213 }

262 public void Traverse(TraversMode mode, int startIndex, int endIndex, Action<Pair<T, TU>> func) {
263 int start = System.Math.Max(startIndex, 0);
264 int end = System.Math.Min(endIndex, m_elements.Count);
265
266 if ( mode == TraversMode.Forwards ) {
267 for ( int i = start; i < end; i++ )
268 func(m_elements[i]);
269 } else if ( mode == TraversMode.Backwards ) {
270 for ( int i = end; i >= start; i-- )
271 func(m_elements[i]);
272 }
273 }
