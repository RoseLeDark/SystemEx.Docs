711 long dst = (long)destinationOffset;
712

749 public Pair<bool, long> CopyFrom ( FixedVector<T> source, ulong sourceOffset, ulong destinationOffset, ulong count ) {
750 long src = (long)sourceOffset;
751 long dst = (long)destinationOffset;
752
753 // Clamp source offset
754 if ( src > source.Length )
755 src = source.Length;
756
757 // Clamp destination offset
758 if ( dst > Length )
759 dst = Length;
760
761 // Compute how much can actually be copied
762 long toCopy = System.Math.Min((long)count,
763 System.Math.Min(
764 System.Math.Max(0, source.Length - src),
765 System.Math.Max(0, Length - dst)
766 ));
767
768 // Nothing fits → return (false, 0)
769 if ( toCopy <= 0 )
770 return new Pair<bool, long>(false, 0);
771
772 // Perform the copy
773 Buffer.LongCopy<T>(source.m_elements, src, m_elements, dst, toCopy);
774
775 // Update logical length only if we wrote beyond current end
776 long end = dst + toCopy;
777 if ( end > m_index )
778 m_index = end;
779
780 return new Pair<bool, long>(true, toCopy);
781 }
