137 private unsafe uint privateCompute ( uint initial, FixedVector<byte> buffer ) {
138
139
140 uint c = initial ^ 0xFFFFFFFF;
141
142 for (long i = 0; i < buffer.Length; ++i) {
143 c = m_table[(c ^ buffer[i]) & 0xFF] ^ (c >> 8);
144 }
145 return c ^ 0xFFFFFFFF;
146 }

153 private void generateTable ( uint polynomial ) {
154
155 for ( uint i = 0 ; i < 256 ; i++ ) {
156 uint c = i;
157 for ( ulong j = 0 ; j < 8 ; j++ ) {
158 if ( (c & 1) == 1 )
159 c = polynomial ^ (c >> 1);
160 else
161 c >>= 1;
162 }
163 m_table[i] = c;
164 }
