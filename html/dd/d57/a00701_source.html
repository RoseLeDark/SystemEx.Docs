492 public int Abort (bool hardt) {
493 m_continuemutex.Lock (-1) ;
494 m_runningMutex.Lock(-1) ;
495
496 if ( !m_bRunning ) {
497 m_runningMutex.Unlock();
498 m_continuemutex.Unlock();
499
500 return 1; // ERR_TASK_NOTRUNNING;
501 }
502
503 // nicht aus sich selbst killen
504 if ( Thread.CurrentThread.ManagedThreadId == m_base.ManagedThreadId ) {
505
506 m_runningMutex.Unlock();
507 m_continuemutex.Unlock();
508
509 return 42; // ERR_TASK_CALLFROMSELFTASK
510 }
511
512
513
514
515 m_runn = false;
516 m_eState = (hardt)? ThreadExState.RequestKill : ThreadExState.RequestAbort;
517 if ( hardt ) m_base.Interrupt();
518
519 if ( IsSuspend() )
520 m_eventGroup.Set(EVENTGROUP_BIT_CONTINUE);
521
522 m_contextMutext.Lock(-1);
523 OnStop(this, hardt);
524 m_contextMutext.Unlock();
525
526 m_runningMutex.Unlock();
527 m_continuemutex.Unlock();
528
529
530
531 return 0; // ERR_TASK_OK;
532 }

735 protected virtual int OnRunning ( ThreadEx e, Object? userdata ) {
736
737
738
739 int _ret = 0;
740
741 while ( m_runn && _ret == 0) {
742
743 if ( m_eventGroup.IsSet(EVENTGROUP_BIT_INVOKESTATE) ) {
744
745 cheackEndContext(false);
746 }
747
748 if( m_eventGroup.IsSet(EVENTGROUP_BIT_INVOKEMESSAGE) ) {
749 m_eventGroup.Clear(EVENTGROUP_BIT_INVOKEMESSAGE);
750
751 while (!m_contextQueue.IsEmpty) {
752 Optional<Triple<byte, ThreadInvokeMessage, object>> _ref = new Triple<byte, ThreadInvokeMessage, object>();
753
754 if ( !m_contextQueue.PopFront(ref _ref) )
755 continue;
756
757
758 var msgType = _ref.Value!.Second;
759 var payload = _ref.Value!.Third;
760
761 switch ( msgType ) {
762
763 case ThreadInvokeMessage.Message:
764 m_messageQueue.PushBack(payload);
765 break;
766
767 case ThreadInvokeMessage.Stack:
768 if ( payload is Pair<byte, object> g )
769 m_coRoutineStack.Push(g.Second, g.First);
770 break;
771
772 case ThreadInvokeMessage.Propertie:
773 break;
774 case ThreadInvokeMessage.CoRoutine:
775 break;
776 }
777 }
778
779 }
780
781 if ( IsSuspend() ) {
782 m_eState = ThreadExState.Pause;
783 WaitOfEvent(EVENTGROUP_BIT_CONTINUE, -1);
784 if ( OnResume != null ) OnResume.Invoke(e, userdata);
785
786
787 m_eventGroup.Clear(EVENTGROUP_BIT_CONTINUE);
788 m_eventGroup.Clear(EVENTGROUP_BIT_PAUSE);
789
790 } else {
791 m_eState = ThreadExState.Running;
792 if ( OnTask != null ) {
793 _ret = OnTask.Invoke(e, userdata);
794 }
795 }
796 }
797
798 return _ret;
799 }

824 private bool cheackEndContext(bool abort) {
825
826 if ( m_CommitWindow.ElapsedMilliseconds <= m_CommitWinOpen )
827 return false;
828
829 if ( abort ) {
830 m_contextQueue.Clear();
831 } else if ( !m_contextQueue.IsEmpty ) {
832 m_eventGroup.Set((byte)(EVENTGROUP_BIT_INVOKESTATE + 1));
833 }
834
835 m_eventGroup.Clear(EVENTGROUP_BIT_INVOKESTATE);
836 m_CommitWindow.Stop();
837 m_CommitWindow.Reset();
838 m_contextMutext.Unlock();
839
840 return true;
841 }

864 private static void ThreadStart ( object? obj ) {
865 int ret = 0;
866
867 if( obj is ThreadEx thread) {
868
869 thread.m_eventGroup.Set(EVENTGROUP_BIT_STARTED);
870 thread.m_eState = ThreadExState.Started;
871
872 // set running
873 thread.m_runningMutex.Lock (-1) ;
874 thread.m_continuemutex.Lock(-1) ;
875 thread.m_bRunning = true;
876 //thread.m_runningMutex.Unlock();
877 thread.ID = Thread.CurrentThread.ManagedThreadId;
878
879
880 if ( thread.OnSetup(thread, thread.m_userData) ) {
881
882 thread.m_runningMutex.Unlock();
883 ret = thread.OnRunning(thread, thread.m_userData);
884
885
886 thread.OnExit(thread, thread.m_userData);
887 }
888
889
890 thread.m_runningMutex.Lock (-1) ;
891 thread.m_bRunning = false;
892 thread.m_eState = ThreadExState.Stopped;
893 thread.m_retval = ret;
894 thread.m_runningMutex.Unlock();
895 thread.m_continuemutex.Unlock();
896
897 // set the join bit
898 thread.m_eventGroup.Set(EVENTGROUP_BIT_JOINABLE);
899 }
900 }
