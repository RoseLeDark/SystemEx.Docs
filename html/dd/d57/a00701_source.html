309 public static FloatE4M3 Add ( FloatE4M3 a, FloatE4M3 b ) {
310 if ( IsNaN(a) || IsNaN(b) ) return FloatE4M3.NaN;
311 if ( IsZero(a) ) return b;
312 if ( IsZero(b) ) return a;
313
314 Fast_Byte expA = a.Exponent;
315 Fast_Byte expB = b.Exponent;
316
317 Fast_Int mantA = (expA == 0 ? a.Mantissa : (a.Mantissa | 0x08));
318 Fast_Int mantB = (expB == 0 ? b.Mantissa : (b.Mantissa | 0x08));
319
320 Fast_Byte exp = expA;
321 Fast_Int diff = expA - expB;
322
323 if ( diff > 0 )
324 mantB >>= diff;
325 else if ( diff < 0 ) {
326 mantA >>= -diff;
327 exp = expB;
328 }
329
330 Fast_Int mant;
331 Fast_Byte sign;
332
333 if ( a.Sign == b.Sign ) {
334 mant = mantA + mantB;
335 sign = (byte)(a.Sign ? 1 : 0);
336 } else {
337 if ( mantA >= mantB ) {
338 mant = mantA - mantB;
339 sign = (byte)(a.Sign ? 1 : 0);
340 } else {
341 mant = mantB - mantA;
342 sign = (byte)(b.Sign ? 1 : 0);
343 }
344 }
345
346 if ( mant == 0 )
347 return FloatE4M3.Zero;
348
349 while ( (mant & 0x08) == 0 ) {
350 mant <<= 1;
351 exp--;
352 }
353
354 if ( exp <= 0 )
355 return FloatE4M3.Zero;
356
357 if ( exp >= 0x0F )
358 return FloatE4M3.NaN;
359
360 Fast_Byte finalMant = (byte)(mant & 0x07);
361
362 return new FloatE4M3(sign.Value, (byte)exp.Value, finalMant.Value);
363 }

367 public static FloatE4M3 Mul ( FloatE4M3 a, FloatE4M3 b ) {
368 if ( IsNaN(a) || IsNaN(b) ) return FloatE4M3.NaN;
369 if ( IsZero(a) || IsZero(b) ) return FloatE4M3.Zero;
370
371 Fast_Byte sign = (byte)((a.Sign ? 1 : 0) ^ (b.Sign ? 1 : 0));
372
373 Fast_Byte exp = a.Exponent + b.Exponent - a.ExponentBias;
374
375 Fast_Int mantA = (a.Exponent == 0 ? a.Mantissa : (a.Mantissa | 0x08));
376 Fast_Int mantB = (b.Exponent == 0 ? b.Mantissa : (b.Mantissa | 0x08));
377
378 Fast_Int mant = mantA * mantB;
379
380 while ( (mant & 0x40) != 0 ) {
381 mant >>= 1;
382 exp++;
383 }
384
385 mant >>= 3;
386
387 if ( exp <= 0 )
388 return FloatE4M3.Zero;
389
390 if ( exp >= 0x0F )
391 return FloatE4M3.NaN;
392
393 byte finalMant = (byte)(mant & 0x07);
394
395 return new FloatE4M3(sign.Value, exp.Value, finalMant);
396 }

400 public static FloatE4M3 Div ( FloatE4M3 a, FloatE4M3 b ) {
401 if ( IsNaN(a) || IsNaN(b) ) return FloatE4M3.NaN;
402 if ( IsZero(b) ) return FloatE4M3.NaN;
403 if ( IsZero(a) ) return FloatE4M3.Zero;
404
405 Fast_Byte sign = (byte)((a.Sign ? 1 : 0) ^ (b.Sign ? 1 : 0));
406
407 Fast_Byte exp = a.Exponent - b.Exponent + a.ExponentBias;
408
409 Fast_Int mantA = (a.Exponent == 0 ? a.Mantissa : (a.Mantissa | 0x08));
410 Fast_Int mantB = (b.Exponent == 0 ? b.Mantissa : (b.Mantissa | 0x08));
411
412 Fast_Int mant = (mantA << 6) / mantB;
413
414 while ( (mant & 0x08) == 0 ) {
415 mant <<= 1;
416 exp--;
417 }
418
419 if ( exp <= 0 )
420 return FloatE4M3.Zero;
421
422 if ( exp >= 0x0F )
423 return FloatE4M3.NaN;
424
425 byte finalMant = (byte)(mant & 0x07);
426
427 return new FloatE4M3(sign.Value, exp.Value, finalMant);
428 }

625 internal static byte Encode ( byte sign, byte exponent, byte mantissa ) {
626 Fast_Byte b = 0;
627
628 b.At(7, sign); // Signbit
629 b.At(3, (byte)((exponent >> 0) & 1));
630 b.At(4, (byte)((exponent >> 1) & 1));
631 b.At(5, (byte)((exponent >> 2) & 1));
632 b.At(6, (byte)((exponent >> 3) & 1));
633
634 b.At(0, (byte)((mantissa >> 0) & 1));
635 b.At(1, (byte)((mantissa >> 1) & 1));
636 b.At(2, (byte)((mantissa >> 2) & 1));
637
638 return b.Value;
639 }
