18namespace SystemEx.Drawing {
28 public class ColorHSV : IColor<ColorHSV>, IEquatable<ColorHSV>, IComparable<ColorHSV> {
29 internal float m_hue; // 0–360°
30 internal float m_saturation; // 0–1
31 internal float m_value; // 0–1
32
37 public static ColorHSV Zero => new ColorHSV(0f, 0f, 0f);
38
43 public static ColorHSV One => new ColorHSV(0f, 0f, 1f);

49 public virtual float H { get => m_hue; set => m_hue = ClampHue(value); }
50
54 public virtual float V { get => m_value; set => m_value = System.Math.Clamp(value, 0f, 1f); }
55
59 public virtual float S { get => m_saturation; set => m_saturation = System.Math.Clamp(value, 0f, 1f); }

64 public ColorHSV(float h, float s, float v) {
65 m_hue = ClampHue(h);
66 m_saturation = System.Math.Clamp(s, 0f, 1f);
67 m_value = System.Math.Clamp(v, 0f, 1f);
68 }

73 public virtual ColorHSV Saturation(float delta) {
74 m_saturation = System.Math.Clamp(m_saturation + delta, 0f, 1f);
75 return this;

77
81 public virtual ColorHSV Brightness(float delta) {
82 m_value = System.Math.Clamp(m_value + delta, 0f, 1f);
83 return this;

90 public virtual ColorHSV Addition(ColorHSV a) {
91 m_saturation = System.Math.Clamp(m_saturation + a.m_saturation, 0f, 1f);
92 m_value = System.Math.Clamp(m_value + a.m_value, 0f, 1f);
93 m_hue = ClampHue(m_hue + a.m_hue);
94 return this;
95 }
96
101 public virtual ColorHSV Subtraction(ColorHSV a) {
102 m_saturation = System.Math.Clamp(m_saturation - a.m_saturation, 0f, 1f);
103 m_value = System.Math.Clamp(m_value - a.m_value, 0f, 1f);
104 m_hue = ClampHue(m_hue - a.m_hue);
105 return this;
106 }
107
112 public virtual ColorHSV Multiplication(ColorHSV a) {
113 m_saturation = System.Math.Clamp(m_saturation * a.m_saturation, 0f, 1f);
114 m_value = System.Math.Clamp(m_value * a.m_value, 0f, 1f);
115 return this;
116 }
117
122 public virtual ColorHSV Division(ColorHSV a) {
123 if ( a.m_saturation != 0f )
124 m_saturation = System.Math.Clamp(m_saturation / a.m_saturation, 0f, 1f);
125
126 if ( a.m_value != 0f )
127 m_value = System.Math.Clamp(m_value / a.m_value, 0f, 1f);
128
129 return this;
130 }
131
138 public ColorHSV Lerp(ColorHSV value, float amount) {
139 float q = amount * amount;
140
141 // Shortest-path hue interpolation
142 float dh = value.m_hue - m_hue;
143 dh = (dh + 540f) % 360f - 180f;
144
145 return new ColorHSV(
146 ClampHue(m_hue + dh * q),
147 m_saturation + (value.m_saturation - m_saturation) * q,
148 m_value + (value.m_value - m_value) * q
149 );
150 }
