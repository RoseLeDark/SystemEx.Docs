177 public ColorHDR Lerp(ColorHDR value, float amount) {
178 // Quadratische Kurve
179 float q = amount * amount;
180
181 // Hue shortest-path interpolation
182 float dh = value.m_hue - m_hue;
183 dh = (dh + 540f) % 360f - 180f;
184
185 return new ColorHDR(
186 ClampHue(m_hue + dh * q),
187 m_saturation + (value.m_saturation - m_saturation) * q,
188 m_value + (value.m_value - m_value) * q
189 );
190 }
