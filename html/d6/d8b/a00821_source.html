154 public static T ToBoundary<T>(this T number, T boundary) where T : INumber<T> {
155 if ( boundary == T.Zero ) return (T)number;
156 T div = number / boundary;
157 T mod = number % boundary;
158 return (T)(mod == T.Zero ? div * boundary : (div + T.One) * boundary);
159 }
