477 private static void DownHeap<T> ( T[] items, int k, int n, CompFunc<T> cmp ) {
478 T temp = items[k - 1];
479
480 while ( k <= n / 2 ) {
481 int child = 2 * k;
482
483 if ( child < n && cmp(items[child - 1], items[child]) == CompareResult.AIsSmallerB )
484 child++;
485
486 if ( cmp(temp, items[child - 1]) == CompareResult.AIsSmallerB ) {
487 items[k - 1] = items[child - 1];
488 k = child;
489 } else break;
490 }
491
492 items[k - 1] = temp;
493 }
