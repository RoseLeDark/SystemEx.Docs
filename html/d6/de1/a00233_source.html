183 public byte IsIt () {
184 uint v = m_value;
185
186 v = v - ((v >> 1) & 0x55555555);
187
188 v = (v & 0x33333333) + ((v >> 2) & 0x33333333);
189
190 v = (v + (v >> 4)) & 0x0F0F0F0F;

192 v = v + (v >> 8);
193
194 v = v + (v >> 16);
