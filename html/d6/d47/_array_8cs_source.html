512 public virtual int InsertRange(int pos, IEnumerable<T> items) {
513 if ( pos < 0 || pos > Size ) return 0;
514
515 // Materialisieren, damit wir Count kennen
516 var list = items as ICollection<T> ?? new List<T>(items);
517 int count = list.Count;
518
519 if ( count == 0 ) return 0;
520
521 // Prüfen ob genug Platz ist
522 int required = m_index + count;
523 if ( required > Size ) {
524 if ( !AutoGrow )
525 return 0;
526
527 // so lange wachsen, bis es passt
528 int newSize = Size;
529 while ( required > newSize )
530 newSize += GrowSize;
531
532 if ( !Resize(newSize) ) return -1;
533 }
534
535
536 // Elemente einfügen
537 int idx = pos;
538 int written = 0;
539 foreach ( var item in list ) {
540 m_elements[idx++] = item;
541 written++;
542 }
543 return written;
544 }

623 public int CopyTo(uint sourceOffset, T[] destination, uint destinationOffset, uint count) {
624 if ( destination == null ) return 0;
625
626 int src = (int)sourceOffset;
627 int dst = (int)destinationOffset;
628
629 if ( src > Size ) src = Size;
630
631 int toCopy = System.Math.Min((int)count,
632 System.Math.Min(System.Math.Max(0, Size - src),
633 System.Math.Max(0, destination.Length - dst)));

635 if ( toCopy <= 0 ) return 0;

637 Buffer.BlockCopy(m_elements, src, destination, dst, toCopy);
638 return toCopy;

645 public int CopyFrom(T[] source, uint sourceOffset, uint destinationOffset, uint count) {
646 if ( source == null ) return 0;
647
648 int src = (int)sourceOffset;
649 int dst = (int)destinationOffset;
650
651 if ( dst > Size ) dst = Size;
652
653 int toCopy = System.Math.Min((int)count,
654 System.Math.Min(System.Math.Max(0, source.Length - src),
655 System.Math.Max(0, Size - dst)));
656
657 // Wenn nichts passt → prüfen ob wir wachsen müssen
658 if ( toCopy <= 0 ) {
659 // Prüfen ob AutoGrow aktiv ist
660 if ( !AutoGrow ) return 0;
661
662 // Wir brauchen mindestens count Bytes Platz ab dst
663 int required = dst + (int)count;
664
665 int newSize = Size;
666 while ( required > newSize )
667 newSize += GrowSize;
668
669 if ( !Resize(newSize) ) return 0;
670
671 // Nach Resize neu berechnen
672 toCopy = System.Math.Min((int)count, System.Math.Min(System.Math.Max(0, source.Length - src), System.Math.Max(0, Size - dst)));
673
674 if ( toCopy <= 0 )
675 return 0;
676 }
677
678 Buffer.BlockCopy(source, src, m_elements, dst, toCopy);
679 return toCopy;
680 }
