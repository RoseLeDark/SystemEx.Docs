49 public void Seed ( uint a, uint b, uint c, uint[]? s = null ) {
50 uint aa, bb, cc, dd, ee, ff, gg, hh;
51 aa = bb = cc = dd = ee = ff = gg = hh = GoldenRatio;
52
53 // Seed‑Array oder Null → wie bei dir m_rsl
54 for ( int i = 0 ; i < Size ; i++ )
55 _rsl[i] = s != null ? s[i] : 0;
56
57 _a = a;
58 _b = b;
59 _c = c;
60
61 Shuffle(ref aa, ref bb, ref cc, ref dd, ref ee, ref ff, ref gg, ref hh);
62 Shuffle(ref aa, ref bb, ref cc, ref dd, ref ee, ref ff, ref gg, ref hh);
63 Shuffle(ref aa, ref bb, ref cc, ref dd, ref ee, ref ff, ref gg, ref hh);
64 Shuffle(ref aa, ref bb, ref cc, ref dd, ref ee, ref ff, ref gg, ref hh);
65
66 for ( int i = 0 ; i < Size ; i += 8 ) {
67 aa += _rsl[i + 0]; bb += _rsl[i + 1]; cc += _rsl[i + 2]; dd += _rsl[i + 3];
68 ee += _rsl[i + 4]; ff += _rsl[i + 5]; gg += _rsl[i + 6]; hh += _rsl[i + 7];
69
70 Shuffle(ref aa, ref bb, ref cc, ref dd, ref ee, ref ff, ref gg, ref hh);
71
72 _mem[i + 0] = aa; _mem[i + 1] = bb; _mem[i + 2] = cc; _mem[i + 3] = dd;
73 _mem[i + 4] = ee; _mem[i + 5] = ff; _mem[i + 6] = gg; _mem[i + 7] = hh;
74 }
75
76 for ( int i = 0 ; i < Size ; i += 8 ) {
77 aa += _mem[i + 0]; bb += _mem[i + 1]; cc += _mem[i + 2]; dd += _mem[i + 3];
78 ee += _mem[i + 4]; ff += _mem[i + 5]; gg += _mem[i + 6]; hh += _mem[i + 7];
79
80 Shuffle(ref aa, ref bb, ref cc, ref dd, ref ee, ref ff, ref gg, ref hh);
81
82 _mem[i + 0] = aa; _mem[i + 1] = bb; _mem[i + 2] = cc; _mem[i + 3] = dd;
83 _mem[i + 4] = ee; _mem[i + 5] = ff; _mem[i + 6] = gg; _mem[i + 7] = hh;
84 }

86 Isaac();

114 private static void Shuffle ( ref uint a, ref uint b, ref uint c, ref uint d,
115 ref uint e, ref uint f, ref uint g, ref uint h ) {
116 a ^= b << 11; d += a; b += c;
117 b ^= c >> 2; e += b; c += d;
118 c ^= d << 8; f += c; d += e;
119 d ^= e >> 16; g += d; e += f;
120 e ^= f << 10; h += e; f += g;
121 f ^= g >> 4; a += f; g += h;
122 g ^= h << 8; b += g; h += a;
123 h ^= a >> 9; c += h; a += b;
124 }

131 private static uint Ind ( uint[] mm, uint x ) {
132 // 32‑Bit: Index = (x & (255 << 2)) / 4
133 int idx = (int)((x & (255u << 2)) >> 2);
134 return mm[idx];
135 }

149 private static void RngStep ( uint mix, ref uint a, ref uint b,
150 uint[] mm, ref int m, ref int m2,
151 uint[] r, ref int rIdx, ref uint x, ref uint y ) {
152 x = mm[m];
153 a = (a ^ mix) + mm[m2++];
154 y = mm[m] = Ind(mm, x) + a + b;
155 r[rIdx++] = b = Ind(mm, y >> 8) + x;
156 m++;
157 }

162 private void Isaac () {
163 uint x = 0, y = 0;
164 uint[] mm = _mem;
165 uint[] r = _rsl;
166
167 uint a = _a;
168 uint b = _b + (++_c);
169
170 int m = 0;
171 int m2 = 128;
172 int rIdx = 0;
173
174 for ( ; m < 128 ; ) {
175 RngStep(a << 13, ref a, ref b, mm, ref m, ref m2, r, ref rIdx, ref x, ref y);
176 RngStep(a >> 6, ref a, ref b, mm, ref m, ref m2, r, ref rIdx, ref x, ref y);
177 RngStep(a << 2, ref a, ref b, mm, ref m, ref m2, r, ref rIdx, ref x, ref y);
178 RngStep(a >> 16, ref a, ref b, mm, ref m, ref m2, r, ref rIdx, ref x, ref y);
179 }
180
181 m2 = 0;
182 for ( ; m2 < 128 ; ) {
183 RngStep(a << 13, ref a, ref b, mm, ref m, ref m2, r, ref rIdx, ref x, ref y);
184 RngStep(a >> 6, ref a, ref b, mm, ref m, ref m2, r, ref rIdx, ref x, ref y);
185 RngStep(a << 2, ref a, ref b, mm, ref m, ref m2, r, ref rIdx, ref x, ref y);
186 RngStep(a >> 16, ref a, ref b, mm, ref m, ref m2, r, ref rIdx, ref x, ref y);
187 }
188
189 _b = b;
190 _a = a;
191 }
