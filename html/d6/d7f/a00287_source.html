425 public void Sort () {
426 if ( m_sorter != null ) {
427 m_sorter(ref m_map, DEFAULTCOMPAR);
428 } else {
429 // Fallback
430 for ( int i = 0 ; i < Count - 1 ; i++ ) {
431 for ( int j = i + 1 ; j < Count ; j++ ) {
432 bool cmp = DEFAULTCOMPAR.Compare( m_map.ElementAt(i), m_map.ElementAt(j) );
433
434 if ( !cmp ) {
435 Swap(i, j);
436 }
437 }

457 private Pair<TElement, TPriority>? GetClosestToMedian () {
458 int n = m_map.Length;
459 if ( n == 0 ) return null;
460
461 Vector<TPriority> prios = new Vector<TPriority>(n);
462
463 var set = Vector<TPriority>.AsMultiSet(ref prios, new Less<TPriority>());
464
465 for(int i= 0 ; i < m_map.Length ; i++ )
466 set.Insert(i, m_map[i].Second);
467
468 TPriority median;
469 if ( (n & 1) == 1 ) {
470 median = prios[n / 2];
471 } else {
472 TPriority a = prios[(n / 2) - 1];
473 TPriority b = prios[n / 2];
474 median = (a + b) / TPriority.CreateChecked(2);
475 }
476
477 Pair<TElement, TPriority> best = m_map[0];
478 TPriority bestDist = Abs(best.Second - median);
479
480 for ( int i = 1; i < n; i++ ) {
481 var p = m_map[i];
482 TPriority dist = Abs(p.Second - median);
483
484 if ( dist < bestDist ) {
485 best = p;
486 bestDist = dist;
487 }
488 }
489
490 return best;
491 }
