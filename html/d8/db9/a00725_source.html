154 public virtual bool Lock ( int millisecondsTimeout = -1) {
155 bool _ret = false;
156 bool _usedTimer = millisecondsTimeout > 0;
157
158
159
160 if ( _usedTimer ) {
161 long start = Environment.TickCount64;
162
163 while ( true ) {
164
165 // Timeout erreicht?
166 if ( millisecondsTimeout >= 0 ) {
167 long elapsed = Environment.TickCount64 - start;
168 if ( elapsed >= millisecondsTimeout )
169 break; // nicht geschafft
170 }
171
172 // Wenn keine Kapazität → sofort spinnen
173 if ( IsHeld ) {
174 Thread.Yield();
175 continue;
176 }
177
178 // Atomar dekrementieren
179 m_counter--;
180 long newValue = m_counter.Value;
181
182 // Erfolg: nicht unter minCapacity gerutscht
183 if ( newValue >= m_min ) {
184 _ret = true;
185 break;
186 }
187
188 // Unter minCapacity gerutscht → revert + spin
189 m_counter++;
190 Thread.Yield();
191 }
192 } else {
193
194 while ( true ) {
195
196 // Wenn keine Kapazität → sofort spinnen
197 if ( IsHeld ) {
198 Thread.SpinWait(1);
199
200 continue;
201 }
202
203 // Atomar dekrementieren
204 m_counter--;
205 long newValue = m_counter.Value;
206
207 // Erfolg: nicht unter minCapacity gerutscht
208 if ( newValue >= m_min ) {
209 _ret = true;
210 break;
211 }
212
213 // Unter minCapacity gerutscht → revert + spin
214 m_counter.Increment();
215 Thread.SpinWait(1);
216 }
217
218 }
219
220 return _ret;
221 }
