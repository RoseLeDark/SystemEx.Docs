215 public virtual double NextDouble ( double minValue, double maxValue ) {
216 if ( minValue > maxValue )
217 throw new ArgumentException("minValue must be <= maxValue");
218
219 double value;
220
221 // Generate a normal IEEE‑754 double using your byte‑based system
222 do {
223 var tmp = RandUtils.GetArray(8);
224 value = tmp.ToDouble();
225 }
226 while ( !double.IsNormal(value) );
227
228
229 // Scale to [minValue, maxValue]
230 return minValue + value * (maxValue - minValue);
231 }
