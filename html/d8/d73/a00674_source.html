171 public static bool IsInteger ( FloatE5M2 value ) {
172 if ( IsNaN(value) ) return false;
173
174 // Unterhalb vom Bias (1.0) ist nur die Null eine Ganzzahl
175 if ( value.Exponent < 15 ) return IsZero(value);
176
177 // Ab Exponent 17 (Bias + 2) schiebt sich das Komma komplett hinter die Mantisse.
178 // Ab hier ist JEDER darstellbare Wert (auch Infinity) eine Ganzzahl.
179 if ( value.Exponent >= 17 ) return !IsInfinity(value);
180
181 int shift = value.Exponent.Value - 15;
182 return (value.Mantissa & (1 >> shift) ) == 0;
183 }

253 public static FloatE5M2 Add ( FloatE5M2 a, FloatE5M2 b ) {
254 // 1. Sonderfälle (NaN und Infinity) direkt auf Bit-Ebene abfangen
255 if ( IsNaN(a) || IsNaN(b) ) return NaN;
256 if ( IsInfinity(a) ) return IsInfinity(b) && (a.Sign != b.Sign) ? NaN : a;
257 if ( IsInfinity(b) ) return b;
258 if ( IsZero(a) ) return b;
259 if ( IsZero(b) ) return a;
260
261 Fast_Byte expA = a.Exponent;
262 Fast_Byte expB = b.Exponent;
263
264 // Implizites Bit (Hidden Bit) hinzufügen: 1.0 bei normalen Zahlen, 0.0 bei subnormalen
265 Fast_Byte mantA = (byte)(a.Mantissa | (expA != 0 ? 0x04 : 0x00));
266 Fast_Byte mantB = (byte)(b.Mantissa | (expB != 0 ? 0x04 : 0x00));
267
268 // Exponenten anpassen (Subnormale Zahlen haben implizit Exponent 1)
269 Fast_Int realExpA = expA == 0 ? 1 : expA;
270 Fast_Int realExpB = expB == 0 ? 1 : expB;
271
272 Fast_Int finalExp = realExpA;
273 byte finalSign = 0;
274
275 // --- Nutzung deines größeren Registers für die Berechnung ---
276 // Wir shiften die extrahierten Basismantissen nach links in das Fixpunktraster
277 Fast_UShort shiftedA = (ushort)(mantA << 4);
278 Fast_UShort shiftedB = (ushort)(mantB << 4);
279 Fast_UShort resMant = 0;
280
281 // 1. Schritt: Mantissen auf denselben Exponenten ausrichten
282 if ( realExpA >= realExpB ) {
283 var shift = realExpA - realExpB;
284 shiftedB >>= (int)shift;
285 finalExp = realExpA;
286 } else {
287 var shift = realExpB - realExpA;
288 shiftedA >>= (int)shift;
289 finalExp = realExpB;
290 }
291
292 // 2. Schritt: Mathematische Addition / Subtraktion basierend auf den echten Vorzeichen
293 if ( a.Sign == b.Sign ) {
294 // Gleiche Vorzeichen -> Einfach addieren
295 resMant = (ushort)(shiftedA + shiftedB);
296 finalSign = (byte)(a.Sign ? 1 : 0);
297 } else {
298 // Unterschiedliche Vorzeichen -> Subtrahieren (Kleinere von Größerer abziehen)
299 if ( shiftedA >= shiftedB ) {
300 resMant = (ushort)(shiftedA - shiftedB);
301 finalSign = (byte)(a.Sign ? 1 : 0);
302 } else {
303 resMant = (ushort)(shiftedB - shiftedA);
304 finalSign = (byte)(b.Sign ? 1 : 0);
305 }
306 }
307
308 if ( resMant == 0 ) return finalSign == 1 ? NegativeZero : Zero;
309
310 // 3. Schritt: Renormalisierung des Ergebnisses im größeren Register
311 // Wir suchen das führende Bit im 4er-Shiftraster (0x08 << 4 entspricht dezimal 128)
312 while ( resMant >= 0x80 ) {
313 resMant >>= 1;
314 finalExp++;
315 }
316 while ( resMant < 0x40 && finalExp > 1 ) // (0x04 << 4 entspricht dezimal 64)
317 {
318 resMant <<= 1;
319 finalExp--;
320 }
321
322 // Zurück in das 2-Bit-E5M2-Raster schieben (Fixpunktraster entfernen)
323 resMant >>= 4;
324 resMant &= 0x03; // Hidden Bit auf Bit-Position 2 entfernen
325
326 // Überlauf zu Infinity prüfen (Der maximale normale Exponent ist 30)
327 if ( finalExp >= 0x1F )
328 return finalSign == 1 ? NegativeInfinity : PositiveInfinity;
329
330 // Unterlauf zu Subnormal prüfen
331 if ( finalExp <= 0 )
332 return new FloatE5M2(finalSign, 0, (byte)resMant);
333
334 return new FloatE5M2(finalSign, (byte)finalExp, (byte)resMant);
335 }

337 public static FloatE5M2 Mul ( FloatE5M2 a, FloatE5M2 b ) {
338 // 1. Sonderfälle auf Bit-Ebene prüfen
339 if ( IsNaN(a) || IsNaN(b) ) return NaN;
340
341 // Kreuzprüfung für Unendlich und Null
342 if ( IsInfinity(a) || IsInfinity(b) )
343 return (IsZero(a) || IsZero(b)) ? NaN : ((a.Sign ^ b.Sign) ? NegativeInfinity : PositiveInfinity);
344
345 if ( IsZero(a) || IsZero(b) )
346 return (a.Sign ^ b.Sign) ? NegativeZero : Zero;
347
348 // Vorzeichen über XOR bestimmen
349 byte finalSign = (byte)(a.Sign ^ b.Sign ? 1 : 0);
350
351 Fast_Byte expA = a.Exponent;
352 Fast_Byte expB = b.Exponent;
353
354 // Mantissen extrahieren inklusive Hidden Bit (Bit Position 2)
355 Fast_Byte mantA = (byte)(a.Mantissa | (expA != 0 ? 0x04 : 0x00));
356 Fast_Byte mantB = (byte)(b.Mantissa | (expB != 0 ? 0x04 : 0x00));
357
358 // Exponenten für subnormale Zahlen korrigieren
359 Fast_Int realExpA = expA == 0 ? 1 : expA;
360 Fast_Int realExpB = expB == 0 ? 1 : expB;
361
362 // Exponenten addieren und den E5M2-Bias (15) abziehen
363 Fast_Int finalExp = realExpA + realExpB - 15;
364
365 // --- Nutzung deines größeren Registers für die Berechnung ---
366 // Multiplikation der Mantissen (Maximalwert: 7 * 7 = 49, passt locker in Fast_UShort / Fast_UInt)
367 Fast_UShort resMant = (ushort)(mantA * mantB);
368
369 if ( resMant == 0 ) return finalSign == 1 ? NegativeZero : Zero;
370
371 // Renormalisierung: Da wir zwei 3-Bit-Zahlen multipliziert haben,
372 // wandert das implizite Hidden Bit standardmäßig auf Position 4 (Wert 16 / 0x10).
373 while ( resMant >= 0x10 ) {
374 resMant >>= 1;
375 finalExp++;
376 }
377 while ( resMant < 0x04 && finalExp > 1 ) {
378 resMant <<= 1;
379 finalExp--;
380 }
381
382 // Hidden Bit auf Position 2 entfernen (Wert 4 ausmaskieren)
383 resMant &= 0x03;
384
385 // Überlauf zu Infinity prüfen
386 if ( finalExp >= 0x1F )
387 return finalSign == 1 ? NegativeInfinity : PositiveInfinity;
388
389 // Unterlauf zu Subnormal prüfen
390 if ( finalExp <= 0 )
391 return new FloatE5M2(finalSign, 0, (byte)resMant);
392
393 return new FloatE5M2(finalSign, (byte)finalExp, (byte)resMant);
394 }

396 public static FloatE5M2 Div ( FloatE5M2 a, FloatE5M2 b ) {
397 // 1. Sonderfälle auf Bit-Ebene abfangen
398 if ( IsNaN(a) || IsNaN(b) ) return NaN;
399 if ( IsZero(b) ) return IsZero(a) ? NaN : (a.Sign ^ b.Sign ? NegativeInfinity : PositiveInfinity);
400 if ( IsInfinity(b) ) return IsInfinity(a) ? NaN : (a.Sign ^ b.Sign ? NegativeZero : Zero);
401 if ( IsZero(a) || IsInfinity(a) ) return a.Sign ^ b.Sign ? Negate(a) : a;
402
403 // Vorzeichen bestimmen
404 byte finalSign = (byte)(a.Sign ^ b.Sign ? 1 : 0);
405
406 Fast_Byte expA = a.Exponent;
407 Fast_Byte expB = b.Exponent;
408
409 Fast_Byte mantA = (byte)(a.Mantissa | (expA != 0 ? 0x04 : 0x00));
410 Fast_Byte mantB = (byte)(b.Mantissa | (expB != 0 ? 0x04 : 0x00));
411
412 Fast_Int realExpA = expA == 0 ? 1 : expA;
413 Fast_Int realExpB = expB == 0 ? 1 : expB;
414
415 // Exponenten subtrahieren und Bias (15) wieder aufschlagen
416 Fast_Int finalExp = realExpA - realExpB + 15;
417
418 // --- Nutzung deines größeren Registers für die Bruchrechnung ---
419 // Wir schieben den Zähler nach links, um genug "Futter" für die Ganzzahldivision zu haben
420 Fast_UInt extendedMantA = (uint)(mantA << 4);
421 Fast_UInt resMant = (uint)(extendedMantA / mantB);
422
423 if ( resMant == 0 ) return finalSign == 1 ? NegativeZero : Zero;
424
425 // Renormalisierung im Shiftraster
426 while ( resMant >= 0x08 ) {
427 resMant >>= 1;
428 finalExp++;
429 }
430 while ( resMant < 0x04 && finalExp > 1 ) {
431 resMant <<= 1;
432 finalExp--;
433 }
434
435 // Hidden Bit isolieren/löschen (nur die untersten 2 Bits behalten)
436 resMant &= 0x03;
437
438 // Grenzen validieren
439 if ( finalExp >= 0x1F )
440 return finalSign == 1 ? NegativeInfinity : PositiveInfinity;
441
442 if ( finalExp <= 0 )
443 return new FloatE5M2(finalSign, 0, (byte)resMant);
444
445 return new FloatE5M2(finalSign, (byte)finalExp, (byte)resMant);
446 }

635 internal static byte Encode ( byte sign, byte exponent, byte mantissa ) {
636 Fast_Byte b = 0;
637
638
639 b.At(0, (byte)((mantissa >> 0) & 1));
640 b.At(1, (byte)((mantissa >> 1) & 1));
641
642 b.At(2, (byte)((exponent >> 0) & 1));
643 b.At(3, (byte)((exponent >> 1) & 1));
644 b.At(4, (byte)((exponent >> 2) & 1));
645 b.At(5, (byte)((exponent >> 3) & 1));
646 b.At(6, (byte)((exponent >> 4) & 1));
647 b.At(7, sign); // Signbit
648
649
650
651 return b.Value;
