99 public static class Layout {
119 public static unsafe bool Check<T>(uint expectedUnmanagedSize) where T : unmanaged {
120 Type t = typeof(T);
121 long _sizeof = sizeof(T);
122 int _marshalSize = Marshal.SizeOf<T>();
123
124 // StructLayout must be Sequential
125 var attr = t.StructLayoutAttribute;
126 if ( attr == null || attr.Value != LayoutKind.Sequential )
127 throw new MissingStructLayoutSequentialException(t);
128
129 // sizeof(T) must match Marshal.SizeOf<T>()
130 if ( _marshalSize != _sizeof )
131 throw new SerializationException(
132 $"{t.Name} size mismatch: MarshalSize={_marshalSize}, sizeof={_sizeof}");
133
134 // Optional unmanaged size check
135 if ( expectedUnmanagedSize != 0 ) {
136 if ( _sizeof != expectedUnmanagedSize )
137 throw new SizeMismatchException(t, _sizeof, expectedUnmanagedSize);
138 }
139
140 return true;
141 }
142 }
