490 public int Abort (bool hardt) {
491 m_continuemutex.Lock (-1) ;
492 m_runningMutex.Lock(-1) ;
493
494 if ( !m_bRunning ) {
495 m_runningMutex.Unlock();
496 m_continuemutex.Unlock();
497
498 return 1; // ERR_TASK_NOTRUNNING;
499 }
500
501 // nicht aus sich selbst killen
502 if ( Thread.CurrentThread.ManagedThreadId == m_base.ManagedThreadId ) {
503
504 m_runningMutex.Unlock();
505 m_continuemutex.Unlock();
506
507 return 42; // ERR_TASK_CALLFROMSELFTASK
508 }
509
510
511
512
513 m_runn = false;
514 m_eState = (hardt)? ThreadExState.RequestKill : ThreadExState.RequestAbort;
515 if ( hardt ) m_base.Interrupt();
516
517 if ( IsSuspend() )
518 m_eventGroup.Set(EVENTGROUP_BIT_CONTINUE);
519
520 m_contextMutext.Lock(-1);
521 OnStop(this, hardt);
522 m_contextMutext.Unlock();
523
524 m_runningMutex.Unlock();
525 m_continuemutex.Unlock();
526
527
528
529 return 0; // ERR_TASK_OK;

733 protected virtual int OnRunning ( ThreadEx e, Object? userdata ) {
734
735
736
737 int _ret = 0;
738
739 while ( m_runn && _ret == 0) {
740
741 if ( m_eventGroup.IsSet(EVENTGROUP_BIT_INVOKESTATE) ) {
742
743 cheackEndContext(false);
744 }
745
746 if( m_eventGroup.IsSet(EVENTGROUP_BIT_INVOKEMESSAGE) ) {
747 m_eventGroup.Clear(EVENTGROUP_BIT_INVOKEMESSAGE);
748
749 while (!m_contextQueue.IsEmpty) {
750 Optional<Triple<byte, ThreadInvokeMessage, object>> _ref = new Triple<byte, ThreadInvokeMessage, object>();
751
752 if ( !m_contextQueue.PopFront(ref _ref) )
753 continue;
754
755
756 var msgType = _ref.Value!.Second;
757 var payload = _ref.Value!.Third;
758
759 switch ( msgType ) {
760
761 case ThreadInvokeMessage.Message:
762 m_messageQueue.PushBack(payload);
763 break;
764
765 case ThreadInvokeMessage.Stack:
766 if ( payload is Pair<byte, object> g )
767 m_coRoutineStack.Push(g.Second, g.First);
768 break;
769
770 case ThreadInvokeMessage.Propertie:
771 break;
772 case ThreadInvokeMessage.CoRoutine:
773 break;
774 }
775 }
776
777 }
778
779 if ( IsSuspend() ) {
780 m_eState = ThreadExState.Pause;
781 WaitOfEvent(EVENTGROUP_BIT_CONTINUE, -1);
782 if ( OnResume != null ) OnResume.Invoke(e, userdata);
783
784
785 m_eventGroup.Clear(EVENTGROUP_BIT_CONTINUE);
786 m_eventGroup.Clear(EVENTGROUP_BIT_PAUSE);
787
788 } else {
789 m_eState = ThreadExState.Running;
790 if ( OnTask != null ) {
791 _ret = OnTask.Invoke(e, userdata);
792 }
793 }
794 }
795
796 return _ret;
797 }

822 private bool cheackEndContext(bool abort) {
823
824 if ( m_CommitWindow.ElapsedMilliseconds <= m_CommitWinOpen )
825 return false;
826
827 if ( abort ) {
828 m_contextQueue.Clear();
829 } else if ( !m_contextQueue.IsEmpty ) {
830 m_eventGroup.Set((byte)(EVENTGROUP_BIT_INVOKESTATE + 1));
831 }
832
833 m_eventGroup.Clear(EVENTGROUP_BIT_INVOKESTATE);
834 m_CommitWindow.Stop();
835 m_CommitWindow.Reset();
836 m_contextMutext.Unlock();
837
838 return true;

874 //thread.m_runningMutex.Unlock();

876
877
878 if ( thread.OnSetup(thread, thread.m_userData) ) {
879
880 thread.m_runningMutex.Unlock();
881 ret = thread.OnRunning(thread, thread.m_userData);
882
883
884 thread.OnExit(thread, thread.m_userData);
885 }
886
887
888 thread.m_runningMutex.Lock (-1) ;
889 thread.m_bRunning = false;
890 thread.m_eState = ThreadExState.Stopped;
891 thread.m_retval = ret;
892 thread.m_runningMutex.Unlock();
893 thread.m_continuemutex.Unlock();
894
895 // set the join bit

897 }
