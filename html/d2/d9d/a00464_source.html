138 public override ulong WriteRange(ulong start, ulong iend, byte[] data) {
139 if ( iend <= start ) return 0;
140 if ( iend > m_totalLength ) iend = m_totalLength;
141
142 ulong rangeLen = iend - start;
143 ulong writableLen = (ulong)data.Length < rangeLen ? (ulong)data.Length : rangeLen;
144
145 ulong totalWritten = 0;
146 ulong pos = start;
147 int dataOffset = 0;
148
149 ulong segmentSize = base.Length;
150
151 while ( totalWritten < writableLen ) {
152 ulong cacheIdx = pos / segmentSize;
153 ulong cacheOff = pos % segmentSize;
154
155 if ( cacheIdx >= (ulong)(m_caches.Length + 1) )
156 break;
157
158 ulong spaceInCache = segmentSize - cacheOff;
159 ulong remaining = writableLen - totalWritten;
160 uint chunkLen = (uint)((remaining < spaceInCache) ? remaining : spaceInCache);
161
162 if ( chunkLen == data.Length && dataOffset == 0 && cacheOff == 0 ) {
163 ulong written = (cacheIdx == 0)
164 ? base.WriteRange(cacheOff, cacheOff + (ulong)chunkLen, data)
165 : m_caches[cacheIdx - 1].WriteRange(cacheOff, cacheOff + (ulong)chunkLen, data);
166
167 if ( written == 0 ) break;
168
169 totalWritten += written;
170 pos += written;
171 dataOffset += (int)written;
172 } else {
173 m_segmentTemp.CopyFrom(new FixedVector<byte>(data), (uint)dataOffset, 0, chunkLen);
174 byte[] internalBuf = m_segmentTemp.ToNative();
175
176 ulong writtenChunk = (cacheIdx == 0)
177 ? base.WriteRange(cacheOff, cacheOff + (ulong)chunkLen, internalBuf)
178 : m_caches[cacheIdx - 1].WriteRange(cacheOff, cacheOff + (ulong)chunkLen, internalBuf);
179
180 if ( writtenChunk == 0 ) break;
