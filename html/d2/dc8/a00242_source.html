119 public Quatd ( Vec3d angles ) {
120 double cos_z_2 = System.Math.Cos(0.5*angles.Z);
121 double cos_y_2 = System.Math.Cos(0.5*angles.Y);
122 double cos_x_2 = System.Math.Cos(0.5*angles.X);
123
124 double sin_z_2 = System.Math.Sin(0.5*angles.Z);
125 double sin_y_2 = System.Math.Sin(0.5*angles.Y);
126 double sin_x_2 = System.Math.Sin(0.5*angles.X);
127
128 // and now compute quaternion
129 m_s = cos_z_2 * cos_y_2 * cos_x_2 + sin_z_2 * sin_y_2 * sin_x_2;
130 m_v = new Vec3d
131 (
132 cos_z_2 * cos_y_2 * sin_x_2 - sin_z_2 * sin_y_2 * cos_x_2,
133 cos_z_2 * sin_y_2 * cos_x_2 + sin_z_2 * cos_y_2 * sin_x_2,
134 sin_z_2 * cos_y_2 * cos_x_2 - cos_z_2 * sin_y_2 * sin_x_2
135 );
136 m_comparebleMode = CompareType.Skalar;
137 }

221 public static Quatd FromAxis ( AxisAngle<Vec3d, double> axisAngle ) {
222
223 double len = Vec3d.LenghtSqrt(axisAngle.Axis);
224 Quatd temp = new Quatd();
225
226 if ( len > 1e-12 ) {
227 double inv = 1.0 / len;
228
229 // Achse normalisieren
230 double nx = axisAngle.Axis.X * inv;
231 double ny = axisAngle.Axis.Y * inv;
232 double nz = axisAngle.Axis.Z * inv;
233
234 double omega = 0.5f * axisAngle.Angle;
235 double s = System.Math.Sin(omega);
236
237 temp.m_v = new Vec3d(s * nx, s * ny, s * nz);
238 temp.m_s = System.Math.Cos(omega);
239 } else {
240 // Ungültige Achse → Identity
241 temp.m_s = 1.0;
242 temp.m_v = Vec3d.Zero;
243 }
244
245 return Normalize(temp);
246

331 public static Quatd Power ( Quatd v, double Exp ) {
332 if ( System.Math.Abs(v.m_s) > .9999f ) {
333 return v;
334 }
335
336 // Erhalte halbe angle alpha (alpha = theta/2)
337 double alpha = System.Math.Acos(v.m_s);
338
339 // Berechne neuen alpha Wert
340 double newAlpha = alpha * Exp;
341
342 // Berechne neuen s wert
343 Quatd result = new Quatd();
344 result.m_s = System.Math.Cos(newAlpha);
345
346 // Berechne neue xyz Werte
347
348 double mult = System.Math.Sin(newAlpha) / System.Math.Sin(alpha);
349 result.m_v.X = v.m_v.X * mult;
350 result.m_v.Y = v.m_v.Y * mult;
351 result.m_v.Z = v.m_v.Z * mult;
352
353 return Normalize(result);
354 }

418 public static Quatd Slerp ( Quatd q0, Quatd q1, double t ) { // Spherical linear interpolation.
419 if ( t <= 0.0 ) return q0;
420 if ( t >= 1.0 ) return q1;
421
422 double cosOmega = DotProduct(q0, q1);
423
424 double q1w = q1.m_s;
425 double q1x = q1.m_v.X;
426 double q1y = q1.m_v.Y;
427 double q1z = q1.m_v.Z;
428
429 if ( cosOmega < 0.0 ) {
430 q1w = -q1w;
431 q1x = -q1x;
432 q1y = -q1y;
433 q1z = -q1z;
434 cosOmega = -cosOmega;
435 }
436
437 Debug.Assert(cosOmega < 1.0);
438
439 // Compute interpolation fraction, checking for quaternions
440 // almost exactly the same
441
442 double k0, k1;
443 if ( cosOmega > 0.9999 ) {
444 k0 = 1.0 - t;
445 k1 = t;
446 } else {
447 // Compute the sin of the angle using the
448 // trig identity sin^2(omega) + cos^2(omega) = 1
449
450 double sinOmega = System.Math.Sqrt(1.0 - cosOmega*cosOmega);
451
452 // Compute the angle from its sin and cosine
453
454 double omega = System.Math.Atan2(sinOmega, cosOmega);
455
456 // Compute inverse of denominator, so we only have
457 // to divide once
458
459 double oneOverSinOmega = 1.0 / sinOmega;
460
461 // Compute interpolation parameters
462
463 k0 = System.Math.Sin((1.0 - t) * omega) * oneOverSinOmega;
464 k1 = System.Math.Sin(t * omega) * oneOverSinOmega;
465 }
466
467 // Interpolate
468
469 Quatd result = new Quatd();
470
471 result.m_v.X = k0 * q0.m_v.X + k1 * q1x;
472 result.m_v.Y = k0 * q0.m_v.Y + k1 * q1y;
473 result.m_v.Z = k0 * q0.m_v.Z + k1 * q1z;
474 result.m_s = k0 * q0.m_s + k1 * q1w;
475
476 // Return it
477
478 return result;
479 }

483 public CompareResult CompareTo ( Quatd a ) {
484
485 CompareResult _ret = CompareResult.Equal;
486
487 double _a = 0;
488 double _b = 0;
489
490 switch ( m_comparebleMode ) {
491 case CompareType.Skalar:
492 _ret = (CompareResult)m_s.CompareTo(a.m_s);
493 break;
494 case CompareType.Norm:
495 _a = m_s * m_s + Vec3d.Lenght(m_v);
496 _b = a.m_s * a.m_s + Vec3d.Lenght(a.m_v);
497 _ret = (CompareResult)_a.CompareTo(_b);
498 break;
499 case CompareType.Rotation:
500 _a = GetRotationAngle();
501 _b = a.GetRotationAngle();
502 _ret = (CompareResult)_a.CompareTo(_b);
503 break;
504 }
505 return _ret;
506 }

580 public static Quatd operator * ( Quatd a, Quatd b ) {
581 Quatd qtmp = new Quatd();
582
583 double ax = a.m_v.X;
584 double ay = a.m_v.Y;
585 double az = a.m_v.Z;
586
587 double bx = b.m_v.X;
588 double by = b.m_v.Y;
589 double bz = b.m_v.Z;
590
591 double as_ = a.m_s;
592 double bs_ = b.m_s;
593
594 // Skalarteil
595 qtmp.m_s = as_ * bs_ - ax * bx - ay * by - az * bz;
596
597 // Vektorteil
598 qtmp.m_v = new Vec3d(
599 as_ * bx + bs_ * ax + ay * bz - az * by,
600 as_ * by + bs_ * ay + az * bx - ax * bz,
601 as_ * bz + bs_ * az + ax * by - ay * bx
602 );
603
604 return qtmp;
605 }
