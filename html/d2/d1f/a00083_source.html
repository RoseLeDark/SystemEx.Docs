56 public static Float8UMX<T> Add ( Float8UMX<T> a, Float8UMX<T> b ) {
57 T[] result = new T[32];
58 byte finalScale = System.Math.Max(a.m_sharedExponent, b.m_sharedExponent);
59
60 for ( int i = 0 ; i < 32 ; i++ ) {
61 // 1. Sonderfälle der Einzelelemente abfangen (über dein Interface)
62 if ( T.IsNaN(a[i]) || T.IsNaN(b[i]) ) { result[i] = T.NaN; continue; }
63 if ( T.IsZero(a[i]) && T.IsZero(b[i]) ) { result[i] = T.Zero; continue; }
64
65 // 2. Skalierungsdifferenz der MX-Blöcke berechnen
66 int scaleDiffA = finalScale - a.m_sharedExponent;
67 int scaleDiffB = finalScale - b.m_sharedExponent;
68
69 // 3. Komponenten direkt als Fast_Byte über dein Interface abgreifen
70 Fast_Byte expA = a[i].Exponent;
71 Fast_Byte expB = b[i].Exponent;
72
73 Fast_Byte mantA = (byte)(a[i].Mantissa | (expA != 0 ? a[i].HiddenBit : 0x00));
74 Fast_Byte mantB = (byte)(b[i].Mantissa | (expB != 0 ? b[i].HiddenBit : 0x00));
75
76 int realExpA = (expA == 0 ? 1 : expA.Value) - scaleDiffA;
77 int realExpB = (expB == 0 ? 1 : expB.Value) - scaleDiffB;
78
79 int finalElementExp = System.Math.Max(realExpA, realExpB);
80
81 // 4. Mantissen ausrichten im Fast_UShort / Fast_UInt Äquivalent
82 ushort shiftedA = (ushort)(mantA.Value << a[i].ShiftRaster.Value);
83 ushort shiftedB = (ushort)(mantB.Value << b[i].ShiftRaster.Value);
84
85 if ( realExpA >= realExpB ) {
86 shiftedB >>= (realExpA - realExpB);
87 } else {
88 shiftedA >>= (realExpB - realExpA);
89 }
90
91 Fast_UShort resMant = (ushort)(shiftedA + shiftedB);
92
93 if ( resMant == 0 ) { result[i] = T.Zero; continue; }
94
95 // 6. Renormalisierung im Rechenregister
96 while ( resMant >= ((Fast_UShort)a[i].HiddenBit << a[i].ShiftRaster ) ) {
97 resMant >>= 1;
98 finalElementExp++;
99 }
100 while ( resMant < (a[i].HiddenBit << ( (byte)a[i].ShiftRaster - 1)) && finalElementExp > 1 ) {
101 resMant <<= 1;
102 finalElementExp--;
103 }
104
105 resMant >>= (byte)a[i].ShiftRaster;
106 resMant &= (byte)a[i].MantissaMask; // Hidden Bit löschen
107
108 // 7. Erzeugung über deine statische Interface-Methode: FromComponent!
109 if ( finalElementExp <= 0 ) {
110 result[i] = T.FromComponent(0, (byte)resMant, 0);
111 } else if ( finalElementExp >= a[i].MaxExponent.Value ) {
112 result[i] = T.PositiveInfinity;
113 } else {
114 result[i] = T.FromComponent(0, (byte)resMant, (byte)finalElementExp);
115 }
116 }
117
118 return RenormalizeBlockOverflow(finalScale, result);

135 Fast_Byte expB = b[i].Exponent;
