159 protected virtual Triple<T, MetaFormat, object> OnDeserialize ( string raw ) {
160 // ---------------------------------------------------------
161 // 1. Versuch: JSON erkennen
162 // ---------------------------------------------------------
163 try {
164 JsonElement root = JsonSerializer.Deserialize<JsonElement>(raw);
165
166 if ( root.ValueKind == JsonValueKind.Object &&
167 root.TryGetProperty("output", out JsonElement outputElement) ) {
168 // Output extrahieren
169 T output;
170
171 if ( typeof(T) == typeof(string) ) {
172 output = (T)(object)outputElement.ToString();
173 } else {
174 output = JsonSerializer.Deserialize<T>(outputElement.GetRawText())
175 ?? throw new InvalidDataException("Failed to convert 'output' to target type.");
176 }
177
178 return new Triple<T, MetaFormat, object>(
179 output,
180 MetaFormat.AI_FORMAT_JSON,
181 root
182 );
183 }
184
185 // JSON erkannt, aber kein "output" → fallback
186 if ( typeof(T) == typeof(string) ) {
187 return new Triple<T, MetaFormat, object>(
188 (T)(object)raw,
189 MetaFormat.AI_FORMAT_JSON,
190 root
191 );
192 }
193
194 throw new InvalidDataException("JSON response missing 'output' field.");
195 } catch {
196 // ---------------------------------------------------------
197 // 2. Kein JSON → Fallback auf TEXT
198 // ---------------------------------------------------------
199 if ( typeof(T) == typeof(string) ) {
200 return new Triple<T, MetaFormat, object>(
201 (T)(object)raw,
202 MetaFormat.AI_FORMAT_TEXT,
203 raw
204 );
205 }
206
207 // ---------------------------------------------------------
208 // 3. Kein JSON + T ist nicht string → Fehler
209 // ---------------------------------------------------------
210 throw new InvalidDataException(
211 "Backend returned non-JSON and T is not string. Override OnDeserialize for custom formats."
212 );
213 }
214 }

291 public async Task<IModelResult<T>> InvokeAsync ( string systemPrompt, IModelPromp<T> input ) {
292 // Payload wird direkt sauber initialisiert
293 var payload = new WebAIBackendPayload<T>
294 {
295 URL = string.Empty,
296 Model = (string)Configuration.GetOrDefault("MODEL_NAME", "gpt-4o-mini"),
297
298 SystemPrompt = systemPrompt ?? string.Empty,
299 PromptRaw = input.Prompt,
300
301 Context = input.Context, // conversation metadata
302
303 Parameters = BuildWebParameters(input.Parameters),
304
305 Tools = new List<object>(),
306 ToolChoice = "no",
307
308 Metadata = new Map<string, object>
309 {
310 ["session_id"] = input.SessionId.HasValue ? input.SessionId.Value! : null,
311 ["tags"] = input.Tags,
312 ["backend"] = "WebBackend"
313 }
314 };
315
316 try {
317 Optional<object> urlOpt = Configuration.Get(WB_CONFIG_URL);
318 if ( urlOpt.IsNull || urlOpt.Value is not string baseUrl || string.IsNullOrWhiteSpace(baseUrl) )
319 throw new KeyNotFoundException("WB_CONFIG_URL is missing or invalid.");
320
321 payload.URL = baseUrl;
322
323 string finalUrl = OnCreateURL(payload);
324 if ( string.IsNullOrWhiteSpace(finalUrl) )
325 throw new InvalidDataException("OnCreateURL returned an empty URL.");
326
327 payload.URL = finalUrl;
328
329
330 var content = OnSerialize(payload);
331 if ( content == null )
332 throw new InvalidDataException("OnSerialize returned null content.");
333
334
335 HttpResponseMessage response;
336 try {
337 response = await m_http.PostAsync(finalUrl, content);
338 } catch ( Exception httpEx ) {
339 throw new HttpRequestException($"HTTP request failed: {httpEx.Message}", httpEx);
340 }
341
342 string raw = await response.Content.ReadAsStringAsync();
343
344 if ( !response.IsSuccessStatusCode ) {
345 throw new HttpRequestException(
346 $"Backend returned HTTP {(int)response.StatusCode} ({response.StatusCode}). Raw: {raw}"
347 );
348 }
349
350
351
352 Triple<T, MetaFormat, object> parsed = OnDeserialize(raw);
353 T output = parsed.First;
354
355 return new ModelResult<T>(
356 result: output,
357 metadata: new Map<ModelMeta, object>
358 {
359 [ModelMeta.AI_META_BACKEND] = BackendName,
360 [ModelMeta.AI_META_BACKEND_TYPE] = "WebAPI",
361 [ModelMeta.AI_META_MODEL] = ModelName ?? "unknown",
362 [ModelMeta.AI_META_MODEL_CAPS] = Capabilities,
363 [ModelMeta.AI_META_RAW] = raw,
364 [ModelMeta.AI_META_HTTP_STATUS] = response.StatusCode,
365 [ModelMeta.AI_META_HTTP_HEADERS] = response.Headers.ToString(),
366 [ModelMeta.AI_META_URL] = finalUrl,
367 [ModelMeta.AI_META_CONTENT_OUT] = parsed.Third,
368 [ModelMeta.AI_META_CONTENT_IN] = content,
369 [ModelMeta.AI_META_EXEC_TIME] = DateTime.UtcNow,
370 [ModelMeta.AI_META_FORMAT] = parsed.Second,
371 },
372 error: null,
373 raw: raw
374 );
375 } catch ( Exception ex ) {
376 return new ModelResult<T>(
377 result: default!,
378 metadata: new Map<ModelMeta, object>
379 {
380 [ModelMeta.AI_META_BACKEND] = BackendName,
381 [ModelMeta.AI_META_BACKEND_TYPE] = "WebAPI",
382 [ModelMeta.AI_META_MODEL] = ModelName ?? "unknown",
383 [ModelMeta.AI_META_MODEL_CAPS] = Capabilities,
384 [ModelMeta.AI_META_ERROR_MESSAGE] = ex.Message,
385 [ModelMeta.AI_META_ERROR_TYPE] = ex.GetType().Name,
386 [ModelMeta.AI_META_ERROR_STACK] = ex.StackTrace ?? "",
387 [ModelMeta.AI_META_ERROR_CODE] = ex.HResult,
388 [ModelMeta.AI_META_URL] = payload.URL
389 },
390 error: ex,
391 raw: null
392 );
393 }

398 private Map<string, object> BuildWebParameters ( Map<string, object> modelParams) {
399
400 var p = new Map<string, object>();
401
402 // 1. Model parameters (temperature, max_tokens, etc.)
403 p.PushBack(modelParams);
404
405 // 2. Configuration (API keys, runtime settings)
406 if ( m_configuration.TryGetValue(WB_CONFIG_TEMPERATUR, out var temp) )
407 p["temperature"] = temp;
408
409 if ( m_configuration.TryGetValue(WB_CONFIG_MAX_TOKENS, out var maxTok) )
410 p["max_tokens"] = maxTok;
411
412 if ( m_configuration.TryGetValue(WB_CONFIG_TOP_P, out var topP) )
413 p["top_p"] = topP;
414
415 if ( m_configuration.TryGetValue(WB_CONFIG_RESPONSE_FORMAT, out var respFmt) )
416 p["response_format"] = respFmt;
417
418 if ( m_configuration.TryGetValue(WB_CONFIG_STREAM, out var stream) )
419 p["stream"] = stream;
420
421
422#if WITH_OTIONAL
423 // 3. Environment (optional, ChatGPT ignoriert diese - nur für passende User APIs
424 foreach ( var kv in m_environment )
425 p[$"env_{kv.First}"] = kv.Second;
426#endif
427 return p;
428 }
