58 public static ColorR8G8B8 ToColorR8G8B8(this ColorXYZ c) {
59 float X = c.X;
60 float Y = c.Y;
61 float Z = c.Z;
62
63 // XYZ → linear RGB
64 float r_lin = 3.2404542f * X - 1.5371385f * Y - 0.4985314f * Z;
65 float g_lin = -0.9692660f * X + 1.8760108f * Y + 0.0415560f * Z;
66 float b_lin = 0.0556434f * X - 0.2040259f * Y + 1.0572252f * Z;
67
68 // linear RGB → sRGB (Gamma)
69 float r = (r_lin <= 0.0031308f) ? 12.92f * r_lin : 1.055f * MathF.Pow(r_lin, 1f / 2.4f) - 0.055f;
70 float g = (g_lin <= 0.0031308f) ? 12.92f * g_lin : 1.055f * MathF.Pow(g_lin, 1f / 2.4f) - 0.055f;
71 float b = (b_lin <= 0.0031308f) ? 12.92f * b_lin : 1.055f * MathF.Pow(b_lin, 1f / 2.4f) - 0.055f;
72
73 // Clamp (sRGB muss 0..1 sein)
74 return new ColorR8G8B8(r < 0f ? 0f : (r > 1f ? 1f : r),
75 g < 0f ? 0f : (g > 1f ? 1f : g),
76 b < 0f ? 0f : (b > 1f ? 1f : b)

134 public static ColorHSV ToColorHSV(this ColorR8G8B8 x) {
135 float max = System.Math.Max(x.Red, System.Math.Max(x.Green, x.Blue));
136 float min = System.Math.Min(x.Red, System.Math.Min(x.Green, x.Blue));
137 float delta = max - min;
138
139 float h = 0f;
140 float s = 0f;
141 float v = max;
142
143 if ( delta > 0f ) {
144 s = delta / max;
145
146 if ( max == x.Red )
147 h = 60f * (((x.Green - x.Blue) / delta) % 6f);
148 else if ( max == x.Green )
149 h = 60f * (((x.Blue - x.Red) / delta) + 2f);
150 else
151 h = 60f * (((x.Red - x.Green) / delta) + 4f);
152
153 if ( h < 0f )
154 h += 360f;
155 }
156
157 return new ColorHSV(h, s, v);

162 public static ColorHWB ToColorHWB(this ColorR8G8B8 color) {
163 var c = color.AsNormalized();
164
165 float max = System.Math.Max(c.Red, System.Math.Max(c.Green, c.Blue));
166 float min = System.Math.Min(c.Red, System.Math.Min(c.Green, c.Blue));
167 float delta = max - min;
168
169 float w = min;
170 float b = 1 - max;
171 float h;
172
173 if ( delta == 0 )
174 h = 0;
175 else if ( max == c.Red )
176 h = (c.Green - c.Blue) / delta;
177 else if ( max == c.Green )
178 h = 2 + (c.Blue - c.Red) / delta;
179 else
180 h = 4 + (c.Red - c.Green) / delta;
181
182 h *= 60;
183 if ( h < 0 ) h += 360;
184
185 return new ColorHWB(h, w *= 100, b *= 100);

190 public static ColorR8G8B8 ToColorR8G8B8(this ColorHWB color) {
191 float h = Math.FMod(color.H, 360.0f) / 60.0f;
192 float w = color.Whiteness / 100.0f;
193 float bl = color.Blackness / 100.0f;
194
195 float v = 1 - bl;
196 float s = v == 0 ? 0 : 1 - w / v;
197 float c = v * s;
198 float x = c * (1 - System.Math.Abs( Math.FMod (h, 2.0f) - 1f));
199 float m = v - c;
200
201 float r, g, b;
202 if ( h < 1 ) { r = c; g = x; b = 0; }
203 else if ( h < 2 ) { r = x; g = c; b = 0; }
204 else if ( h < 3 ) { r = 0; g = c; b = x; }
205 else if ( h < 4 ) { r = 0; g = x; b = c; }
206 else if ( h < 5 ) { r = x; g = 0; b = c; }
207 else { r = c; g = 0; b = x; }
208
209 return new ColorR8G8B8(
210 ((r + m)),
211 ((g + m)),
212 ((b + m))
213 );

219 public static ColorR8G8B8 ToColorR8G8B8(this ColorHSV color) {
220 float C = color.V * color.S; // Chroma
221 float Hp = Math.FMod(color.H / 60f, 6f); // Hue' (0..6)
222 float X = C * (1f - System.Math.Abs(Math.FMod(Hp, 2f) - 1f));
223 float m = color.V - C;
224
225 float r, g, b;
226
227 if ( Hp < 1f ) { r = C; g = X; b = 0; }
228 else if ( Hp < 2f ) { r = X; g = C; b = 0; }
229 else if ( Hp < 3f ) { r = 0; g = C; b = X; }
230 else if ( Hp < 4f ) { r = 0; g = X; b = C; }
231 else if ( Hp < 5f ) { r = X; g = 0; b = C; }
232 else if ( Hp < 6f ) { r = C; g = 0; b = X; }
233 else { r = 0; g = 0; b = 0; }
234
235 return new ColorR8G8B8(r + m, g + m, b + m);
