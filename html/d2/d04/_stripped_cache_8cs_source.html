77 public override FlexSpan<byte> AsFlexSpan ( long start, FlexSpanMode mode = FlexSpanMode.System ) {
78 if ( start < 0 || start >= (long)LongLength )
79 throw new ArgumentOutOfRangeException(nameof(start));
80
81 long segmentIndex = start / (long)base.LongLength;
82 long localOffset = start % (long)base.LongLength;
83
84 if ( segmentIndex == 0 ) {
85 // Base cache
86 return base.AsFlexSpan(localOffset, mode);
87 } else {
88 // Other segments
89 return m_caches[segmentIndex - 1].AsFlexSpan(localOffset, mode);
90 }
91 }

99 public StrippedCache (int cacheCount, int cacheSize) : base(cacheSize, CacheType.Both) {
100
101 m_caches = new Cache[cacheCount-1];
102 for ( int i = 0; i < cacheCount-1; i++ )
103 m_caches[i] = new Cache(cacheSize, CacheType.Both);
104
105 LongLength = (ulong)(cacheCount * cacheSize);
106 // m_currentCache = 0;
107 SetSavePosition(0);
108
109 m_segmentTemp = new FixedArray<byte>(cacheSize);
110 }

119 public override ulong WriteRange(ulong position, byte[] data) {
120 ulong written = WriteRange(Position + position, (ulong)data.LongLength, data);
121 SetSavePosition(Position + written);
122 return written;

133 public override ulong WriteRange(ulong start, ulong iend, byte[] data) {
134 if ( iend <= start ) return 0;
135 if ( iend > LongLength ) iend = LongLength;
136
137 ulong rangeLen = iend - start;
138 ulong writableLen = (ulong)data.Length < rangeLen ? (ulong)data.Length : rangeLen;
139
140 ulong totalWritten = 0;
141 ulong pos = start;
142 int dataOffset = 0;
143 ulong segmentSize = base.LongLength;
144
145 while ( totalWritten < writableLen ) {
146 ulong cacheIdx = pos / segmentSize;
147 ulong cacheOff = pos % segmentSize;
148 if ( cacheIdx >= (ulong)(m_caches.Length + 1) ) break;
149
150 ulong spaceInCache = segmentSize - cacheOff;
151 ulong remaining = writableLen - totalWritten;
152 uint chunkLen = (uint)((remaining < spaceInCache) ? remaining : spaceInCache);
153
154 // Fast path: ganzes data-Array passt und kein Offset nötig
155 if ( chunkLen == data.Length && dataOffset == 0 && cacheOff == 0 ) {
156
157 ulong written = (cacheIdx == 0)
158 ? base.WriteRange(cacheOff, cacheOff + (ulong)chunkLen, data)
159 : m_caches[cacheIdx - 1].WriteRange(cacheOff, cacheOff + (ulong)chunkLen, data);
160
161 if ( written == 0 ) break;
162
163 totalWritten += written;
164 pos += written;
165 dataOffset += (int)written;
166
167 } else {
168
169 m_segmentTemp.CopyFrom(data, (uint)dataOffset, 0, chunkLen);
170 // Hole internen Buffer (ToArray liefert m_elements)
171 byte[] internalBuf = m_segmentTemp.ToArray();
172
173 // Schreibe den Chunk in den jeweiligen Cache
174 ulong writtenChunk = (cacheIdx == 0)
175 ? base.WriteRange(cacheOff, cacheOff + (ulong)chunkLen, internalBuf)
176 : m_caches[cacheIdx - 1].WriteRange(cacheOff, cacheOff + (ulong)chunkLen, internalBuf);
177
178 if ( writtenChunk == 0 ) break;
179
180 totalWritten += writtenChunk;
181 pos += writtenChunk;
182 dataOffset += (int)writtenChunk;
183 }
184 }
185
186 return totalWritten;
187 }
