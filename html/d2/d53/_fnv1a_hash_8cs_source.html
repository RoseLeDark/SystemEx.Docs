20namespace SystemEx.Hash {
24 public sealed class Fnv1aHash : IHash {
25 private const ulong Offset64 = 14695981039346656037UL;
26 private const ulong Prime64 = 1099511628211UL;
27
28 private const uint Prime32 = 16777619u;
29 private const uint Offset32 = 2166136261u;
37 public Hash32 Compute ( Array<byte> input, uint seed, Endian endian ) {
38 uint hash = seed == 0 ? Offset32 : seed;
39
40 for ( int i = 0 ; i < input.Count ; i++ ) {
41 hash ^= input[i];
42 hash *= Prime32;
43 }
44
45 hash ^= (hash >> 16);
46 return new Hash32(hash);
47 }
