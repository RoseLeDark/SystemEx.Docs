194 public ulong Seek(SeekOrigin org, int pos) {
195 if ( m_rawBuffer.Size == 0 ) return 0;
196
197 switch ( org ) {
198
199 case SeekOrigin.Begin:
200 m_position = (ulong)System.Math.Max(pos, 0);
201 break;
202 case SeekOrigin.Current:
203 if ( pos >= 0 )
204 m_position += (ulong)pos;
205 else {
206 ulong neg = (ulong)(-pos);
207 m_position = (m_position > neg) ? (m_position - neg) : 0;
208 }
209 break;
210 case SeekOrigin.End:
211 if ( pos >= 0 ) {
212 ulong p = (ulong)pos;
213 m_position = (p > LongLength) ? 0 : (LongLength - p);
214 } else {
215 // End + negative pos = End - (-pos) = End + pos
216 ulong add = (ulong)(-pos);
217 m_position = LongLength + add; // wird unten gekappt
218 }
219 break;
220 }
221
222 if ( m_position > LongLength )
223 m_position = LongLength;
224
225 return m_position;
226 }

237 public virtual ulong WriteRange(ulong start, ulong iend, byte[] data) {
238 if ( m_isLocked ) throw new InvalidOperationException("is Locked");
239
240 // Start ungültig?
241 if ( start < 0 || start >= (ulong)m_rawBuffer.Size )
242 return 0;
243
244 // End über Größe → kappen
245 if ( iend > (ulong)m_rawBuffer.Size )
246 iend = (ulong)m_rawBuffer.Size;
247
248 // Bereich ungültig?
249 if ( iend <= start )
250 return 0;
251
252 ulong rangeLen = iend - start;
253 ulong writable = System.Math.Min((uint)rangeLen, (uint)data.Length);
254
255 for ( ulong i = 0; i < (ulong)writable; i++ )
256 m_rawBuffer[(int)(start + i)] = data[i];
257
258 if ( (start + writable) > m_maxUsedAdress ) m_maxUsedAdress = start + writable;
259
260 return (ulong)writable;
261 }

559 public int Read(byte[] buffer, int offset, int count) {
560 if ( m_isLocked ) throw new InvalidOperationException("is Locked");
561 if ( offset < 0 || count < 0 ) throw new ArgumentOutOfRangeException("offset,count");
562 ArgumentNullException.ThrowIfNull(buffer);
563 ArgumentOutOfRangeException.ThrowIfGreaterThan(offset, buffer.Length);
564
565 if ( offset + count > buffer.Length ) {
566 // Wenn der Caller mehr verlangt als im Zielbuffer Platz ist,
567 // lesen wir nur so viel, wie in den Zielbuffer passt.
568 count = buffer.Length - offset;
569 }
570
571 // Wenn die Leseposition bereits am oder jenseits des Endes ist: EOF
572 if ( (ulong)m_rawBuffer.Size <= m_position ) return 0;
573
574 // Verfügbare Bytes im Rohpuffer ab aktueller Position
575 ulong available = (ulong)m_rawBuffer.Size - m_position;
576 int toRead = (int)System.Math.Min((ulong)count, available);
577
578 if ( toRead <= 0 ) return 0;
579
580 // ReadRange erwartet position als ulong und count als uint
581 byte[]? chunk = ReadRange(m_position, (uint)toRead);
582 if ( chunk == null || chunk.Length == 0 ) return 0;
583
584 Array.Copy(chunk, 0, buffer, offset, chunk.Length);
585
586 // interne Position vorziehen
587 m_position += (ulong)chunk.Length;
588
589 return chunk.Length;
590 }

601 public int Write(byte[] buffer, int offset, int count) {
602 if ( m_isLocked ) throw new InvalidOperationException("is Locked");
603 if ( offset < 0 || count < 0 ) throw new ArgumentOutOfRangeException("offset,count");
604
605 ArgumentNullException.ThrowIfNull(buffer);
606 ArgumentOutOfRangeException.ThrowIfGreaterThan(offset, buffer.Length);
607
608
609
610 // Adjust count if caller asked for more than fits in the target buffer
611 if ( offset + count > buffer.Length ) {
612 count = buffer.Length - offset;
613 }
614
615 // If write position is already at or beyond end, nothing to write
616 if ( (ulong)m_rawBuffer.Size <= m_position ) return 0;
617
618 // How many bytes are available in the raw buffer from current position
619 ulong available = (ulong)m_rawBuffer.Size - m_position;
620 int toWrite = (int)System.Math.Min((ulong)count, available);
621
622 if ( toWrite <= 0 ) return 0;
623
624 for ( int i = 0; i < toWrite; i++ ) {
625 m_rawBuffer[(int)(m_position + (ulong)i)] = buffer[offset + i];
626 }
627
628 // advance internal position
629 m_position += (ulong)toWrite;
630
631 if ( m_position > m_maxUsedAdress ) m_maxUsedAdress = m_position;
632
633 return toWrite;
634 }
