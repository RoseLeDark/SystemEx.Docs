223 public virtual Task WriteAsync(ICache src, CancellationToken cancellationToken = default) {
224 if ( !m_cache.CanWrite ) throw new InvalidOperationException("dest is Locked");
225 if ( !src.CanRead ) throw new InvalidOperationException("src is Locked");
226
227 if ( cancellationToken.IsCancellationRequested )
228 return Task.FromCanceled(cancellationToken);
229
230 try {
231 const int CHUNK = 64 * 1024; // 64 KiB
232 ulong total = (ulong)src.Length;
233 ulong pos = 0;
234
235 while ( pos < total ) {
236 if ( cancellationToken.IsCancellationRequested )
237 return Task.FromCanceled(cancellationToken);
238
239 int toRead = (int)System.Math.Min((ulong)CHUNK, total - pos);
240 byte[]? chunk = src.ReadRange(pos, (uint)toRead);
241
242 if ( chunk == null || chunk.Length == 0 )
243 break;
244
245 m_cache.WriteRange(pos, chunk);
246 pos += (ulong)chunk.Length;
247 }
248
249 return Task.CompletedTask;
250 } catch ( OperationCanceledException oce ) {
251 return Task.FromCanceled(oce.CancellationToken);
252 } catch ( Exception ex ) {
253 return Task.FromException(ex);
254 }
