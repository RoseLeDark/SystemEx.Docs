175 public static bool IsInteger ( FloatUE5M2 value ) {
176 if ( IsNaN(value) ) return false;
177
178 // Unterhalb vom Bias (1.0) ist nur die Null eine Ganzzahl
179 if ( value.Exponent < 15 ) return IsZero(value);
180
181 // Ab Exponent 17 (Bias + 2) schiebt sich das Komma komplett hinter die Mantisse.
182 // Ab hier ist JEDER darstellbare Wert (auch Infinity) eine Ganzzahl.
183 if ( value.Exponent >= 17 ) return !IsInfinity(value);
184
185 int shift = value.Exponent.Value - 15;
186 return (value.Mantissa & (1 >> shift)) == 0;

219 public static FloatUE5M2 Floor ( FloatUE5M2 value ) {
220 if ( IsNaN(value) || IsInfinity(value) ) return value;
221
222 byte exp = (byte)(value.Exponent);
223
224 // Werte kleiner als 1.0 werden immer zu 0 abgerundet
225 if ( exp < 15 ) return Zero;
226
227 // Ab Exponent 17 (Wert >= 4.0) ist jede darstellbare Zahl bereits eine Ganzzahl
228 if ( exp >= 17 ) return value;
229
230 // Bereich dazwischen (Exponent 15 und 16)
231 int shift = (int)(exp - 15);
232
233 // Wir erstellen eine Bitmaske für die Nachkommastellen der Mantisse
234 // Bei Exp 15 (shift = 0): Alle 2 Bits der Mantisse sind Nachkomma
235 // Bei Exp 16 (shift = 1): Das unterste Bit (Bit 0) ist Nachkomma
236 byte fractionMask = (byte)( 0x03 >> shift);
237
238 // Wenn keine Nachkommastellen gesetzt sind, ist es bereits eine Ganzzahl
239 if ( (value.Mantissa & fractionMask) == 0 ) return value;
240
241 // Nachkommastellen wegschneiden (Bits auf 0 setzen)
242 byte newMantissa = (byte)(value.Mantissa & ~fractionMask);
243 return new FloatUE5M2(0, exp, newMantissa);
244 }

249 public static FloatUE5M2 Ceil ( FloatUE5M2 value ) {
250 if ( IsNaN(value) || IsInfinity(value) ) return value;
251
252 // Werte kleiner oder gleich 0 bleiben 0
253 if ( IsZero(value) ) return Zero;
254
255 byte exp = (byte)(value.Exponent);
256
257 // Werte zwischen 0.0 (exklusive) und 1.0 werden immer zu 1.0 aufgerundet
258 if ( exp < 15 ) return One;
259
260 // Ab Exponent 17 (Wert >= 4.0) ist jede darstellbare Zahl bereits eine Ganzzahl
261 if ( exp >= 17 ) return value;
262
263 // Bereich dazwischen (Exponent 15 und 16)
264 int shift = exp - 15;
265 byte fractionMask = (byte)(0x03 >> shift);
266
267 // Wenn keine Nachkommastellen gesetzt sind, ist es bereits eine Ganzzahl
268 if ( (value.Mantissa & fractionMask) == 0 ) return value;
269
270 // Nachkommastellen wegschneiden und 1 auf den Ganzzahl-Anteil addieren
271 // Wir addieren den Wert 1 an der Stelle, wo das Komma sitzt (1 << shift)
272 byte steppedMantissa = (byte)((value.Mantissa & ~fractionMask) + (1 << shift));
273 byte finalExp = exp;
274
275 // Falls die Mantisse durch das Aufrunden überläuft (wird >= 4 bzw. Bit 2 wird gesetzt)
276 if ( (steppedMantissa & 0x04) != 0 ) {
277 steppedMantissa >>= 1;
278 finalExp++;
279 }
280
281 steppedMantissa &= 0x03; // Hidden Bit abschneiden
282
283 if ( finalExp >= 0x1F ) return PositiveInfinity;
284
285 return new FloatUE5M2(0, finalExp, steppedMantissa);
286 }

306 public static FloatUE5M2 Add ( FloatUE5M2 a, FloatUE5M2 b ) {
307 // 1. Sonderfälle (NaN und Infinity) direkt auf Bit-Ebene abfangen
308 if ( IsNaN(a) || IsNaN(b) ) return NaN;
309 if ( IsInfinity(a) ) return IsInfinity(b) && (a.Sign != b.Sign) ? NaN : a;
310 if ( IsInfinity(b) ) return b;
311 if ( IsZero(a) ) return b;
312 if ( IsZero(b) ) return a;
313
314 byte expA = a.Exponent.Value;
315 byte expB = b.Exponent.Value;
316
317 // Implizites Bit (Hidden Bit) hinzufügen: 1.0 bei normalen Zahlen, 0.0 bei subnormalen
318 byte mantA = (byte)(a.Mantissa | (expA != 0 ? 0x04 : 0x00));
319 byte mantB = (byte)(b.Mantissa | (expB != 0 ? 0x04 : 0x00));
320
321 // Exponenten anpassen (Subnormale Zahlen haben implizit Exponent 1)
322 ushort realExpA = (ushort)(expA == 0 ? 1 : expA);
323 ushort realExpB = (ushort)(expB == 0 ? 1 : expB);
324
325 Fast_Int finalExp = realExpA;
326
327 // --- Nutzung deines größeren Registers für die Berechnung ---
328 // Wir shiften die extrahierten Basismantissen nach links in das Fixpunktraster
329 ushort shiftedA = (ushort)(mantA << 4);
330 ushort shiftedB = (ushort)(mantB << 4);
331 ushort resMant = 0;
332
333 // 1. Schritt: Mantissen auf denselben Exponenten ausrichten
334 if ( realExpA >= realExpB ) {
335 var shift = realExpA - realExpB;
336 shiftedB >>= shift;
337 finalExp = realExpA;
338 } else {
339 var shift = realExpB - realExpA;
340 shiftedA >>= shift;
341 finalExp = realExpB;
342 }
343
344
345 // Gleiche Vorzeichen -> Einfach addieren
346 resMant = (ushort)(shiftedA + shiftedB);
347
348 if ( resMant == 0 ) return Zero;
349
350
351 while ( resMant >= 0x80 ) {
352 resMant >>= 1;
353 finalExp++;
354 }
355 while ( resMant < 0x40 && finalExp > 1 ) {
356 resMant <<= 1;
357 finalExp--;
358 }
359
360 resMant >>= 4;
361 resMant &= 0x03; // Hidden Bit auf Bit-Position 2 entfernen
362
363 if ( finalExp >= 0x1F ) return PositiveInfinity;
364
365 // Unterlauf zu Subnormal prüfen
366 if ( finalExp <= 0 )
367 return new FloatUE5M2(0, 0, (byte)resMant);
368
369 return new FloatUE5M2(0, (byte)finalExp, (byte)resMant);
370 }

372 public static FloatUE5M2 Mul ( FloatUE5M2 a, FloatUE5M2 b ) {
373 // 1. Sonderfälle auf Bit-Ebene prüfen
374 if ( IsNaN(a) || IsNaN(b) ) return NaN;
375
376 // Kreuzprüfung für Unendlich und Null
377 if ( IsInfinity(a) || IsInfinity(b) )
378 return (IsZero(a) || IsZero(b)) ? NaN : PositiveInfinity;
379
380 if ( IsZero(a) || IsZero(b) )
381 return Zero;
382
383 Fast_Byte expA = a.Exponent;
384 Fast_Byte expB = b.Exponent;
385
386 // Mantissen extrahieren inklusive Hidden Bit (Bit Position 2)
387 Fast_Byte mantA = (byte)(a.Mantissa | (expA != 0 ? 0x04 : 0x00));
388 Fast_Byte mantB = (byte)(b.Mantissa | (expB != 0 ? 0x04 : 0x00));
389
390 // Exponenten für subnormale Zahlen korrigieren
391 Fast_Int realExpA = expA == 0 ? 1 : expA;
392 Fast_Int realExpB = expB == 0 ? 1 : expB;
393
394 // Exponenten addieren und den E5M2-Bias (15) abziehen
395 Fast_Int finalExp = realExpA + realExpB - 15;
396
397 // --- Nutzung deines größeren Registers für die Berechnung ---
398 // Multiplikation der Mantissen (Maximalwert: 7 * 7 = 49, passt locker in Fast_UShort / Fast_UInt)
399 Fast_UShort resMant = (ushort)(mantA * mantB);
400
401 if ( resMant == 0 ) return Zero;
402
403 // Renormalisierung: Da wir zwei 3-Bit-Zahlen multipliziert haben,
404 // wandert das implizite Hidden Bit standardmäßig auf Position 4 (Wert 16 / 0x10).
405 while ( resMant >= 0x10 ) {
406 resMant >>= 1;
407 finalExp++;
408 }
409 while ( resMant < 0x04 && finalExp > 1 ) {
410 resMant <<= 1;
411 finalExp--;
412 }
413
414 // Hidden Bit auf Position 2 entfernen (Wert 4 ausmaskieren)
415 resMant &= 0x03;
416
417 // Überlauf zu Infinity prüfen
418 if ( finalExp >= 0x1F )
419 return PositiveInfinity;
420
421 // Unterlauf zu Subnormal prüfen
422 if ( finalExp <= 0 )
423 return new FloatUE5M2(0, 0, (byte)resMant);
424
425 return new FloatUE5M2(0, (byte)finalExp, (byte)resMant);
426 }

428 public static FloatUE5M2 Div ( FloatUE5M2 a, FloatUE5M2 b ) {
429 // 1. Sonderfälle auf Bit-Ebene abfangen
430 if ( IsNaN(a) || IsNaN(b) ) return NaN;
431 if ( IsZero(b) ) return IsZero(a) ? NaN : PositiveInfinity;
432 if ( IsInfinity(b) ) return IsInfinity(a) ? NaN : Zero;
433 if ( IsZero(a) || IsInfinity(a) ) return a;
434
435 Fast_Byte expA = a.Exponent;
436 Fast_Byte expB = b.Exponent;
437
438 Fast_Byte mantA = (byte)(a.Mantissa | (expA != 0 ? 0x04 : 0x00));
439 Fast_Byte mantB = (byte)(b.Mantissa | (expB != 0 ? 0x04 : 0x00));
440
441 Fast_Int realExpA = expA == 0 ? 1 : expA;
442 Fast_Int realExpB = expB == 0 ? 1 : expB;
443
444 // Exponenten subtrahieren und Bias (15) wieder aufschlagen
445 Fast_Int finalExp = realExpA - realExpB + 15;
446
447 // --- Nutzung deines größeren Registers für die Bruchrechnung ---
448 // Wir schieben den Zähler nach links, um genug "Futter" für die Ganzzahldivision zu haben
449 Fast_UInt extendedMantA = (uint)(mantA << 4);
450 Fast_UInt resMant = (uint)(extendedMantA / mantB);
451
452 if ( resMant == 0 ) return Zero;
453
454 // Renormalisierung im Shiftraster
455 while ( resMant >= 0x08 ) {
456 resMant >>= 1;
457 finalExp++;
458 }
459 while ( resMant < 0x04 && finalExp > 1 ) {
460 resMant <<= 1;
461 finalExp--;
462 }
463
464 // Hidden Bit isolieren/löschen (nur die untersten 2 Bits behalten)
465 resMant &= 0x03;
466
467 // Grenzen validieren
468 if ( finalExp >= 0x1F )
469 return PositiveInfinity;
470
471 if ( finalExp <= 0 )
472 return new FloatUE5M2(0, 0, (byte)resMant);
473
474 return new FloatUE5M2(0, (byte)finalExp, (byte)resMant);
475 }

477 public static FloatUE5M2 Sub ( FloatUE5M2 a, FloatUE5M2 b ) {
478 // 1. Sonderfälle auf Bit-Ebene abfangen
479 if ( IsNaN(a) || IsNaN(b) ) return NaN;
480
481 // Unendlich-Regeln
482 if ( IsInfinity(a) ) return IsInfinity(b) ? NaN : PositiveInfinity;
483 if ( IsInfinity(b) ) return Zero; // Sättigung: Irgendwas minus Unendlich wird zu 0 gekappt
484 if ( IsZero(b) ) return a;
485 if ( IsZero(a) ) return Zero; // Sättigung: 0 minus Irgendwas wird zu 0 gekappt
486
487 // Wenn b größer oder gleich a ist, laufen wir garantiert unter oder gegen Null
488 if ( a <= b ) return Zero;
489
490 Fast_Byte expA = a.Exponent;
491 Fast_Byte expB = b.Exponent;
492
493 // Hidden Bit hinzufügen
494 Fast_Byte mantA = (byte)(a.Mantissa | (expA != 0 ? 0x04 : 0x00));
495 Fast_Byte mantB = (byte)(b.Mantissa | (expB != 0 ? 0x04 : 0x00));
496
497 Fast_Int realExpA = expA == 0 ? 1 : expA;
498 Fast_Int realExpB = expB == 0 ? 1 : expB;
499
500 Fast_Int finalExp = realExpA;
501
502 // Nutzen deines größeren Registers, damit beim Shiften nichts verloren geht
503 Fast_UShort shiftedA = (ushort)(mantA << 4);
504 Fast_UShort shiftedB = (ushort)(mantB << 4);
505 Fast_UShort resMant = 0;
506
507 // Mantissen auf denselben Exponenten ausrichten
508 if ( realExpA >= realExpB ) {
509 var shift = realExpA - realExpB;
510 shiftedB >>= (int)shift;
511 finalExp = realExpA;
512 } else {
513 var shift = realExpB - realExpA;
514 shiftedA >>= (int)shift;
515 finalExp = realExpB;
516 }
517
518 // Da a > b garantiert ist, ist shiftedA immer größer als shiftedB
519 resMant = (ushort)(shiftedA - shiftedB);
520
521 if ( resMant == 0 ) return Zero;
522
523 // Renormalisierung im Shiftraster (Wir suchen das führende Bit)
524 while ( resMant >= 0x80 ) {
525 resMant >>= 1;
526 finalExp++;
527 }
528 while ( resMant < 0x40 && finalExp > 1 ) {
529 resMant <<= 1;
530 finalExp--;
531 }
532
533 // Zurück in das 2-Bit-Raster stutzen
534 resMant >>= 4;
535 resMant &= 0x03; // Hidden Bit löschen
536
537 // Überlauf zu Infinity prüfen
538 if ( finalExp >= 0x1F ) return PositiveInfinity;
539
540 // Unterlauf zu Subnormal prüfen
541 if ( finalExp <= 0 ) return new FloatUE5M2(0, 0, (byte)resMant);
542
543 return new FloatUE5M2(0, (byte)finalExp, (byte)resMant);
544 }
