179 public byte IsIt () {
180 ulong v = m_value;
181
182 // subtract half‑bit groups
183 v = v - ((v >> 1) & 0x5555555555555555UL);
184
185 // collapse into 2‑bit groups
186 v = (v & 0x3333333333333333UL) + ((v >> 2) & 0x3333333333333333UL);
187
188 // collapse into 4‑bit groups
189 v = (v + (v >> 4)) & 0x0F0F0F0F0F0F0F0FUL;
190
191 // collapse into 8‑bit groups
192 v = v + (v >> 8);
193
194 // collapse into 16‑bit groups
195 v = v + (v >> 16);
196
197 // collapse into 32‑bit groups
198 v = v + (v >> 32);

200 // Final result: lower 7 bits contain the popcount (0–64)
