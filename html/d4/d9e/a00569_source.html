316 private static long GreatestCommonDivisor ( long a, long b ) {
317 long temp;
318
319 if ( a > b ) {
320 temp = b;
321 b = a;
322 a = temp;
323 }
324
325 while ( b != 0 ) {
326 temp = a % b;
327 a = b;
328 b = temp;
329 }
330
331 return a;
332 }
