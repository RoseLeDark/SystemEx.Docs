101 private static void SetMessage ( uint[] buffer, FixedVector<byte> input, PaddingState s, ulong inlen, uint STATEBYTES, uint STATECOLS ) {
102
103 uint i = 0;
104 for ( i = 0 ; i < s.BytesInBlock ; i++ )
105 buffer[BYTESLICE(i, (int)STATECOLS)] = input[(int)i];
106
107 if ( s.BytesInBlock != STATEBYTES ) {
108 if ( s.FirstPaddingBlock == 1 ) {
109 buffer[BYTESLICE(i, (int)STATECOLS)] = 0x80;
110 i++;
111 }
112
113 for ( ; i < STATEBYTES ; i++ )
114 buffer[BYTESLICE(i, (int)STATECOLS)] = 0;
115
116 if ( s.Last_PaddingBlock == 1 ) {
117 ulong blocks = inlen / STATEBYTES;
118
119 blocks += (uint)((s.FirstPaddingBlock == s.Last_PaddingBlock) ? 1 : 2);
120
121 for ( int j = (int)STATEBYTES - 8 ; j < STATEBYTES ; j++ ) {
122 int pos = (int)STATEBYTES - j - 1;
123
124 buffer[BYTESLICE((uint)j, (int)STATECOLS)] = (byte)((blocks >> (8 * pos)) & 0xFF);
125 }
126 }
127 }

133 [MethodImpl(MethodImplOptions.AggressiveInlining)]
134 private static void Mul2 ( ref uint x ) {
135 uint t = x & 0x80808080u; // extract MSBs of each byte
136 x ^= t; // flip MSBs before shift
137 x <<= 1; // shift left (multiply by 2)
138 t >>= 7; // move MSBs into LSB position
139 t ^= (t << 1); // GF(2^8) reduction step
140 x ^= t; // apply reduction
141 x ^= (t << 3); // final reduction term
142 }

151 //#define BYTESLICE(i) (((i)%8)*STATECOLS+(i)/8)
152 return ( ((input) % 8) * (uint)state + (input) / 8);
153 }
154 private static void MixBytes ( uint[,] a, uint[] b, int s ) {
155 int i;
156 uint t0, t1, t2;
157
158 // b[i] = a[i][s]
159 for ( i = 0 ; i < 8 ; i++ )
160 b[i] = a[i, s];
161
162 // y_i = a_{i+6}
163 for ( i = 0 ; i < 8 ; i++ )
164 a[i, s] = b[(i + 2) % 8];
165
166 // t_i = a_i + a_{i+1}
167 for ( i = 0 ; i < 7 ; i++ )
168 b[i] ^= b[(i + 1) % 8];
169 b[7] ^= a[6, s];
170
171 // y_i = a_{i+6} + t_i
172 for ( i = 0 ; i < 8 ; i++ )
173 a[i, s] ^= b[(i + 4) % 8];
174
175 // y_i = y_i + t_{i+2}
176 for ( i = 0 ; i < 8 ; i++ )
177 a[i, s] ^= b[(i + 6) % 8];
178
179 // x_i = t_i + t_{i+3}
180 t0 = b[0];
181 t1 = b[1];
182 t2 = b[2];
183
184 for ( i = 0 ; i < 5 ; i++ )
185 b[i] ^= b[(i + 3) % 8];
186
187 b[5] ^= t0;
188 b[6] ^= t1;
189 b[7] ^= t2;
190
191 // z_i = 02 * x_i
192 for ( i = 0 ; i < 8 ; i++ )
193 Mul2(ref b[i]);
194
195 // w_i = z_i + y_{i+4}
196 for ( i = 0 ; i < 8 ; i++ )
197 b[i] ^= a[i, s];
198
199 // v_i = 02 * w_i
200 for ( i = 0 ; i < 8 ; i++ )
201 Mul2(ref b[i]);
202
203 // b_i = v_{i+3} + y_{i+4}
204 for ( i = 0 ; i < 8 ; i++ )
205 a[i, s] ^= b[(i + 3) % 8];
206 }

223 private static void Permutation ( uint[] x, int q, int state, int cols, int rR, byte[,] shift, uint[] ColumnConstant ) {
224 uint[] tmp = new uint[cols];
225 uint constant = 0;
226
227 for ( int round = 0 ; round < rR ; round++ ) {
228 constant += 0x01010101u;
229
230 // AddRoundConstant
231 if ( q == 0 ) {
232 for ( int j = 0 ; j < ColumnConstant.Length ; j++ )
233 x[j] ^= ColumnConstant[j] ^ constant;
234 } else {
235 for ( int i = 0 ; i < state ; i++ )
236 x[i] = ~x[i];
237
238 for ( int j = 0 ; j < ColumnConstant.Length ; j++ )
239 x[state - ColumnConstant.Length + j] ^= ColumnConstant[j] ^ constant;
240 }
241
242 // SubBytes + ShiftBytes
243 for ( int row = 0 ; row < 8 ; row++ ) {
244 int baseIndex = row * cols;
245
246 // Load row
247 for ( int col = 0 ; col < cols ; col++ )
248 tmp[col] = x[baseIndex + col];
249
250 // Apply S-box + shift
251 for ( int col = 0 ; col < cols ; col++ ) {
252 int idx = (col + shift[q, row]) % cols;
253 byte b = (byte)(tmp[idx] & 0xFF);
254 x[baseIndex + col] = (x[baseIndex + col] & 0xFFFFFF00u) | Ss[b];
255 }
256 }
257
258 // MixBytes
259 uint[,] mat = new uint[8, cols];
260 for ( int r = 0 ; r < 8 ; r++ )
261 for ( int c = 0 ; c < cols ; c++ )
262 mat[r, c] = x[r * cols + c];
263
264 for ( int c = 0 ; c < cols ; c++ )
265 MixBytes(mat, tmp, c);
266
267 for ( int r = 0 ; r < 8 ; r++ )
268 for ( int c = 0 ; c < cols ; c++ )
269 x[r * cols + c] = mat[r, c];
270 }
271 }

323 private static byte[] Compute_Hash ( FixedVector<byte> input, uint CRYPTO_BYTES, byte[,] shift, uint[] ColumnConstant ) {
324
325 byte ROUNDS = (byte)(CRYPTO_BYTES <= 32 ? 10 : 14);
326 byte STATEBYTES = (byte)(CRYPTO_BYTES <= 32 ? 64 : 128);
327 byte STATEWORDS = (byte)(STATEBYTES / 4 );
328 byte STATECOLS = (byte)(STATEBYTES / 8 );
329
330 uint[] ctx = new uint[STATEWORDS];
331 uint[] buffer = new uint[STATEWORDS];
332
333 ulong rlen = (ulong)input.Length;
334 PaddingState s = new PaddingState { BytesInBlock = STATEBYTES, FirstPaddingBlock = 0, Last_PaddingBlock = 0 };
335
336
337
338 // Initial value
339 for ( int i = 0 ; i < STATEWORDS ; i++ )
340 ctx[i] = 0;
341
342 ((uint[])ctx)[BYTESLICE((byte)(STATEBYTES - 2), STATECOLS)] = ((CRYPTO_BYTES * 8) >> 8);
343 ((uint[])ctx)[BYTESLICE((byte)(STATEBYTES - 1), STATECOLS)] = ((CRYPTO_BYTES * 8));
344
345 int inPos = 0;
346
347 while ( s.Last_PaddingBlock == 0 ) {
348 if ( rlen < STATEBYTES ) {
349 if ( s.FirstPaddingBlock == 0 ) {
350 s.BytesInBlock = (byte)rlen;
351 s.FirstPaddingBlock = 1;
352 s.Last_PaddingBlock = (s.BytesInBlock < STATEBYTES - 8) ? (byte)1 : (byte)0;
353 } else {
354 s.BytesInBlock = 0;
355 s.FirstPaddingBlock = 0;
356 s.Last_PaddingBlock = 1;
357 }
358 } else
359 rlen -= STATEBYTES;
360
361
362
363 // Compression
364
365 SetMessage(buffer, input, s, (ulong)input.Length, STATEBYTES, STATECOLS);
366 MemXor(buffer, ctx, STATEWORDS);
367 Permutation(buffer, 0, STATEWORDS, STATECOLS, (byte)ROUNDS, shift, ColumnConstant);
368 MemXor(ctx, buffer, STATEWORDS);
369 SetMessage(buffer, input, s, (ulong)input.Length, STATEBYTES, STATECOLS);
370 Permutation(buffer, 1, STATEWORDS, STATECOLS, (byte)ROUNDS, shift, ColumnConstant);
371 MemXor(ctx, buffer, STATEWORDS);
372
373 inPos += (int)STATEBYTES;
374 }
375
376 // Output transformation
377 for ( int i = 0 ; i < STATEWORDS ; i++ )
378 buffer[i] = ctx[i];
379
380 Permutation(buffer, 0, STATEWORDS, STATECOLS, (byte)ROUNDS, shift, ColumnConstant);
381
382 MemXor(ctx, buffer, STATEWORDS);
383
384 // Truncate
385 byte[] outx = new byte[CRYPTO_BYTES];
386 for ( uint i = STATEBYTES - CRYPTO_BYTES ; i < STATEBYTES ; i++ ) {
387 outx[i - (STATEBYTES - CRYPTO_BYTES)] = (byte)ctx[BYTESLICE(i, STATECOLS)];
388 }
389
390 return outx;
391 }
