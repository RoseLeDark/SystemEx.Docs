112 public T[] Extract ( long index ) {
113 T[] _ret = Array.Empty<T>();
114
115 long count = m_pKeys.Count;
116
117 if ( index >= 0 && index < count ) {
118 long extractCount = count - index;
119 _ret = new T[extractCount];
120
121 // copy extracted elements
122 for ( long i = 0 ; i < extractCount ; i++ ) {
123 var itm = m_pKeys.ElementAt(index + i);
124
125 if(itm.IsSome)
126 {
127
128 _ret[i] = itm.Value!;
129 }
130 }
131
132 // remove extracted elements from container
133 for ( long i = count - 1 ; i >= index ; i-- ) {
134 m_pKeys.Erase(i);
135 }
136 }
137
138 return _ret;

288 public long LowerBound ( T key ) {
289 long _ret = -1;
290
291 long lo = 0;
292 long hi = m_pKeys.Count;
293
294 while ( lo < hi ) {
295 long mid = (lo + hi) >> 1;
296 Optional<T> val = m_pKeys.ElementAt(mid);
297
298 if ( val.IsNull) {
299 lo = mid + 1;
300 continue;
301 }
302
303 // val < key → move right
304 if ( !m_compare.Compare(val.Value!, key) ) {
305 lo = mid + 1;
306 } else {
307 _ret = mid;
308 hi = mid;
309 }
310 }
