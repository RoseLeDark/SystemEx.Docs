50 public Hash32 Compute ( Vector<byte> input, uint seed) {
51 uint h1 = seed;
52
53 // 4‑Byte Blöcke
54 for ( ArrayRandomAccessIterator<byte> it = input.First ; it != input.End ; it.Forward(4) ) {
55 // Wenn weniger als 4 Bytes übrig sind → Tail
56 if ( Iterator.Distance(it.Clone(), input.End.Clone()) < 4 )
57 break;
58
59 uint k1;
60
61 byte[] bvalue = new byte[4] {
62 it.Current,
63 Iterator.Advance<byte>(it, 1).Current,
64 Iterator.Advance<byte>(it, 2).Current,
65 Iterator.Advance<byte>(it, 3).Current
66 };
67
68 k1 = bvalue.ToUInt(m_endian);
69
70 // Murmur3‑Mixing
71 k1 *= 3432918353U;
72 k1 = RotateLeft(k1, 15);
73 k1 *= 461845907U;
74
75 h1 ^= k1;
76 h1 = RotateLeft(h1, 13);
77 h1 = h1 * 5 + 0xe6546b64;
78 }
79
80 // --- TAIL ---
81 int remainder = Iterator.Distance(input.First, input.End) % 4;
82
83 if ( remainder > 0 ) {
84
85 ArrayRandomAccessIterator<byte> tailIt = input.At(input.Length - remainder);
86
87 uint k1 = 0;
88
89 switch ( remainder ) {
90 case 3:
91 k1 |= (uint)tailIt.Current;
92 k1 |= (uint)Iterator.Advance<byte>(tailIt, 1).Current << 8;
93 k1 |= (uint)Iterator.Advance<byte>(tailIt, 2).Current << 16;
94 break;
95
96 case 2:
97 k1 |= (uint)tailIt.Current;
98 k1 |= (uint)Iterator.Advance<byte>(tailIt, 1).Current << 8;
99 break;
100
101 case 1:
102 k1 |= tailIt.Current;
103 break;
104 }
105
106 k1 *= 3432918353U;
107 k1 = RotateLeft(k1, 15);
108 k1 *= 461845907U;
109
110 h1 ^= k1;
111 }
112
113 h1 ^= (uint)input.Count;
114 h1 = FMix(h1);
115
116 return new Hash32(h1);
117 }

124 public Hash64 ComputeLong ( Array<byte> input, ulong seed ) {
125 int length = input.Count;
126 ulong h = seed;
127
128 // 8‑Byte Blöcke
129 for ( ArrayRandomAccessIterator<byte> it = input.First ; it != input.End ; it.Forward(8) ) {
130
131 if ( Iterator.Distance(it, input.End) < 8 )
132 break;
133
134 ulong k = 0;
135
136 byte[] bvalue = new byte[8] {
137 it.Current,
138 Iterator.Advance<byte>(it, 1).Current,
139 Iterator.Advance<byte>(it, 2).Current,
140 Iterator.Advance<byte>(it, 3).Current,
141 Iterator.Advance<byte>(it, 4).Current,
142 Iterator.Advance<byte>(it, 5).Current,
143 Iterator.Advance<byte>(it, 6).Current,
144 Iterator.Advance<byte>(it, 7).Current
145 };
146
147 k = bvalue.ToULong(m_endian);
148
149 // Murmur‑like mixing
150 k *= 0x87c37b91114253d5UL;
151 k = RotateLeft64(k, 31);
152 k *= 0x4cf5ad432745937fUL;
153
154 h ^= k;
155 h = RotateLeft64(h, 27);
156 h = h * 5 + 0x52dce729;
157 }
158
159 int remainder = Iterator.Distance(input.End, input.First) % 8;
160
161
162 // Tail
163 if ( remainder > 0 ) {
164 ArrayRandomAccessIterator<byte> tailIt = input.At(input.Count - remainder);
165
166 ulong k = 0;
167
168 for ( int i = 0 ; i < remainder ; i++ ) {
169 byte b = Iterator.Advance<byte>(tailIt, i).Current;
170 k |= (ulong)b << (8 * i);
171 }
172
173 k *= 0x87c37b91114253d5UL;
174 k = RotateLeft64(k, 31);
175 k *= 0x4cf5ad432745937fUL;
176
177 h ^= (ulong)k;
178 }
179
180 h ^= (ulong)input.Count;
181 h = FMix64(h);
182
183 return new Hash64(h);
184 }

197 k ^= k >> 33;

200 }
