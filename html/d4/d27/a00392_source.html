122 internal DeviceSharedBuffer( DeviceBuffer cache, TDeviceSharedBackend backend, int flags, object? config) {
123
124 m_cache = cache;
125
126 if ( cache.Type == Collections.Generic.CacheType.ToDevice )
127 m_type = SharedCacheType.WriteOnly;
128 else if ( cache.Type == Collections.Generic.CacheType.FromDevice )
129 m_type = SharedCacheType.ReadOnly;
130 else if ( cache.Type == Collections.Generic.CacheType.Both )
131 m_type = SharedCacheType.ReadWrite;
132 else
133 throw new NotSupportedException("DeviceBuffer is System Only ");
134
135 m_backend = backend;
136 m_hardwareBuffer = new object();
137 }
138
147 public bool Begin() {
148 m_cache.Lock();
149
150 if ( m_type != SharedCacheType.ReadOnly ) {
151 m_backend.CreateWriteHardwareBuffer(m_cache.ToArray(), out m_hardwareBuffer);
152 } else {
153 m_backend.CreateReadHardwareBuffer((int)m_cache.Length, out m_hardwareBuffer);
154 }
155
156 return true;
157 }
158
165 public void End() {
166
167 if ( m_type != SharedCacheType.ReadOnly ) {
168 byte[] cachetmp;
169 long startPos = m_backend.ReciveFromHardwareBuffer(out cachetmp, ref m_hardwareBuffer);
170
171 m_cache.WriteRange((ulong)startPos, cachetmp);
172 } else {
173 m_backend.CloseHardwareBuffer(ref m_hardwareBuffer);
174 }
175
176 m_cache.Unlock();
177 }
178 }
